11/// A directed graph seen as undirected.
12///
13/// `out_edges` is the *whole* incidence run, anchored at the query vertex --
14/// which is exactly what `graph_adaptor.hh:199-207` does by routing to
15/// `_all_edges_out`, and what `graph_adjacency.hh:1102-1108` implements by
16/// building an `out_edge_iterator` (not an `all_edge_iterator`) over
17/// `es.begin()..es.end()`, so `make_out_edge::def` sets `src == u` for the
18/// in-half too.
19///
20/// One of the six source designs read that backwards, wired `out_edges` to a
21/// canonical all-edges iterator, and thereby made `target(e)` return the query
22/// vertex itself for half the edges. The anchored
23/// [`Incident`](crate::adj::Incident) makes that
24/// mistake unrepresentable.
25///
26/// The field is private; construct with [`Undirect::undirect`].
27#[repr(transparent)]
28#[derive(Clone, Copy, Debug)]
29pub struct Und<G: HasDir<Dir = Directed>>(G);
30
31impl<G: HasDir<Dir = Directed>> Und<G> {
32    #[inline]
33    pub(crate) const fn wrap(g: G) -> Self {
34        Und(g)
35    }
36
37    /// The underlying directed view.
38    ///
39    /// Note this is *not* an escape hatch to directed incidence on an
40    /// undirected view by accident: it is explicit at the call site, whereas
41    /// `Undirected<G>(pub G)` would let `u.0.in_edges(v)` compile silently.
42    #[inline]
43    pub fn into_directed(self) -> G {
44        self.0
45    }
46}
47
48impl<G: HasDir<Dir = Directed>> HasDir for Und<G> {
49    type Dir = Undirected;
50}

52/// Viewing a directed graph as undirected.
53///
54/// Idempotent (`Und<G>::undirect() == Und<G>`) and absorbing
55/// (`Rev<G>::undirect() == Und<G>`, since reversing an undirected graph is a
56/// no-op). Together these close the algebra at the type level.
57pub trait Undirect {
58    /// The resulting undirected view.
59    type Out: HasDir<Dir = Undirected>;
60    /// Take the undirected view.
61    fn undirect(self) -> Self::Out;
62}
63
64impl<G: HasDir<Dir = Directed>> Undirect for Und<G> {
65    type Out = Und<G>;
66    #[inline]
67    fn undirect(self) -> Und<G> {
68        self
69    }
70}

72impl<G: HasDir<Dir = Directed>> Undirect for Rev<G> {
73    type Out = Und<G>;
74    #[inline]
75    fn undirect(self) -> Und<G> {
76        Und::wrap(self.into_forward())
77    }

147impl<G: Endpoints + HasDir<Dir = Directed>> Endpoints for Und<G> {
148    #[inline]
149    fn endpoints(self, e: EdgeId) -> Option<(VertexId, VertexId)> {
150        self.0.endpoints(e)
151    }
152    #[inline]
153    fn opposite(self, e: EdgeId, v: VertexId) -> Option<VertexId> {
154        self.0.opposite(e, v)
155    }
156    /// Tries both orientations, and returns the hit **in storage orientation**.
157    ///
158    /// `edge(u, v, undirected_adaptor)` does `std::swap(res.first.s, res.first.t)`
159    /// on the reverse hit (`graph_adaptor.hh:161`), which makes the C++'s
160    /// undirected lookup disagree with its own iteration. Here iteration
161    /// yields anchored incidences and lookup yields a canonical
162    /// [`EdgeRef`], so the two cannot disagree: they answer different
163    /// questions and say so in their types.
164    fn find_edge(self, s: VertexId, t: VertexId) -> Option<EdgeRef> {
165        self.0.find_edge(s, t).or_else(|| self.0.find_edge(t, s))
166    }
167}

169/// `Arc<AdjList>` seen as undirected, by moving the handle into the wrapper.
170///
171/// This is the replacement for `std::reinterpret_pointer_cast<ug_t>(u)`
172/// (`graph_filtering.cc:92`), which produces a `shared_ptr` to an
173/// `undirected_adaptor` object that was never constructed. `Und<Arc<AdjList>>`
174/// is `#[repr(transparent)]` over `Arc<AdjList>` -- the same layout the cast
175/// produces -- but is built by a move.
176pub fn own_undirected<H: crate::adj::Lookup>(
177    g: std::sync::Arc<crate::adj::AdjList<H>>,
178) -> Und<std::sync::Arc<crate::adj::AdjList<H>>> {
179    Und::wrap(g)
180}
