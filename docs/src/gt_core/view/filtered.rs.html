1//! The filtered view.
2//!
3//! ## Type parameter, not const generic, not runtime predicate ([DESIGN](crate::design) D3)
4//!
5//! * **Type parameter** (chosen). [`KeepAll`] is a ZST, so `Filtered<G, KeepAll>`
6//!   carries only the two memoised counters and `F::TRIVIAL` const-folds the
7//!   predicate chain away. Instantiation count: 3 directedness shapes x 2
8//!   filter shapes = 6, matching `graph_filtering.hh:127` exactly.
9//! * **Const generic** `Filtered<G, const ON: bool>`: the same six, but a const
10//!   erases no *data*, so the mask fields (two fat pointers) exist in the
11//!   unfiltered arm and must be populated with dummies. Strictly worse.
12//! * **Runtime predicate** (`&dyn Fn(EdgeId) -> bool`): one instantiation
13//!   instead of six, a real compile-time win, but the inner loop gains an
14//!   unspeculatable indirect call that blocks unrolling. Rejected on hot
15//!   paths; acceptable behind the single cold dispatch boundary.
16

24/// A predicate over both identifier spaces.
25///
26/// One parameter, not two. `graph_filtering.cc:35` already collapses
27/// `_edge_filter_active || _vertex_filter_active` into a single branch.
28pub trait Filter: Copy + Send + Sync {
29    /// `true` when this filter keeps everything, letting every probe fold away.
30    const TRIVIAL: bool;
31    /// Whether `v` survives.
32    fn keep_vertex(self, v: VertexId) -> bool;
33    /// Whether `e` survives.
34    fn keep_edge(self, e: EdgeId) -> bool;
35}
36
37/// The trivial filter. A ZST.
38#[derive(Clone, Copy, Default, Debug)]
39pub struct KeepAll;
40
41impl Filter for KeepAll {
42    const TRIVIAL: bool = true;
43    #[inline(always)]
44    fn keep_vertex(self, v: VertexId) -> bool {
45        true
46    }
47    #[inline(always)]
48    fn keep_edge(self, e: EdgeId) -> bool {
49        true
50    }
51}

53/// A filter backed by two `uint8` masks, as graph-tool's `MaskFilter`
54/// (`graph_filtering.hh:42`) is.
55///
56/// The borrow is load-bearing: while any [`Filtered`] holding this exists, the
57/// mask storage cannot be resized or replaced. graph-tool instead relies on
58/// the programmer remembering `_gviews.clear()` in `set_vertex_filter_property`
59/// (`graph_filtering.cc:179`) and `set_edge_filter_property` (`:196`).
60#[derive(Clone, Copy, Debug)]
61pub struct MaskFilter<'m> {
62    vmask: &'m [u8],
63    emask: &'m [u8],
64}
65
66impl Filter for MaskFilter<'_> {
67    const TRIVIAL: bool = false;
68
69    /// Non-zero keeps, exactly as `set_vertex_filter` documents ("only the
70    /// vertices with value different than `False` are kept",
71    /// `graph_tool/__init__.py:3505-3507`).
72    ///
73    /// An out-of-range probe answers *filtered out*. `MaskFilter::operator()`
74    /// is `boost::get(_filtered_property, d)` (`graph_filtering.hh:51-56`) on
75    /// an **unchecked** map, whose `operator[]` is an unguarded `_data[i]`
76    /// (`fast_vector_property_map.hh:218-221`, defect #11): a mask shorter
77    /// than the index space reads past the end. That case is unreachable
78    /// through [`Filtered::masked`], which refuses a short mask; this clause
79    /// is what makes it unreachable through the one remaining door -- a
80    /// `MaskFilter` copied out of [`Filtered::filter`] and handed to
81    /// [`Filtered::new`] on a *different*, larger graph. There it
82    /// under-filters, deterministically, instead of reading out of bounds,
83    /// and the memoised counts still agree with iteration because both go
84    /// through this predicate.
85    #[inline]
86    fn keep_vertex(self, v: VertexId) -> bool {
87        self.vmask.get(v.index()).is_some_and(|&keep| keep != 0)
88    }
89
90    /// Non-zero keeps. Out of range is filtered out; see
91    /// [`keep_vertex`](Self::keep_vertex).
92    #[inline]
93    fn keep_edge(self, e: EdgeId) -> bool {
94        self.emask.get(e.index()).is_some_and(|&keep| keep != 0)
95    }
96}

98/// A view restricted to the vertices and edges a [`Filter`] keeps.
99///
100/// Both counts are memoised at construction from [`EdgeList::edges`] and
101/// [`VertexList::vertices`], using the *same* predicate the iterators use, so
102/// `num_edges() == edges().count()` holds. Computing the edge count by summing
103/// out-degrees instead is how a filtered undirected view comes to report three
104/// edges where the answer is one.
105#[derive(Clone, Copy, Debug)]
106pub struct Filtered<G, F> {
107    inner: G,
108    filter: F,
109    n_vertices: usize,
110    n_edges: usize,
111}
112
113/// Whether an edge survives, endpoints included. The single predicate shared
114/// by [`Filtered::new`] and every filtered iterator.
115#[inline]
116pub(crate) fn keeps_edge<F: Filter>(f: F, e: EdgeRef) -> bool {
117    F::TRIVIAL || (f.keep_edge(e.id()) && f.keep_vertex(e.source()) && f.keep_vertex(e.target()))
118}
119
120/// Whether an incidence survives, given that its anchor already survives.
121#[inline]
122pub(crate) fn keeps_incident<F: Filter>(f: F, i: Incident) -> bool {
123    F::TRIVIAL || (f.keep_edge(i.edge) && f.keep_vertex(i.other))
124}

126impl<G: GraphRef + VertexList + EdgeList, F: Filter> Filtered<G, F> {
127    /// Build a filtered view, memoising both honest counts.
128    ///
129    /// Short-circuits on `F::TRIVIAL`, so the unfiltered arm pays no prepass.
130    /// Otherwise O(V + E), which is the price of an honest `num_vertices`;
131    /// `filt_graph` gets O(1) by returning the wrong number.
132    pub fn new(inner: G, filter: F) -> Self {
133        if F::TRIVIAL {
134            return Filtered {
135                n_vertices: inner.num_vertices(),
136                n_edges: inner.num_edges(),
137                inner,
138                filter,
139            };
140        }
141        // `keeps_edge` here and in `FilterEdges::next` are the *same*
142        // function, which is the whole of defect #16: `num_edges` counted by
143        // one rule and `edges()` yielded by another is how
144        // `distance(ei, eiend) != num_edges(g)` (`graph_filtered.hh:301-312`)
145        // happens. Summing degrees instead would be a third rule, and on an
146        // undirected view a wrong one.
147        let n_vertices = inner.vertices().filter(|&v| filter.keep_vertex(v)).count();
148        let n_edges = inner.edges().filter(|&e| keeps_edge(filter, e)).count();
149        Filtered {
150            inner,
151            filter,
152            n_vertices,
153            n_edges,
154        }
155    }
156}

171impl<'m, G: GraphRef + VertexList + EdgeList> Filtered<G, MaskFilter<'m>> {
172    /// Build a mask-filtered view, validating the masks against *this graph's*
173    /// own bounds.
174    ///
175    /// Validation and use are one statement. A free-standing
176    /// `MaskFilter::new(vmask, emask, vbound, ebound)` is `Copy` and remembers
177    /// nothing, so a mask admitted for a 3-vertex graph is silently accepted
178    /// by a 10-vertex one and panics on first probe -- which is the same
179    /// structural defect as `graph_filtering.cc:42-46` reserving separately
180    /// from `MaskFilter::operator()`.
181    pub fn masked(inner: G, vmask: &'m [u8], emask: &'m [u8]) -> Result<Self, PropError> {
182        // Against the *bound*, never against `num_vertices`: the mask is
183        // indexed by the unfiltered index space, which is exactly the
184        // distinction `graph_filtering.cc:42-46` gets right
185        // (`get_edge_index_range()`, `num_vertices(*u)`) and
186        // `graph_copy.cc:66-73` gets wrong.
187        let vb = inner.vertex_bound();
188        if vmask.len() < vb.len() {
189            return Err(PropError::ShortMask {
190                have: vmask.len(),
191                need: vb.len(),
192            });
193        }
194        let eb = inner.edge_bound();
195        if emask.len() < eb.len() {
196            return Err(PropError::ShortMask {
197                have: emask.len(),
198                need: eb.len(),
199            });
200        }
201        Ok(Filtered::new(inner, MaskFilter { vmask, emask }))
202    }
203}

250impl<G: GraphRef, F: Filter> GraphRef for Filtered<G, F> {
251    type Out = FilterIncident<G::Out, F>;
252    type All = FilterIncident<G::All, F>;
253    #[inline]
254    fn out_edges(self, v: VertexId) -> Self::Out {
255        FilterIncident {
256            inner: self.inner.out_edges(v),
257            filter: self.filter,
258        }
259    }
260    #[inline]
261    fn all_edges(self, v: VertexId) -> Self::All {
262        FilterIncident {
263            inner: self.inner.all_edges(v),
264            filter: self.filter,
265        }
266    }
267    /// O(deg), by probing. `out_degree(u, filt_graph)` counts the filtered
268    /// range (`graph_filtered.hh:383-392`) rather than forwarding, and so does
269    /// this. The anchor `v` itself is *not* probed, matching
270    /// `out_edges(u, g)`, which applies `out_edge_pred` (edge and *target*)
271    /// and never tests `u`: the caller reached `v` through `vertices()`, which
272    /// already filtered it, and this is what keeps
273    /// `sum(out_degree) == num_edges` true over a directed filtered view.
274    #[inline]
275    fn out_degree(self, v: VertexId) -> usize {
276        if F::TRIVIAL {
277            return self.inner.out_degree(v);
278        }
279        self.out_edges(v).count()
280    }
281    /// O(deg), by probing the whole incidence run.
282    ///
283    /// Counted from [`all_edges`](GraphRef::all_edges), not as
284    /// `in_degree + out_degree` the way `degree(u, filt_graph)`
285    /// (`graph_filtered.hh:395-399`) is: that form needs a `Bidirectional`
286    /// underneath, so on an undirected view it reaches
287    /// `in_edges(u, undirected_adaptor)` -- the default-constructed empty
288    /// range of `graph_adaptor.hh:219-227`, defect #12 -- and silently halves
289    /// the answer.
290    #[inline]
291    fn degree(self, v: VertexId) -> usize {
292        if F::TRIVIAL {
293            return self.inner.degree(v);
294        }
295        self.all_edges(v).count()
296    }
297}

343impl<G: Endpoints + GraphRef, F: Filter> Endpoints for Filtered<G, F> {
344    /// `None` when the edge or either endpoint is filtered out.
345    ///
346    /// `source(e, filt_graph)` and `target(e, filt_graph)` forward to the
347    /// unfiltered graph unconditionally (`graph_filtered.hh:341-355`), so a
348    /// descriptor that the view's own `edges()` never yields still resolves to
349    /// a pair of vertices its own `vertices()` never yields. The predicate
350    /// here is `keeps_edge` -- the same one `edges()` and `num_edges()` use.
351    #[inline]
352    fn endpoints(self, e: EdgeId) -> Option<(VertexId, VertexId)> {
353        let (s, t) = self.inner.endpoints(e)?;
354        keeps_edge(self.filter, EdgeRef::new(e, s, t)).then_some((s, t))
355    }
356    #[inline]
357    fn opposite(self, e: EdgeId, v: VertexId) -> Option<VertexId> {
358        let (s, t) = self.endpoints(e)?;
359        EdgeRef::new(e, s, t).opposite(v)
360    }
361    /// The first *surviving* `s -> t` edge, in this view's orientation.
362    ///
363    /// The scan is the point. `inner.find_edge(s, t).filter(..)` would answer
364    /// `None` whenever the first parallel edge is filtered out and a later one
365    /// survives -- a lookup disagreeing with the view's own `out_edges(s)`,
366    /// which is defect #16 in miniature. `edge(u, v, filt_graph)` scans too
367    /// (`graph_filtered.hh:502-540` drives `edge_range_iter`), so this is also
368    /// what the C++ does; what the C++ then omits is the vertex predicate,
369    /// testing only `g._edge_pred(e)` (`:529`), so its `edge()` returns edges
370    /// whose endpoints its `vertices()` has removed. Here one predicate
371    /// serves both.
372    ///
373    /// O(deg) when filtering is on, O(1)-or-`Lookup` when it is not.
374    fn find_edge(self, s: VertexId, t: VertexId) -> Option<EdgeRef> {
375        if F::TRIVIAL {
376            return self.inner.find_edge(s, t);
377        }
378        if !self.filter.keep_vertex(s) {
379            return None;
380        }
381        let f = self.filter;
382        let hit = self
383            .inner
384            .out_edges(s)
385            .find(|&i| i.other == t && keeps_incident(f, i))?;
386        // Orientation comes from the inner view, so a reversed view reports
387        // reversed endpoints and an undirected one reports storage order --
388        // each exactly as its own unfiltered `find_edge` would.
389        let (es, et) = self.inner.endpoints(hit.edge)?;
390        Some(EdgeRef::new(hit.edge, es, et))
391    }
392}

398impl<I: Iterator<Item = Incident>, F: Filter> Iterator for FilterIncident<I, F> {
399    type Item = Incident;
400    /// `Iterator::find` rather than a hand-rolled `while let`, so the probe
401    /// runs inside the inner iterator's own `try_fold` and keeps the
402    /// unrolled slice loop that `IncidentIter::fold` exists to expose.
403    #[inline]
404    fn next(&mut self) -> Option<Incident> {
405        if F::TRIVIAL {
406            return self.inner.next();
407        }
408        let f = self.filter;
409        self.inner.find(|&i| keeps_incident(f, i))
410    }
411    #[inline]
412    fn size_hint(&self) -> (usize, Option<usize>) {
413        (0, self.inner.size_hint().1)
414    }
415    #[inline]
416    fn fold<B, Fun: FnMut(B, Incident) -> B>(self, init: B, mut fun: Fun) -> B {
417        if F::TRIVIAL {
418            return self.inner.fold(init, fun);
419        }
420        let f = self.filter;
421        self.inner.fold(init, move |acc, i| {
422            if keeps_incident(f, i) {
423                fun(acc, i)
424            } else {
425                acc
426            }
427        })
428    }
429}

439impl<I: Iterator<Item = EdgeRef>, F: Filter> Iterator for FilterEdges<I, F> {
440    type Item = EdgeRef;
441    /// `keeps_edge`, the same predicate [`Filtered::new`] counted with --
442    /// which is what makes `num_edges() == edges().count()` a fact rather
443    /// than a hope.
444    #[inline]
445    fn next(&mut self) -> Option<EdgeRef> {
446        if F::TRIVIAL {
447            return self.inner.next();
448        }
449        let f = self.filter;
450        self.inner.find(|&e| keeps_edge(f, e))
451    }
452    #[inline]
453    fn size_hint(&self) -> (usize, Option<usize>) {
454        (0, self.inner.size_hint().1)
455    }
456    #[inline]
457    fn fold<B, Fun: FnMut(B, EdgeRef) -> B>(self, init: B, mut fun: Fun) -> B {
458        if F::TRIVIAL {
459            return self.inner.fold(init, fun);
460        }
461        let f = self.filter;
462        self.inner.fold(
463            init,
464            move |acc, e| {
465                if keeps_edge(f, e) { fun(acc, e) } else { acc }
466            },
467        )
468    }
469}
