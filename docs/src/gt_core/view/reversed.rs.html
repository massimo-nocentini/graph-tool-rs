9/// A directed graph traversed against its edges.
10///
11/// `out_edges` is the underlying graph's **in-edges only**. This is what
12/// `graph_reverse.hh:78-80` says: `reversed_graph`'s `graph_traits`
13/// specialisation swaps the `out_edge_iterator` and `in_edge_iterator`
14/// typedefs. A design that yielded the union of both halves would be an
15/// undirected view wearing a directed type.
16///
17/// The field is private; construct with [`Reverse::reverse`].
18#[repr(transparent)]
19#[derive(Clone, Copy, Debug)]
20pub struct Rev<G: HasDir<Dir = Directed>>(G);
21
22impl<G: HasDir<Dir = Directed>> Rev<G> {
23    #[inline]
24    pub(crate) const fn wrap(g: G) -> Self {
25        Rev(g)
26    }
27
28    /// The underlying forward view.
29    #[inline]
30    pub fn into_forward(self) -> G {
31        self.0
32    }
33}

39/// Reversing a directed view. An involution: `g.reverse().reverse()` is `g`,
40/// as a *type*, not merely as a value.
41///
42/// There is no implementation for [`Und`](super::Und), so reversing an
43/// undirected view is `error[E0599]` rather than the silent no-op that
44/// `graph_filtering.hh:77`'s `reversed && is_directed_v<directed_t>` guard
45/// produces.
46pub trait Reverse {
47    /// The resulting reversed view.
48    type Out: HasDir<Dir = Directed>;
49    /// Take the reversed view.
50    fn reverse(self) -> Self::Out;
51}
52
53impl<G: HasDir<Dir = Directed>> Reverse for Rev<G> {
54    type Out = G;
55    #[inline]
56    fn reverse(self) -> G {
57        self.0
58    }
59}
