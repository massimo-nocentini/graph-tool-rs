1//! Identifiers.
2//!
3//! ## The width decision ([DESIGN](crate::design) D5)
4//!
5//! graph-tool instantiates `adj_list<size_t>` and nothing else
6//! (`src/graph/graph.hh:137`), so an adjacency entry costs 16 bytes and a
7//! 64-byte cache line holds four. It then stores adjacency *positions* as
8//! `pair<uint32_t,uint32_t>` (`graph_adjacency.hh:620`) under that `size_t`
9//! vertex, so the two widths already disagree and nothing asserts it.
10//!
11//! Here the width is **one crate-wide type alias**, not a type parameter.
12//! [`Raw`] is `u32` by default (8 entries per line, 2x graph-tool on the
13//! dominant memory stream) and `u64` under the `wide-index` feature. Making it
14//! an alias rather than a parameter removes an entire monomorphisation axis
15//! from every signature in the port -- see the instantiation budget in
16//! [DESIGN](crate::design) section 10.
17
18use std::fmt;
19use std::hash::{Hash, Hasher};
20use std::marker::PhantomData;
21use std::num::NonZeroU64;
22
23/// The raw integer width of every vertex and edge identifier.
24#[cfg(not(feature = "wide-index"))]
25pub type Raw = u32;
26/// The raw integer width of every vertex and edge identifier.
27#[cfg(feature = "wide-index")]
28pub type Raw = u64;
29
30/// Largest representable index. One value is reserved so that a future
31/// niche-carrying id stays free.
32pub const MAX_INDEX: usize = (Raw::MAX - 1) as usize;
33
34/// Marker distinguishing one identifier space from another.
35///
36/// Replaces `adj_edge_descriptor`'s `Vertex s, t, idx` (`graph_adjacency.hh:211`),
37/// where all three fields are the same type and are therefore mutually
38/// substitutable at every call site.
39pub trait IdTag: Copy + Eq + Ord + Hash + fmt::Debug + Send + Sync + 'static {
40    /// Human-readable name, used in error messages.
41    const NAME: &'static str;
42}
43
44/// Tag for the vertex index space.
45#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
46pub struct VertexTag;
47impl IdTag for VertexTag {
48    const NAME: &'static str = "vertex";
49}
50
51/// Tag for the edge index space.
52#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
53pub struct EdgeTag;
54impl IdTag for EdgeTag {
55    const NAME: &'static str = "edge";
56}
57
58/// A tagged index into one identifier space.
59///
60/// `PhantomData<fn() -> T>` keeps `Id` unconditionally `Copy + Send + Sync`
61/// and covariant without bounding `T`.
62#[repr(transparent)]
63pub struct Id<T: IdTag>(Raw, PhantomData<fn() -> T>);
64
65/// A vertex identifier.
66pub type VertexId = Id<VertexTag>;
67/// An edge identifier. This, and never an [`adj::EdgeRef`](crate::adj::EdgeRef),
68/// is an edge's identity.
69pub type EdgeId = Id<EdgeTag>;
70
71const _: () = assert!(size_of::<VertexId>() == size_of::<Raw>());
72const _: () = assert!(size_of::<EdgeId>() == size_of::<Raw>());
73
74impl<T: IdTag> Id<T> {
75    /// Construct from a raw index, or `None` if it exceeds [`MAX_INDEX`].
76    #[inline]
77    pub const fn new(i: usize) -> Option<Self> {
78        if i <= MAX_INDEX {
79            Some(Id(i as Raw, PhantomData))
80        } else {
81            None
82        }
83    }
84
85    /// Construct from a raw index, panicking if out of range.
86    ///
87    /// For call sites that have already proved the bound (an iterator over
88    /// `0..n`, a value read back out of the same structure).
89    #[inline]
90    pub const fn from_index(i: usize) -> Self {
91        match Self::new(i) {
92            Some(id) => id,
93            None => panic!("index exceeds the identifier width"),
94        }
95    }
96
97    /// The index, for use as a slice offset.
98    #[inline]
99    pub const fn index(self) -> usize {
100        self.0 as usize
101    }
102
103    /// The raw representation.
104    #[inline]
105    pub const fn raw(self) -> Raw {
106        self.0
107    }
108}
109
110// Hand-written so that no `T: Clone`-style bound leaks into public signatures.
111impl<T: IdTag> Clone for Id<T> {
112    #[inline]
113    fn clone(&self) -> Self {
114        *self
115    }
116}
117impl<T: IdTag> Copy for Id<T> {}
118impl<T: IdTag> PartialEq for Id<T> {
119    #[inline]
120    fn eq(&self, other: &Self) -> bool {
121        self.0 == other.0
122    }
123}
124impl<T: IdTag> Eq for Id<T> {}
125impl<T: IdTag> PartialOrd for Id<T> {
126    #[inline]
127    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
128        Some(self.cmp(other))
129    }
130}
131impl<T: IdTag> Ord for Id<T> {
132    #[inline]
133    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
134        self.0.cmp(&other.0)
135    }
136}
137impl<T: IdTag> Hash for Id<T> {
138    #[inline]
139    fn hash<H: Hasher>(&self, state: &mut H) {
140        self.0.hash(state);
141    }
142}
143impl<T: IdTag> fmt::Debug for Id<T> {
144    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
145        write!(f, "{}#{}", T::NAME, self.0)
146    }
147}

149/// Process-unique identity of one graph.
150///
151/// Carried by [`crate::bound::Bound`] and by [`crate::prop::DenseProp`] so that
152/// sizing a map from graph A and indexing it with graph B's descriptors is
153/// caught once, at kernel entry, by a single comparison. graph-tool has no
154/// analogue: vertex descriptors are bare `size_t` and the confusion is the
155/// mechanism behind the `copy_property` out-of-bounds write
156/// (`graph_copy.cc:66-73` sizing from a *filtered* count then writing at
157/// *unfiltered* indices).
158#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug)]
159pub struct GraphId(NonZeroU64);
160
161impl GraphId {
162    /// Mint a fresh, process-unique identity.
163    pub fn fresh() -> Self {
164        static N: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(1);
165        let v = N.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
166        GraphId(NonZeroU64::new(v).expect("counter starts at 1"))
167    }
168
169    /// The raw value, for diagnostics only.
170    #[inline]
171    pub const fn get(self) -> u64 {
172        self.0.get()
173    }
174}
