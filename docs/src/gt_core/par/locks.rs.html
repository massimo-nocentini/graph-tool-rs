1//! Row locking for concurrent commits.
2//!
3//! ## Why this exists at all ([DESIGN](crate::design) D9)
4//!
5//! Three of the six source designs independently removed graph-tool's
6//! concurrent mutation and each scored the removal as a win. It is not a win.
7//! `blockmodel/state.hh:152` builds `_group_mutex`, `:343-348` takes ordered
8//! pair row locks via `do_lock<lock_t::shared>`, and `:451` sizes
9//! `_m_entries_pool` to `get_num_threads()` -- the library commits moves from
10//! several threads, and the scratch pool exists precisely for that.
11//!
12//! What Rust adds is not a prohibition but an enforcement: `partition.hh:78-84`
13//! mixes a plain `_count[r] += w` between two `#pragma omp atomic` statements,
14//! correct only if every caller holds `_group_mutex[r]` -- a convention, not a
15//! type. Here the row can only be reached through its guard.
16
17use std::sync::{Mutex, MutexGuard, PoisonError};
18
19/// Two distinct rows, or one row named twice.
20///
21/// The variants matter. Returning `None` for `r == s` forces every caller to
22/// hand-write a diagonal branch, and the diagonal is not an error case: it is
23/// the block-graph self-loop `_mrs[r][r]`, which is common. `do_ulock_pair`
24/// (`parallel_util.hh:247-256`) handles it transparently by locking once.
25pub enum Pair<'a, T> {
26    /// `r` and `s` were distinct; the references are in **caller order**.
27    Two(&'a mut T, &'a mut T),
28    /// `r == s`; one reference.
29    Same(&'a mut T),
30}
31
32/// Borrow two rows of a slice mutably, in caller order.
33///
34/// Caller order is the whole point. A version returning `[&mut T; 2]` sorted
35/// by index silently transposes: `pair_mut(xs, Group(3), Group(1))` hands back
36/// `[&mut xs[1], &mut xs[3]]`, so a caller writing `pr` and `ps` positionally
37/// updates the wrong group. graph-tool does not have that bug -- it sorts only
38/// for `std::lock`'s deadlock avoidance and keeps `r` and `s` by name -- and a
39/// port must not introduce it.
40///
41/// Returns `None` if either index is out of range, `Same` if `r == s`.
42pub fn pair_mut<T>(xs: &mut [T], r: usize, s: usize) -> Option<Pair<'_, T>> {
43    // The diagonal first: `get_disjoint_mut` rejects `[r, r]` as overlapping,
44    // and `r == s` is not an error here -- it is `_mrs[r][r]`.
45    if r == s {
46        return xs.get_mut(r).map(Pair::Same);
47    }
48
49    // Sorted only to mirror `std::lock`'s deadlock-avoidance discipline in
50    // `do_ulock_pair`; the pair is then re-normalised to caller order, which
51    // is the part `do_ulock_pair` keeps and a sorted-array API loses. The
52    // `Err` covers both out-of-range indices; overlap is impossible here.
53    let (lo, hi) = if r < s { (r, s) } else { (s, r) };
54    let [a, b] = xs.get_disjoint_mut([lo, hi]).ok()?;
55
56    Some(if r < s {
57        Pair::Two(a, b)
58    } else {
59        Pair::Two(b, a)
60    })
61}

63/// Per-row mutexes, acquired in `(min, max)` order.
64///
65/// The port of `_group_mutex` (`blockmodel/state.hh:152`).
66///
67/// ## Two deliberate divergences from the C++
68///
69/// * `do_lock` (`parallel_util.hh:84-86`) calls `std::lock`, whose order is
70///   an unspecified try-lock dance. A fixed `(min, max)` order is equally
71///   deadlock-free and additionally *deterministic*, which is the property
72///   the rest of `par` is built around.
73/// * A `std::mutex` has no poison state. The guarded datum here is `()`, so
74///   there is no invariant a panicking closure can leave broken; a poisoned
75///   row is recovered rather than propagated, and the panic reaches the
76///   caller by itself. Otherwise one panicking commit would wedge its two
77///   groups for the rest of the process.
78pub struct RowLocks {
79    rows: Box<[Mutex<()>]>,
80}
81
82/// Acquire one row, ignoring poison. See the note on [`RowLocks`].
83#[inline]
84fn acquire(m: &Mutex<()>) -> MutexGuard<'_, ()> {
85    m.lock().unwrap_or_else(PoisonError::into_inner)
86}
87
88impl RowLocks {
89    /// One mutex per row.
90    pub fn new(n: usize) -> Self {
91        RowLocks {
92            rows: (0..n).map(|_| Mutex::new(())).collect(),
93        }
94    }
95
96    /// Number of rows.
97    #[inline]
98    pub fn len(&self) -> usize {
99        self.rows.len()
100    }
101    /// Whether there are no rows.
102    #[inline]
103    pub fn is_empty(&self) -> bool {
104        self.rows.is_empty()
105    }
106
107    /// Run `f` holding the locks for `r` and `s`, taken in `(min, max)` order
108    /// so that two commits can never deadlock against each other.
109    ///
110    /// `r == s` locks **once**: `std::sync::Mutex` is not re-entrant, so the
111    /// transparent diagonal of `do_ulock_pair` (`parallel_util.hh:247-256`) is
112    /// not a convenience here, it is the difference between working and
113    /// self-deadlocking.
114    ///
115    /// # Panics
116    ///
117    /// If `r` or `s` is not a row, matching `_group_mutex[r]`.
118    pub fn with_pair<R>(&self, r: usize, s: usize, f: impl FnOnce() -> R) -> R {
119        if r == s {
120            return self.with_row(r, f);
121        }
122        let (lo, hi) = if r < s { (r, s) } else { (s, r) };
123        // Both bounds are checked before either mutex is taken, so an
124        // out-of-range `hi` cannot panic with `lo` held.
125        let (mlo, mhi) = (&self.rows[lo], &self.rows[hi]);
126        let _glo = acquire(mlo);
127        let _ghi = acquire(mhi);
128        f()
129    }
130
131    /// Run `f` holding one row's lock.
132    ///
133    /// # Panics
134    ///
135    /// If `r` is not a row.
136    pub fn with_row<R>(&self, r: usize, f: impl FnOnce() -> R) -> R {
137        let _g = acquire(&self.rows[r]);
138        f()
139    }
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    /// D9's named case, by value rather than by index.
147    #[test]
148    fn pair_mut_is_in_caller_order() {
149        let mut xs = [10, 11, 12, 13];
150        match pair_mut(&mut xs, 3, 1) {
151            Some(Pair::Two(a, b)) => {
152                assert_eq!(*a, 13, "first reference must be r's row");
153                assert_eq!(*b, 11, "second reference must be s's row");
154            }
155            _ => panic!("expected Two"),
156        }
157    }
158
159    #[test]
160    fn pair_mut_diagonal_is_same() {
161        let mut xs = [10, 11, 12, 13];
162        match pair_mut(&mut xs, 2, 2) {
163            Some(Pair::Same(a)) => assert_eq!(*a, 12),
164            _ => panic!("expected Same"),
165        }
166    }
167
168    #[test]
169    fn pair_mut_out_of_range_is_none() {
170        let mut xs = [10, 11, 12, 13];
171        assert!(pair_mut(&mut xs, 4, 1).is_none());
172        assert!(pair_mut(&mut xs, 1, 4).is_none());
173        assert!(pair_mut(&mut xs, 4, 4).is_none());
174        let empty: &mut [i32] = &mut [];
175        assert!(pair_mut(empty, 0, 0).is_none());
176    }
177
178    #[test]
179    fn row_locks_are_reentrant_on_the_diagonal() {
180        let l = RowLocks::new(4);
181        assert_eq!(l.len(), 4);
182        assert!(!l.is_empty());
183        assert_eq!(l.with_pair(2, 2, || 7), 7);
184    }
