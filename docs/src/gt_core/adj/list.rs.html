14/// A graph: contiguous adjacency, dense edge indices, O(1) endpoint lookup.
15///
16/// `H` selects whether an `(s,t)` hash index is maintained. It is the only
17/// type parameter; the slot table is unconditional (see
18/// the private `index` module) and the index width is the crate-wide
19/// [`Raw`](crate::ids::Raw) alias rather than a parameter.
20///
21/// `AdjList` is *not* directed or undirected. graph-tool's base type is not
22/// either -- `graph_filtering.hh:70-73` builds `directed_t` by wrapping
23/// `adj_list<size_t>` in an adaptor. Directedness is a property of the
24/// [`view`](crate::view), and `&AdjList` is the directed one.
25#[derive(Clone, Debug)]
26pub struct AdjList<H: Lookup = NoLookup> {
27    blocks: Vec<Block>,
28    ids: EdgeIds,
29    slots: EdgeSlots,
30    lookup: H,
31    id: GraphId,
32    /// Reused across `clear_vertex`/`remove_vertex` so that a steady-state
33    /// sweep allocates nothing. graph-tool's non-epos `clear_vertex` is an
34    /// in-place `remove_if` with zero allocations; a per-call `Vec` of doomed
35    /// descriptors costs two allocations per call on a path this library runs
36    /// in loops.
37    scratch: Vec<EdgeId>,
38}
39
40impl AdjList<NoLookup> {
41    /// An empty graph.
42    pub fn new() -> Self {
43        AdjList::with_lookup(NoLookup)
44    }
45
46    /// An empty graph with `n` isolated vertices.
47    ///
48    /// One `Vec` allocation, not `n` calls to
49    /// [`add_vertex`](Self::add_vertex): `add_vertex(g, n)`
50    /// (`graph_adjacency.hh:1317-1333`) special-cases the bulk form the same
51    /// way, `_edges.resize(v + n)`.
52    ///
53    /// # Panics
54    ///
55    /// If `n` exceeds the vertex index space, i.e. `n - 1 > MAX_INDEX`. The
56    /// C++ has no bound at all: `Vertex` is also the index type and
57    /// `null_vertex()` is `numeric_limits<Vertex>::max()` (`:924`), so a graph
58    /// grown past that silently collides with the null descriptor.
59    pub fn with_vertices(n: usize) -> Self {
60        AdjList::with_vertices_and_lookup(n, NoLookup)
61    }
62}
63
64impl Default for AdjList<NoLookup> {
65    fn default() -> Self {
66        Self::new()
67    }

83    /// An empty graph with `n` isolated vertices and the given `(s,t)` index.
84    ///
85    /// The generic form of [`AdjList::with_vertices`]; `pub(crate)` because
86    /// the public constructor's `AdjList::with_vertices(n)` would otherwise
87    /// stop inferring `H = NoLookup` at every call site. `ParBuilder::build`
88    /// (U6) is the intended caller.
89    pub(crate) fn with_vertices_and_lookup(n: usize, lookup: H) -> Self {
90        assert!(
91            n == 0 || n - 1 <= MAX_INDEX,
92            "vertex index space exhausted (max {MAX_INDEX})"
93        );
94        let mut g = AdjList::with_lookup(lookup);
95        g.blocks = vec![Block::new(); n];
96        g
97    }

210    /// The first edge from `s` to `t`, if any.
211    ///
212    /// Returns `Option`, not a `{max,max,max}` descriptor paired with a `bool`
213    /// that callers may drop (`graph_adjacency.hh:943, :949, :962`). O(1) with
214    /// [`EHash`](super::EHash), otherwise a scan of the shorter half.
215    pub fn find_edge(&self, s: VertexId, t: VertexId) -> Option<EdgeRef> {
216        let sb = self.blocks.get(s.index())?;
217        let tb = self.blocks.get(t.index())?;
218        if H::ENABLED {
219            // `iter->second.front()` (`graph_adjacency.hh:949`): the bucket's
220            // first id, which is the oldest surviving parallel edge.
221            return self
222                .lookup
223                .find(s, t)
224                .first()
225                .map(|&id| EdgeRef::new(id, s, t));
226        }
227        // `:952-971`: scan whichever half is shorter. The two arms answer the
228        // same question, so they must agree; they do here because the entry's
229        // `other` is the *other* endpoint in either half (D2), where the C++
230        // rebuilds an `edge_descriptor(s, t, idx)` per arm from the arguments
231        // and could not disagree even if storage did.
232        let hit = if sb.out_degree() < tb.in_degree() {
233            sb.out().iter().find(|e| e.other == t)
234        } else {
235            tb.inc().iter().find(|e| e.other == s)
236        };
237        hit.map(|e| EdgeRef::new(e.idx, s, t))
238    }

242    /// Append an isolated vertex.
243    ///
244    /// `Block::new()` does not allocate, so an isolated vertex costs the 32
245    /// bytes of its record and no malloc.
246    pub fn add_vertex(&mut self) -> Result<VertexId, GraphError> {
247        let v = VertexId::new(self.blocks.len())
248            .ok_or(GraphError::VertexIdSpaceExhausted { max: MAX_INDEX })?;
249        self.blocks.push(Block::new());
250        Ok(v)
251    }

253    /// Add an edge. O(1) amortised in both halves.
254    ///
255    /// Both endpoints are checked *before* an index is taken, so a failed call
256    /// leaves the index space untouched. `add_edge` (`:1190`) calls
257    /// `get_free_idx()` first and indexes `g._edges[s]` afterwards.
258    pub fn add_edge(&mut self, s: VertexId, t: VertexId) -> Result<EdgeRef, GraphError> {
259        let n = self.blocks.len();
260        if s.index() >= n {
261            return Err(GraphError::NoSuchVertex(s));
262        }
263        if t.index() >= n {
264            return Err(GraphError::NoSuchVertex(t));
265        }
266        let id = self.ids.alloc()?;
267        match self.splice_in(id, s, t) {
268            Ok(e) => {
269                self.audit();
270                Ok(e)
271            }
272            Err(err) => {
273                // Unreachable given the bounds above, and written anyway: an
274                // index taken and not linked is a leak the allocator's own
275                // invariant (`live() == bound() - |free|`) would not notice.
276                self.ids.release(id);
277                Err(err)
278            }
279        }
280    }

282    /// Remove an edge, by identity.
283    ///
284    /// Takes an [`EdgeId`], not a descriptor: the endpoints come from the slot
285    /// table, so there is no caller-supplied orientation that can be wrong.
286    pub fn remove_edge(&mut self, e: EdgeId) -> Result<(), GraphError> {
287        self.splice_out(e)?;
288        // The count is `EdgeIds::live()`, moved only here and in `alloc`.
289        // `remove_edge` (`:1310-1312`) instead decrements `_n_edges` at the
290        // call site, which is why `clear_vertex` can and does get it wrong
291        // (`:1403-1410`, defect #1).
292        self.ids.release(e);
293        self.audit();
294        Ok(())
295    }

303    /// Remove the incident edges of `v` that satisfy `pred`.
304    ///
305    /// One algorithm: collect the doomed identities into the reusable scratch
306    /// buffer, then call [`remove_edge`](Self::remove_edge) on each. graph-tool
307    /// has two bodies for this selected by a runtime flag
308    /// (`graph_adjacency.hh:1343-1414` and `:1416-1434`); only one of them is
309    /// wrong, which is the point.
310    /// `pred` is called once per *edge*, as `pred(id, other)`, where `other`
311    /// is the endpoint that is not `v` -- `v` itself for a self-loop. A
312    /// self-loop occupies both halves of `v`'s block and the C++ offers it to
313    /// the predicate **twice** (`:1411-1419`: `pred(ed)` is evaluated before
314    /// the `j >= pos && e.first == v` guard discards the in-half occurrence),
315    /// which a `FnMut` with a side effect can observe. Here the in-half
316    /// occurrence of a self-loop is skipped before the call, so a predicate
317    /// sees each incident edge exactly once; the *set* of removed edges is the
318    /// same, because the C++ discards that occurrence regardless of the answer.
319    pub fn clear_vertex_where<F>(&mut self, v: VertexId, mut pred: F) -> Result<(), GraphError>
320    where
321        F: FnMut(EdgeId, VertexId) -> bool,
322    {
323        if v.index() >= self.blocks.len() {
324            return Err(GraphError::NoSuchVertex(v));
325        }
326
327        // `mem::take` rather than a fresh `Vec`: the buffer keeps its capacity
328        // across calls, and the empty `Vec` left behind has no allocation to
329        // free. A per-call `Vec<edge_descriptor>` -- which is literally what
330        // `:1421-1424` does, `res.reserve(es.size())` -- costs two allocations
331        // per call on a path that runs once per vertex.
332        let mut scratch = core::mem::take(&mut self.scratch);
333        scratch.clear();
334        {
335            let block = &self.blocks[v.index()];
336            let out_len = block.out_degree();
337            for (j, e) in block.all().iter().enumerate() {
338                if j >= out_len && e.other == v {
339                    // The out-half occurrence of this self-loop already
340                    // decided its fate.
341                    continue;
342                }
343                if pred(e.idx, e.other) {
344                    scratch.push(e.idx);
345                }
346            }
347        }
348
349        let mut outcome = Ok(());
350        for &id in &scratch {
351            if let Err(err) = self.remove_edge(id) {
352                outcome = Err(err);
353                break;
354            }
355        }
356
357        scratch.clear();
358        self.scratch = scratch;
359        outcome?;
360        self.audit();
361        Ok(())
362    }

364    /// Remove `v`, moving the last vertex into its slot and relabelling.
365    ///
366    /// Relabelling is expressed as unlink-then-relink through the two splice
367    /// primitives, so both the slot table and the `(s,t)` index are notified
368    /// for *both* directions. `remove_vertex_fast` (`:1471-1535`) instead
369    /// patches endpoints in place and touches `_ehash` only via
370    /// `out_edges(back)` and `out_edges(v)`, leaving every neighbour that held
371    /// `back` as a target keyed on a dead vertex. This is slower and it is why
372    /// the class is gone rather than merely absent today.
373    pub fn swap_remove_vertex(&mut self, v: VertexId) -> Result<(), GraphError> {
374        let n = self.blocks.len();
375        if v.index() >= n {
376            return Err(GraphError::NoSuchVertex(v));
377        }
378        // `clear_vertex(v, g)` first, exactly as `:1475`.
379        self.clear_vertex(v)?;
380
381        let back = VertexId::from_index(n - 1);
382        if v != back {
383            // Collect `back`'s incident identities before touching anything:
384            // every splice below relocates entries within this very block.
385            // A self-loop is named once, by its out-half entry.
386            let mut scratch = core::mem::take(&mut self.scratch);
387            scratch.clear();
388            {
389                let block = &self.blocks[back.index()];
390                let out_len = block.out_degree();
391                for (j, e) in block.all().iter().enumerate() {
392                    if j >= out_len && e.other == back {
393                        continue;
394                    }
395                    scratch.push(e.idx);
396                }
397            }
398            let outcome = self.relabel(&scratch, back, v);
399            scratch.clear();
400            self.scratch = scratch;
401            outcome?;
402        }
403
404        let emptied = self.blocks.pop();
405        debug_assert_eq!(
406            emptied.map_or(0, |b| b.degree()),
407            0,
408            "the vacated block still holds entries"
409        );
410        self.audit();
411        Ok(())
412    }

414    /// Move every edge in `ids` off `from` and onto `onto`.
415    ///
416    /// Unlink then relink, per edge, through the two primitives -- so the slot
417    /// table *and* the `(s,t)` index are notified for both directions and for
418    /// both endpoints. `remove_vertex_fast` (`:1497-1522`) patches
419    /// `es[i].first` in place and repairs `_ehash` only through
420    /// `out_edges(back)` and `out_edges(v)` (`:1479-1484`, `:1524-1529`), so a
421    /// neighbour `u` that held `back` as a **target** keeps a bucket keyed on
422    /// a vertex that no longer exists and `edge(u, v, g)` answers false
423    /// afterwards. That is defect #3, and it is a consequence of `_ehash`
424    /// being keyed on the source alone (`:624`); keying on the ordered pair
425    /// leaves no direction to forget.
426    ///
427    /// The cost is the stated loss: two splices per incident edge where the
428    /// C++ writes one word.
429    fn relabel(
430        &mut self,
431        ids: &[EdgeId],
432        from: VertexId,
433        onto: VertexId,
434    ) -> Result<(), GraphError> {
435        for &id in ids {
436            let (s, t) = self.splice_out(id)?;
437            let s = if s == from { onto } else { s };
438            let t = if t == from { onto } else { t };
439            self.splice_in(id, s, t)?;
440        }
441        Ok(())
442    }

444    /// Release excess capacity across all blocks.
445    ///
446    /// Ports the per-vertex `_edges[i].second.shrink_to_fit()` and the
447    /// `_edges.shrink_to_fit()` of `shrink_to_fit` (`graph_adjacency.hh:926-960`).
448    /// It does **not** recompute the edge index range: `:928-935` sets
449    /// `_edge_idx_range = max(idx) + 1` over the adjacency and drops the free
450    /// list above it, which reclaims the tail of the space while every
451    /// interior hole survives -- and every edge property map keeps paying for
452    /// them. Renumbering is [`EdgeIds::compact`](super::EdgeIds::compact),
453    /// which returns the permutation so property maps can follow.
454    pub fn shrink_to_fit(&mut self) {
455        for b in &mut self.blocks {
456            b.shrink();
457        }
458        self.blocks.shrink_to_fit();
459        self.scratch.shrink_to_fit();
460    }

464    /// Link an already-allocated edge id into both halves.
465    ///
466    /// `add_edge`'s body from `:1195` on, in order: out-half first (which may
467    /// displace the in-half's first entry -- the O(1) push-swap of
468    /// `:1203-1210`), *then* the in-half, then the derived indexes. The
469    /// displaced entry's notification is applied before the second splice
470    /// runs, because for a self-loop the second splice is into the same block.
471    fn splice_in(&mut self, id: EdgeId, s: VertexId, t: VertexId) -> Result<EdgeRef, GraphError> {
472        let n = self.blocks.len();
473        if s.index() >= n {
474            return Err(GraphError::NoSuchVertex(s));
475        }
476        if t.index() >= n {
477            return Err(GraphError::NoSuchVertex(t));
478        }
479
480        let (out_pos, displaced) =
481            self.blocks[s.index()].insert_out(AdjEntry { other: t, idx: id });
482        if let Some(m) = displaced {
483            // `g.get_epos(s_es.back().second).second = s_es.size() - 1`
484            // (`:1208-1209`), as a value rather than a reference into `_epos`.
485            self.slots.on_move(m);
486        }
487        let in_pos = self.blocks[t.index()].insert_in(AdjEntry { other: s, idx: id });
488
489        self.slots.on_insert(
490            id,
491            EdgeSlot {
492                src: s,
493                tgt: t,
494                out_pos,
495                in_pos,
496            },
497        );
498        self.lookup.on_link(s, t, id);
499        Ok(EdgeRef::new(id, s, t))
500    }

502    /// Unlink a live edge from both halves, leaving the id allocated.
503    ///
504    /// The in-half must be re-located *after* the out-half is spliced, because
505    /// the out splice can move it (self-loop, or a boundary promotion). The
506    /// borrow checker forces that ordering: `remove_at` returns its
507    /// notifications by value, so the `&mut Block` ends before the index hooks
508    /// run, and two blocks cannot be borrowed mutably at once -- which is
509    /// exactly the `s_es`/`t_es` aliasing that `:1243-1249` hides.
510    fn splice_out(&mut self, id: EdgeId) -> Result<(VertexId, VertexId), GraphError> {
511        let (s, t) = self.slots.endpoints(id).ok_or(GraphError::NoSuchEdge(id))?;
512        let (owner, out_pos) = self
513            .slots
514            .locate(&self.blocks, id, End::Out)
515            .ok_or(GraphError::NoSuchEdge(id))?;
516        debug_assert_eq!(owner, s, "the slot's source is not the out-half's owner");
517
518        for m in self.blocks[owner.index()]
519            .remove_at(out_pos, End::Out)
520            .into_iter()
521            .flatten()
522        {
523            self.slots.on_move(m);
524        }
525
526        // Re-located here, and not before the splice above, because that
527        // splice can have moved it: removing a self-loop's out-entry promotes
528        // the loop's *own* in-entry across the out/in boundary, and a
529        // boundary promotion moves an unrelated in-entry in every other case.
530        // The C++ reads both positions up front through two references
531        // (`s_es`, `t_es`, `:1252-1257`) that are the same object when
532        // `s == t`, and the second `remove_e` then trusts a `_epos` value the
533        // first one rewrote.
534        let (owner, in_pos) =
535            self.slots
536                .locate(&self.blocks, id, End::In)
537                .ok_or(GraphError::Invariant(
538                    "a live edge's in-half is not locatable",
539                ))?;
540        debug_assert_eq!(owner, t, "the slot's target is not the in-half's owner");
541
542        for m in self.blocks[owner.index()]
543            .remove_at(in_pos, End::In)
544            .into_iter()
545            .flatten()
546        {
547            self.slots.on_move(m);
548        }
549
550        self.slots.on_remove(id);
551        self.lookup.on_unlink(s, t, id);
552        Ok((s, t))
553    }

557    /// Run [`validate`](Self::validate) after a mutation, in debug builds.
558    ///
559    /// This is [DESIGN](crate::design)'s answer to defect #46 -- `check_epos` exists in
560    /// graph-tool and every call site is commented out (`:698`, `:1226`,
561    /// `:1277`, `:1306`, `:1433`) -- wired up.
562    ///
563    /// It carries a size budget, which the skeleton's wording ("after every
564    /// mutation") did not. `validate` is O(V + E); running it unconditionally
565    /// after each O(1) splice makes every debug build quadratic, which is how
566    /// an always-on audit becomes an audit someone turns off. Below the
567    /// budget -- every unit test in this port, and any graph small enough for
568    /// a human to have written down -- it runs on every single mutation.
569    #[inline]
570    fn audit(&self) {
571        #[cfg(debug_assertions)]
572        {
573            /// Entries plus vertices. 4096 makes the audit's own cost O(1) in
574            /// the size of the graphs it actually guards.
575            const BUDGET: usize = 4096;
576            if self.blocks.len() + 2 * self.num_edges() <= BUDGET
577                && let Err(e) = self.validate()
578            {
579                panic!("AdjList invariant broken by the preceding mutation: {e}");
580            }
581        }
582    }

584    /// Cross-check every derived index against the adjacency.
585    ///
586    /// This is `check_epos` (`graph_adjacency.hh:686-718`) actually wired up:
587    /// graph-tool has the function and comments out every call site (`:698`,
588    /// `:1226`, `:1277`, `:1306`, `:1433`). Called by the private `audit`
589    /// after every mutation under `debug_assertions`, and by every test.
590    ///
591    /// Five claims, in order:
592    ///
593    /// 1. **Both halves.** Every entry of every block is named by the matching
594    ///    half of a live slot, at exactly the position stored there, and the
595    ///    slot's endpoints agree with the entry's `other`. `check_epos`
596    ///    (`:701-717`) only walks `edges(g)` -- the out-halves -- so an
597    ///    in-half position that drifted is invisible to it. Checking one half
598    ///    is how a `validate` that never inspects the in-half at all ships.
599    /// 2. **Injectivity, hence completeness.** Distinct entries in the same
600    ///    half map to distinct `(owner, pos)` pairs, so claim 1 makes the
601    ///    out-entries an injection into the live slots; with claim 3 it is a
602    ///    bijection.
603    /// 3. **The count is the adjacency's.** `num_edges()` equals the number of
604    ///    out-entries and the number of in-entries. This is the assertion
605    ///    defect #1 fails: `:1403-1410` decrements `_n_edges` by two for one
606    ///    removed edge, counting `std::remove_if`'s moved-from tail.
607    /// 4. **No orphan slots.** The number of live slots over the whole edge
608    ///    index range equals `num_edges()`, so a released id cannot keep a
609    ///    live slot, and a live id cannot lack entries.
610    /// 5. **The `(s,t)` index agrees**, in multiplicity, not merely in
611    ///    membership -- and only when one exists, so the `NoLookup` audit
612    ///    allocates nothing and costs nothing.
613    pub fn validate(&self) -> Result<(), GraphError> {
614        let n = self.blocks.len();
615        let mut out_entries = 0usize;
616        let mut in_entries = 0usize;
617
618        for (vi, block) in self.blocks.iter().enumerate() {
619            let owner = VertexId::from_index(vi);
620            let out_len = block.out_degree();
621            for (j, entry) in block.all().iter().enumerate() {
622                if entry.other.index() >= n {
623                    return Err(GraphError::Invariant(
624                        "an adjacency entry names a vertex outside the graph",
625                    ));
626                }
627                let end = if j < out_len { End::Out } else { End::In };
628                let (src, tgt) = self
629                    .slots
630                    .endpoints(entry.idx)
631                    .ok_or(GraphError::Invariant(
632                        "an adjacency entry names a released edge",
633                    ))?;
634                let (holder, pos) = self.slots.locate(&self.blocks, entry.idx, end).ok_or(
635                    GraphError::Invariant("an adjacency entry is unreachable from the slot table"),
636                )?;
637                if holder != owner || pos as usize != j {
638                    return Err(GraphError::Invariant(
639                        "the slot table records a different position for an entry",
640                    ));
641                }
642                match end {
643                    End::Out => {
644                        if src != owner || tgt != entry.other {
645                            return Err(GraphError::Invariant(
646                                "the slot's endpoints disagree with an out-half entry",
647                            ));
648                        }
649                        out_entries += 1;
650                    }
651                    End::In => {
652                        if tgt != owner || src != entry.other {
653                            return Err(GraphError::Invariant(
654                                "the slot's endpoints disagree with an in-half entry",
655                            ));
656                        }
657                        in_entries += 1;
658                    }
659                }
660            }
661        }
662
663        if out_entries != in_entries {
664            return Err(GraphError::Invariant(
665                "the out-halves and the in-halves hold different numbers of entries",
666            ));
667        }
668        if out_entries != self.num_edges() {
669            return Err(GraphError::Invariant(
670                "num_edges() disagrees with the adjacency",
671            ));
672        }
673
674        let mut live_slots = 0usize;
675        for i in 0..self.ids.bound() {
676            if self.slots.endpoints(EdgeId::from_index(i)).is_some() {
677                live_slots += 1;
678            }
679        }
680        if live_slots != self.num_edges() {
681            return Err(GraphError::Invariant(
682                "the slot table holds entries the adjacency does not",
683            ));
684        }
685
686        if H::ENABLED {
687            for (vi, block) in self.blocks.iter().enumerate() {
688                let src = VertexId::from_index(vi);
689                for entry in block.out() {
690                    let bucket = self.lookup.find(src, entry.other);
691                    if !bucket.contains(&entry.idx) {
692                        return Err(GraphError::Invariant(
693                            "the (s,t) index does not hold an edge the adjacency does",
694                        ));
695                    }
696                    let multiplicity = block
697                        .out()
698                        .iter()
699                        .filter(|x| x.other == entry.other)
700                        .count();
701                    if bucket.len() != multiplicity {
702                        return Err(GraphError::Invariant(
703                            "the (s,t) index disagrees with the adjacency about multiplicity",
704                        ));
705                    }
706                    if bucket
707                        .iter()
708                        .any(|&id| self.slots.endpoints(id) != Some((src, entry.other)))
709                    {
710                        return Err(GraphError::Invariant(
711                            "the (s,t) index holds an edge whose endpoints are not (s,t)",
712                        ));
713                    }
714                }
715            }
716        }
717
718        Ok(())
719    }
