1//! Adjacency entries and the two edge descriptor types.
2//!
3//! ## Why there are two ([DESIGN](crate::design) D2)
4//!
5//! graph-tool has one, `adj_edge_descriptor {Vertex s, t, idx}`
6//! (`graph_adjacency.hh:186-206`), and it is overloaded:
7//!
8//! * `operator==` compares `idx` **only** (`:196`), so
9//!   `edge_descriptor(1,2,7) == edge_descriptor(9,9,7)`;
10//! * `reverse_edge` (`:571`) mutates `s`/`t` in place, so a descriptor's
11//!   endpoints can disagree with the topology while `==` still says "same
12//!   edge";
13//! * the default value `{max,max,max}` (`:188-190`) is what `edge(s,t,g)`
14//!   returns on failure (`:943`), and compares equal to every other failure.
15//!
16//! Splitting it gives:
17//!
18//! * [`Incident`] -- what *incidence iteration* yields. It is **anchored**:
19//!   `other` is the neighbour, never the query vertex. This is exactly what
20//!   `_all_edges_out` does (`graph_adjacency.hh:1102-1108` builds an
21//!   `out_edge_iterator` over the whole list, so `make_out_edge::def` sets
22//!   `src == u` for the in-half too) and it is what `graph_adaptor.hh:199-207`
23//!   routes an undirected view to.
24//! * [`EdgeRef`] -- what the *global* edge list yields, in canonical storage
25//!   orientation.
26//!
27//! Neither implements `PartialEq`/`Hash`. An edge's identity is its
28//! [`EdgeId`], obtained by `.id()` or `From`, so porting
29//! `unordered_set<edge_descriptor>` (which hashes `e.idx`, `:1618`) to
30//! `HashSet<EdgeRef>` does not compile rather than silently changing results.
31
32use crate::ids::{EdgeId, VertexId};
33
34/// One half-edge as stored: the other endpoint, and the edge's dense index.
35///
36/// The `pair<vertex_t, vertex_t>` of `graph_adjacency.hh:222`, at half the
37/// size: 8 bytes with a 32-bit [`Raw`](crate::ids::Raw), so eight per cache
38/// line where `adj_list<size_t>` gets four.
39#[repr(C)]
40#[derive(Clone, Copy, PartialEq, Eq, Debug)]
41pub struct AdjEntry {
42    /// The endpoint that is not the vertex whose block this is.
43    pub other: VertexId,
44    /// The edge's dense index.
45    pub idx: EdgeId,
46}
47
48const _: () = assert!(size_of::<AdjEntry>() == 2 * size_of::<crate::ids::Raw>());

50/// An edge as seen *from* a vertex.
51///
52/// Yielded by [`out_edges`](crate::graph::GraphRef::out_edges),
53/// [`in_edges`](crate::graph::Bidirectional::in_edges) and
54/// [`all_edges`](crate::graph::GraphRef::all_edges). `other` is always the
55/// neighbour.
56#[derive(Clone, Copy, Debug)]
57pub struct Incident {
58    /// The neighbour.
59    pub other: VertexId,
60    /// The edge's identity.
61    pub edge: EdgeId,
62}
63
64impl Incident {
65    /// The edge's identity.
66    #[inline]
67    pub const fn id(self) -> EdgeId {
68        self.edge
69    }
70}
71
72impl From<Incident> for EdgeId {
73    #[inline]
74    fn from(i: Incident) -> EdgeId {
75        i.edge
76    }
77}

79/// An edge with both endpoints, in canonical storage orientation.
80///
81/// Yielded by [`edges`](crate::graph::EdgeList::edges) and returned by
82/// [`find_edge`](crate::adj::AdjList::find_edge). There is no null value: a
83/// failed lookup is `None`.
84#[derive(Clone, Copy, Debug)]
85pub struct EdgeRef {
86    id: EdgeId,
87    src: VertexId,
88    tgt: VertexId,
89}
90
91impl EdgeRef {
92    /// Construct. Crate-private: an `EdgeRef` always comes from a graph.
93    #[inline]
94    pub(crate) const fn new(id: EdgeId, src: VertexId, tgt: VertexId) -> Self {
95        EdgeRef { id, src, tgt }
96    }
97
98    /// The edge's identity.
99    #[inline]
100    pub const fn id(self) -> EdgeId {
101        self.id
102    }
103    /// The stored source.
104    #[inline]
105    pub const fn source(self) -> VertexId {
106        self.src
107    }
108    /// The stored target.
109    #[inline]
110    pub const fn target(self) -> VertexId {
111        self.tgt
112    }
113
114    /// The same edge with its endpoints exchanged.
115    ///
116    /// Returns a value. `reverse_edge` (`graph_adjacency.hh:571`) mutates the
117    /// descriptor in place, and the result is then accepted by `remove_edge`,
118    /// which is how a caller-supplied orientation can drive a position lookup
119    /// against the wrong half.
120    #[inline]
121    pub const fn reversed(self) -> Self {
122        EdgeRef {
123            id: self.id,
124            src: self.tgt,
125            tgt: self.src,
126        }
127    }
128
129    /// The endpoint that is not `v`, or `None` if `v` is neither.
130    #[inline]
131    pub fn opposite(self, v: VertexId) -> Option<VertexId> {
132        if v == self.src {
133            Some(self.tgt)
134        } else if v == self.tgt {
135            Some(self.src)
136        } else {
137            None
138        }
139    }
140
141    /// View this edge from `v`, if `v` is an endpoint.
142    #[inline]
143    pub fn anchored_at(self, v: VertexId) -> Option<Incident> {
144        self.opposite(v).map(|other| Incident {
145            other,
146            edge: self.id,
147        })
148    }
149}
150
151impl From<EdgeRef> for EdgeId {
152    #[inline]
153    fn from(e: EdgeRef) -> EdgeId {
154        e.id
155    }
156}
