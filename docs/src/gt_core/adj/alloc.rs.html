1//! The dense edge-index allocator.
2
3use crate::error::GraphError;
4use crate::ids::{EdgeId, MAX_INDEX};
5
6/// Owner of the edge index space.
7///
8/// Merges graph-tool's three separate fields -- `_n_edges` (`:607`),
9/// `_edge_idx_range` (`:608`) and `_free_idx` (`:612`) -- into one type whose
10/// only mutators are [`alloc`](Self::alloc) and [`release`](Self::release).
11///
12/// `num_edges()` is therefore *derived*, never accumulated by caller
13/// arithmetic. That deletes an entire defect class: `clear_vertex`
14/// (`graph_adjacency.hh:1404-1413`) computes
15/// `k += count_if(iter, es.begin()+pos, ...)` over the moved-from tail of
16/// `std::remove_if`, which by that algorithm's definition holds the *kept*
17/// elements; a filtered `clear_vertex` (reachable from
18/// `graph_filtered.hh:573-584`) on a vertex with a self-loop plus another
19/// out-edge decrements `_n_edges` by two for one removed edge.
20///
21/// ## The invariant
22///
23/// `live() == bound() - free.len()` holds after every operation and is
24/// `debug_assert`ed there. It is the one relation graph-tool cannot state,
25/// because its three fields are `friend`-visible to every mutating free
26/// function (`:755-818`) and each of them re-derives the count by hand.
27#[derive(Clone, Debug, Default)]
28pub struct EdgeIds {
29    next: usize,
30    free: Vec<EdgeId>,
31    live: usize,
32}
33
34impl EdgeIds {
35    /// An empty allocator.
36    #[inline]
37    pub const fn new() -> Self {
38        EdgeIds {
39            next: 0,
40            free: Vec::new(),
41            live: 0,
42        }
43    }
44
45    /// Take an index from the free list, or extend the range.
46    ///
47    /// Ports `get_free_idx` (`graph_adjacency.hh:627-655`), serial branch:
48    /// LIFO off the back of the free list, else `_edge_idx_range++`. The
49    /// concurrent branch (`_free_idx_m[get_thread_num()]`) has no counterpart
50    /// -- [DESIGN](crate::design) defect #51: `&mut AdjList` refuses concurrent mutation and
51    /// `ParBuilder` replaces the bulk case deterministically, so there is no
52    /// per-thread free list whose contents depend on the schedule.
53    ///
54    /// Exhaustion is a `Result`, not a wrap: `get_free_idx` returns
55    /// `_edge_idx_range++` with no bound at all, and `Vertex` is also the
56    /// index type, so the range silently collides with `null_vertex()` at the
57    /// top of the space.
58    pub fn alloc(&mut self) -> Result<EdgeId, GraphError> {
59        let id = match self.free.pop() {
60            Some(id) => id,
61            None => {
62                let id = EdgeId::new(self.next)
63                    .ok_or(GraphError::EdgeIdSpaceExhausted { max: MAX_INDEX })?;
64                self.next += 1;
65                id
66            }
67        };
68        self.live += 1;
69        debug_assert_eq!(self.live, self.next - self.free.len());
70        Ok(id)
71    }
72
73    /// Return an index to the free list.
74    ///
75    /// Ports `put_free_index` (`graph_adjacency.hh:659-664`). `id` must be
76    /// live; releasing an id twice, or one this allocator never issued, is a
77    /// caller bug that [`AdjList::validate`](super::AdjList::validate) and the
78    /// `debug_assert`s here exist to catch. In a release build the count
79    /// saturates rather than wrapping, because a `live` of `usize::MAX` would
80    /// propagate into every `num_edges()`-sized allocation in the port.
81    pub fn release(&mut self, id: EdgeId) {
82        debug_assert!(
83            id.index() < self.next,
84            "release of an id outside the allocated range"
85        );
86        debug_assert!(self.live > 0, "release with no live edge");
87        self.free.push(id);
88        self.live = self.live.saturating_sub(1);
89        debug_assert_eq!(self.live, self.next - self.free.len());
90    }
91
92    /// Number of live edges. This is `num_edges(g)`.
93    #[inline]
94    pub const fn live(&self) -> usize {
95        self.live
96    }
97
98    /// Size of the index space. This is `get_edge_index_range(g)`, and it is
99    /// what a property map must be sized to -- not [`live`](Self::live), since
100    /// the space is sparse after removals.
101    #[inline]
102    pub const fn bound(&self) -> usize {
103        self.next
104    }
105
106    /// Largest representable index.
107    #[inline]
108    pub const fn max_index() -> usize {
109        MAX_INDEX
110    }
111
112    /// Compact the index space, remapping live edges onto `0..live`.
113    ///
114    /// Returns the permutation so that callers can rewrite edge property maps.
115    ///
116    /// ## What "the permutation" means
117    ///
118    /// The result is indexed by the **old** id and has length the **old**
119    /// [`bound`](Self::bound); `p[old.index()]` is `old`'s new id. It is a
120    /// total permutation of `0..old_bound`, not a partial map with holes:
121    ///
122    /// * the live ids go, order-preservingly, onto `0..live()`;
123    /// * the ids that were on the free list go onto `live()..old_bound`, also
124    ///   order-preservingly.
125    ///
126    /// So an edge property map is rewritten by permuting in place over its
127    /// whole length and then truncating to the new [`bound`](Self::bound) --
128    /// no validity test per element, and no sentinel id to represent "was
129    /// free", which is what a `live`-length map would have needed. The
130    /// adjacency entries and the slot table are rewritten by the same table:
131    /// `entry.idx = p[entry.idx.index()]`.
132    ///
133    /// Afterwards `bound() == live()` and the free list is empty.
134    ///
135    /// There is no counterpart in graph-tool. `shrink_to_fit` (`:541-576`)
136    /// recomputes `_edge_idx_range` as `max(idx) + 1` over the adjacency and
137    /// drops the free-list entries above it, which reclaims the tail of the
138    /// index space but never renumbers an edge -- so a graph that had its
139    /// *interior* edges removed keeps every hole, and every edge property map
140    /// keeps paying for them.
141    pub fn compact(&mut self) -> Vec<EdgeId> {
142        let bound = self.next;
143        let mut freed = vec![false; bound];
144        for id in &self.free {
145            debug_assert!(id.index() < bound, "free id outside the range");
146            freed[id.index()] = true;
147        }
148
149        let mut perm = Vec::with_capacity(bound);
150        let mut live = 0usize;
151        // The dead ids are numbered after the live ones, so the map is a
152        // permutation of the whole old range and the caller needs no "is this
153        // id live" test while rewriting.
154        let mut dead = self.live;
155        for is_free in freed {
156            if is_free {
157                perm.push(EdgeId::from_index(dead));
158                dead += 1;
159            } else {
160                perm.push(EdgeId::from_index(live));
161                live += 1;
162            }
163        }
164        // Fires if an id was released twice: the free list is then longer than
165        // the set of ids it names, and the two counters cross.
166        debug_assert_eq!(live, self.live, "free list disagrees with the count");
167        debug_assert_eq!(dead, bound, "free list disagrees with the range");
168
169        self.next = self.live;
170        self.free.clear();
171        perm
172    }
173}
