1//! Derived indexes: the edge slot table, and the optional `(s,t)` lookup.
2//!
3//! ## The slot table is not optional ([DESIGN](crate::design) D1)
4//!
5//! graph-tool makes `_epos` a runtime flag (`_keep_epos`, `graph_adjacency.hh:619`)
6//! and then writes `clear_vertex` **twice**, once for each setting
7//! (`:1344-1414` and `:1415-1434`). Only one of the two is wrong. Here the
8//! slot table always exists, so there is one algorithm, and there is no
9//! configuration in which two monomorphisations can disagree about whether an
10//! input is valid.
11//!
12//! The slot additionally carries the **endpoints**, not only the positions.
13//! That is what lets [`AdjList::remove_edge`](super::AdjList::remove_edge)
14//! take a bare [`EdgeId`]: there is no caller-supplied orientation left to be
15//! wrong, which retires the `reverse_edge`/`remove_edge` interaction at
16//! `graph_adjacency.hh:578` entirely rather than guarding it.
17//!
18//! ## Liveness without a second table
19//!
20//! `_epos` (`:620`) is a bare `vector<pair<uint32_t,uint32_t>>`: it says
21//! nothing about whether `idx` names a live edge, and every reader has to have
22//! established that some other way. Here [`EdgeSlot::out_pos`] carries the
23//! answer. [`MAX_INDEX`] reserves the top raw value, so `Raw::MAX` is not a
24//! representable adjacency position and can serve as the tombstone --
25//! [`EdgeSlots::endpoints`] and [`EdgeSlots::locate`] are therefore total
26//! functions on *any* [`EdgeId`], which is what the `Option` in their return
27//! types promises.
28

34/// Everything the graph knows about one edge besides its adjacency entries.
35///
36/// 16 bytes at a 32-bit [`Raw`](crate::ids::Raw). Positions are `Raw`, not a
37/// hardcoded `u32`: graph-tool stores `_epos` as `pair<uint32_t,uint32_t>`
38/// (`:620`) beneath a `size_t` vertex, and nothing asserts the relation.
39#[repr(C)]
40#[derive(Clone, Copy, PartialEq, Eq, Debug)]
41pub struct EdgeSlot {
42    /// Stored source.
43    pub src: VertexId,
44    /// Stored target.
45    pub tgt: VertexId,
46    /// Position of this edge's entry in `src`'s out-half.
47    pub out_pos: Raw,
48    /// Position of this edge's entry in `tgt`'s in-half, relative to the block
49    /// start (i.e. including the out-half offset).
50    pub in_pos: Raw,
51}
52
53const _: () = assert!(size_of::<EdgeSlot>() == 4 * size_of::<Raw>());
54// The tombstone below is only unambiguous because the identifier space stops
55// one short of the raw width (`ids::MAX_INDEX`). If that ever changes, a
56// legitimate adjacency position could collide with `EdgeSlot::TOMB`.
57const _: () = assert!(MAX_INDEX < Raw::MAX as usize);
58
59impl EdgeSlot {
60    /// The slot of an index that names no live edge.
61    ///
62    /// The endpoints are arbitrary and are never read: every accessor tests
63    /// [`live`](Self::live) first.
64    const TOMB: EdgeSlot = EdgeSlot {
65        src: VertexId::from_index(0),
66        tgt: VertexId::from_index(0),
67        out_pos: Raw::MAX,
68        in_pos: Raw::MAX,
69    };
70
71    /// Whether this slot names a live edge.
72    #[inline]
73    const fn live(self) -> bool {
74        self.out_pos != Raw::MAX
75    }
76
77    /// The vertex whose block holds the `end` half, and the position within it.
78    ///
79    /// This is the whole of the orientation logic, in one place. graph-tool
80    /// spreads the same two lines over every caller of `_epos` (`:1208`,
81    /// `:1225`, `:1295`, `:1301-1304`, `:1517-1518`), each of which re-derives which
82    /// member of the pair goes with which endpoint from a descriptor the
83    /// caller supplied -- and `reverse_edge` (`:578`) lets that descriptor
84    /// disagree with storage.
85    #[inline]
86    const fn half(self, end: End) -> (VertexId, Raw) {
87        match end {
88            End::Out => (self.src, self.out_pos),
89            End::In => (self.tgt, self.in_pos),
90        }
91    }
92}
93
94/// Narrow a position or a length to [`Raw`].
95///
96/// Every value that reaches this is bounded by a container the allocator
97/// already refused to grow past [`MAX_INDEX`], so the cast is lossless; the
98/// assertion is what keeps that a fact rather than a belief.
99#[inline]
100fn raw(i: usize) -> Raw {
101    debug_assert!(i <= MAX_INDEX, "position exceeds the identifier width");
102    i as Raw
103}

105/// The edge slot table.
106#[derive(Clone, Debug, Default)]
107pub struct EdgeSlots {
108    slot: Vec<EdgeSlot>,
109}
110
111impl EdgeSlots {
112    /// An empty table.
113    #[inline]
114    pub const fn new() -> Self {
115        EdgeSlots { slot: Vec::new() }
116    }
117
118    /// Endpoints of a live edge.
119    #[inline]
120    pub fn endpoints(&self, id: EdgeId) -> Option<(VertexId, VertexId)> {
121        let s = *self.slot.get(id.index())?;
122        s.live().then_some((s.src, s.tgt))
123    }
124
125    /// Locate one half of an edge, verifying the entry actually found there.
126    ///
127    /// The verification lives *here*, checked against the correct half, rather
128    /// than at the caller. An ad-hoc caller-side guard cannot be correct: it
129    /// passes whenever the two halves happen to share a position.
130    ///
131    /// Concretely: `EdgeSlot { out_pos: 0, in_pos: 0, .. }` is a perfectly
132    /// ordinary slot -- the edge is the first out-entry of `src` and, because
133    /// `src` has no in-edges of its own, also the entry at offset 0 of `tgt`
134    /// (a self-loop on an isolated vertex is the smallest case). A guard that
135    /// reads `block.all()[pos]` and compares `idx` therefore succeeds for
136    /// `End::In` while having read the **out** entry. The check below selects
137    /// the half *first* and treats a position on the wrong side of `out_len`
138    /// as a miss, so the two halves can never be confused for one another.
139    pub fn locate(&self, blocks: &[Block], id: EdgeId, end: End) -> Option<(VertexId, Raw)> {
140        let s = *self.slot.get(id.index())?;
141        if !s.live() {
142            return None;
143        }
144        let (owner, pos) = s.half(end);
145        let entry = blocks.get(owner.index())?.get(pos, end)?;
146        (entry.idx == id).then_some((owner, pos))
147    }
148
149    /// Record a new edge.
150    pub(crate) fn on_insert(&mut self, id: EdgeId, s: EdgeSlot) {
151        debug_assert!(
152            s.live(),
153            "a live edge cannot sit at the reserved position Raw::MAX"
154        );
155        let i = id.index();
156        if self.slot.len() <= i {
157            // Ports `set_epos`'s `soft_resize` (`graph_adjacency.hh:672-677`):
158            // the table is indexed by the *edge index range*, not the edge
159            // count, so it is sparse after removals and grows only at the top.
160            self.slot.resize(i + 1, EdgeSlot::TOMB);
161        }
162        self.slot[i] = s;
163    }
164
165    /// Apply a relocation notification.
166    pub(crate) fn on_move(&mut self, m: Moved) {
167        let Some(slot) = self.slot.get_mut(m.id.index()) else {
168            debug_assert!(false, "relocation names an edge with no slot");
169            return;
170        };
171        // A relocation can name the edge currently being spliced out: removing
172        // a self-loop's out-entry promotes its own in-entry across the
173        // out/in boundary. The slot is still live at that point, and must be
174        // updated, or the second half of the splice locates a stale position.
175        debug_assert!(slot.live(), "relocation names a released edge");
176        match m.end {
177            End::Out => slot.out_pos = m.to,
178            End::In => slot.in_pos = m.to,
179        }
180    }
181
182    /// Forget an edge.
183    pub(crate) fn on_remove(&mut self, id: EdgeId) {
184        if let Some(slot) = self.slot.get_mut(id.index()) {
185            debug_assert!(slot.live(), "double release of an edge slot");
186            *slot = EdgeSlot::TOMB;
187        } else {
188            debug_assert!(false, "release names an edge with no slot");
189        }
190    }
191
192    /// Rebuild from scratch. Ports `rebuild_epos` (`:679-699`).
193    ///
194    /// `bound` is the edge *index range* (`_edge_idx_range`), not the edge
195    /// count: the table is indexed by edge id and the space is sparse after
196    /// removals.
197    ///
198    /// **Serial, deliberately.** `rebuild_epos` runs its vertex loop under
199    /// `#pragma omp parallel for` and the plan's skeleton described that as
200    /// "par_iter over blocks, writing disjoint slots". The slots are *not*
201    /// disjoint per block: the edge `(s,t)` is written by `s`'s block (the
202    /// `out_pos` field) and by `t`'s block (the `in_pos` field). C++ gets away
203    /// with it because distinct non-bitfield members are distinct memory
204    /// locations; in Rust that decomposition of a `&mut [EdgeSlot]` has no
205    /// safe expression, and this crate is `#![forbid(unsafe_code)]`. The safe
206    /// parallel forms -- an SoA of atomics, or a bucket-by-edge-id scatter --
207    /// both cost a second pass over the edge set, and this is a cold path
208    /// (bulk construction and compaction), so the single streaming pass wins.
209    pub(crate) fn rebuild(&mut self, blocks: &[Block], bound: usize) {
210        self.slot.clear();
211        self.slot.resize(bound, EdgeSlot::TOMB);
212        for (v, block) in blocks.iter().enumerate() {
213            let owner = VertexId::from_index(v);
214            let out_len = block.out_degree();
215            // `for j in 0..es.size() { if (j < pos) _epos[idx].first = j else
216            // _epos[idx].second = j }` (`:689-696`), plus the endpoints, which
217            // `_epos` does not carry. Both halves agree about them because
218            // `AdjEntry::other` is the *other* endpoint in either half (D2),
219            // so the order in which blocks are scanned does not matter.
220            for (j, e) in block.all().iter().enumerate() {
221                let Some(slot) = self.slot.get_mut(e.idx.index()) else {
222                    // `rebuild` is also the repair path, so an entry naming an
223                    // id outside the range is dropped rather than panicking.
224                    debug_assert!(false, "adjacency names an edge outside the index range");
225                    continue;
226                };
227                let pos = raw(j);
228                if j < out_len {
229                    slot.src = owner;
230                    slot.tgt = e.other;
231                    slot.out_pos = pos;
232                } else {
233                    slot.src = e.other;
234                    slot.tgt = owner;
235                    slot.in_pos = pos;
236                }
237            }
238        }
239    }
240}

242/// Optional `(source, target) -> edges` index.
243///
244/// One table keyed on the **ordered pair**, replacing
245/// `vector<gt_hash_map<Vertex, vector<Vertex>>>` (`graph_adjacency.hh:624`) --
246/// one hash table *object* per vertex, keyed by the source only.
247///
248/// The asymmetry is what causes the second confirmed defect:
249/// `remove_vertex_fast` (`:1471-1535`) patches `_ehash` only through
250/// `out_edges(back)` and `out_edges(v)`, so a neighbour `u` that held `back`
251/// as a *target* keeps a key naming a dead vertex, and `edge(u, v, g)`
252/// afterwards returns false. With no privileged endpoint there is no direction
253/// a relabelling can forget.
254pub trait Lookup: Default + Clone + Send + Sync + 'static {
255    /// Whether this index is live. `false` makes every hook compile away.
256    const ENABLED: bool;
257
258    /// Note an edge's existence.
259    fn on_link(&mut self, s: VertexId, t: VertexId, id: EdgeId);
260    /// Note an edge's removal.
261    fn on_unlink(&mut self, s: VertexId, t: VertexId, id: EdgeId);
262    /// Every edge from `s` to `t`.
263    fn find(&self, s: VertexId, t: VertexId) -> &[EdgeId];
264    /// Rebuild from scratch. Ports `rebuild_ehash` (`:720-732`).
265    fn rebuild(&mut self, blocks: &[Block]);
266}

286/// Hash index on the ordered endpoint pair.
287#[derive(Clone, Default, Debug)]
288pub struct EHash {
289    map: FxHashMap<(VertexId, VertexId), Vec<EdgeId>>,
290    /// Where each edge sits inside its own bucket.
291    ///
292    /// The port of `_ehpos` (`graph_adjacency.hh:625`), and the reason
293    /// [`on_unlink`](Lookup::on_unlink) is O(1) rather than O(multiplicity):
294    /// `remove_ehash` (`:743-752`) swap-removes at the recorded position
295    /// instead of searching the bucket. Indexed by edge id, so it is sparse
296    /// after removals, exactly like [`EdgeSlots`].
297    pos: Vec<Raw>,
298}
299
300impl Lookup for EHash {
301    const ENABLED: bool = true;
302
303    fn on_link(&mut self, s: VertexId, t: VertexId, id: EdgeId) {
304        let bucket = self.map.entry((s, t)).or_default();
305        bucket.push(id);
306        let at = raw(bucket.len() - 1);
307        let i = id.index();
308        if self.pos.len() <= i {
309            self.pos.resize(i + 1, 0);
310        }
311        self.pos[i] = at;
312    }
313
314    fn on_unlink(&mut self, s: VertexId, t: VertexId, id: EdgeId) {
315        let Some(&at) = self.pos.get(id.index()) else {
316            debug_assert!(false, "unlink of an edge this index never saw");
317            return;
318        };
319        let at = at as usize;
320        let Some(bucket) = self.map.get_mut(&(s, t)) else {
321            debug_assert!(false, "unlink names a pair this index never saw");
322            return;
323        };
324        if bucket.get(at) != Some(&id) {
325            debug_assert!(false, "`pos` disagrees with the bucket");
326            return;
327        }
328        // `remove_ehash` (`:743-752`), minus its stale write: the C++ sets
329        // `_ehpos[es.back()] = pos` before popping, which for `pos ==
330        // es.size()-1` records a position for the edge it is about to forget.
331        bucket.swap_remove(at);
332        if let Some(&now) = bucket.get(at) {
333            self.pos[now.index()] = raw(at);
334        }
335        if bucket.is_empty() {
336            // `_ehash[e.s].erase(e.t)` (`:750-751`). Not an optimisation: without
337            // it the key set of the index stops being the set of adjacent
338            // pairs, which is what `rebuild` reconstructs and what
339            // `AdjList::validate` compares against.
340            self.map.remove(&(s, t));
341        }
342    }
343
344    fn find(&self, s: VertexId, t: VertexId) -> &[EdgeId] {
345        self.map.get(&(s, t)).map_or(&[], Vec::as_slice)
346    }
347
348    fn rebuild(&mut self, blocks: &[Block]) {
349        self.map.clear();
350        self.pos.clear();
351        // The out-half alone is the right domain: every edge is an out-entry
352        // of exactly one vertex, so this visits each edge exactly once. It is
353        // also literally what `rebuild_ehash` iterates (`:728-730`,
354        // `out_edges(v, *this)`).
355        for (v, block) in blocks.iter().enumerate() {
356            let s = VertexId::from_index(v);
357            for e in block.out() {
358                self.on_link(s, e.other, e.idx);
359            }
360        }
361    }
362}

465    // -- a real adjacency to check the table against ------------------------

467    /// The two splice primitives, and nothing else.
468    ///
469    /// This is `AdjList::{splice_in, splice_out}` (U5) in miniature. It exists
470    /// so that the slot table is verified against an adjacency that was built
471    /// the way the graph builds one -- `Block::insert_out`'s push-swap, the
472    /// promotion `remove_at` performs across the out/in boundary -- rather
473    /// than against positions a test made up.
474    struct Mini {
475        blocks: Vec<Block>,
476        slots: EdgeSlots,
477        hash: EHash,
478        next: usize,
479        live: Vec<(usize, usize, EdgeId)>,
480    }
481
482    impl Mini {
483        fn new(n: usize) -> Self {
484            Mini {
485                blocks: vec![Block::new(); n],
486                slots: EdgeSlots::new(),
487                hash: EHash::default(),
488                next: 0,
489                live: Vec::new(),
490            }
491        }
492
493        fn add(&mut self, s: usize, t: usize) -> EdgeId {
494            let id = e(self.next);
495            self.next += 1;
496            let (out_pos, moved) = self.blocks[s].insert_out(AdjEntry {
497                other: v(t),
498                idx: id,
499            });
500            if let Some(m) = moved {
501                self.slots.on_move(m);
502            }
503            let in_pos = self.blocks[t].insert_in(AdjEntry {
504                other: v(s),
505                idx: id,
506            });
507            self.slots.on_insert(
508                id,
509                EdgeSlot {
510                    src: v(s),
511                    tgt: v(t),
512                    out_pos,
513                    in_pos,
514                },
515            );
516            self.hash.on_link(v(s), v(t), id);
517            self.live.push((s, t, id));
518            id
519        }
520
521        fn remove(&mut self, id: EdgeId) {
522            let (s, t) = self.slots.endpoints(id).expect("live");
523            let (owner, out_pos) = self
524                .slots
525                .locate(&self.blocks, id, End::Out)
526                .expect("the out half is locatable");
527            assert_eq!(owner, s);
528            for m in self.blocks[s.index()]
529                .remove_at(out_pos, End::Out)
530                .into_iter()
531                .flatten()
532            {
533                self.slots.on_move(m);
534            }
535            // Re-located *after* the out splice, which can have moved it: a
536            // self-loop's in-entry is promoted across the boundary by the very
537            // call above. This ordering is the whole of `splice_out`'s
538            // contract, and reading `in_pos` before the first splice is the
539            // aliasing bug `graph_adjacency.hh:1252-1257` leaves to the reader.
540            let (owner, in_pos) = self
541                .slots
542                .locate(&self.blocks, id, End::In)
543                .expect("the in half is locatable");
544            assert_eq!(owner, t);
545            for m in self.blocks[t.index()]
546                .remove_at(in_pos, End::In)
547                .into_iter()
548                .flatten()
549            {
550                self.slots.on_move(m);
551            }
552            self.slots.on_remove(id);
553            self.hash.on_unlink(s, t, id);
554            let k = self.live.iter().position(|x| x.2 == id).expect("live");
555            self.live.swap_remove(k);
556        }
557
558        /// `check_epos` (`graph_adjacency.hh:701-718`), which graph-tool defines
559        /// and never calls (`:698`, `:1226`, `:1277`, `:1306` and `:1433` are
560        /// all commented out).
561        ///
562        /// Both directions: every live edge is locatable in both halves and
563        /// the entry found there names it, *and* every entry in every block is
564        /// accounted for by exactly one half of one slot.
565        fn check(&self) {
566            let mut seen = 0usize;
567            for &(s, t, id) in &self.live {
568                assert_eq!(self.slots.endpoints(id), Some((v(s), v(t))));
569                let (o, p) = self.slots.locate(&self.blocks, id, End::Out).unwrap();
570                assert_eq!(o, v(s));
571                let entry = self.blocks[s].get(p, End::Out).unwrap();
572                assert_eq!((entry.idx, entry.other), (id, v(t)));
573                let (o, p) = self.slots.locate(&self.blocks, id, End::In).unwrap();
574                assert_eq!(o, v(t));
575                let entry = self.blocks[t].get(p, End::In).unwrap();
576                assert_eq!((entry.idx, entry.other), (id, v(s)));
577                seen += 2;
578            }
579            let entries: usize = self.blocks.iter().map(Block::degree).sum();
580            assert_eq!(entries, seen, "blocks hold entries no slot points at");
581            // …and the `(s,t)` index agrees with the same edge set.
582            let mut model: BTreeMap<(usize, usize), Vec<EdgeId>> = BTreeMap::new();
583            for &(s, t, id) in &self.live {
584                model.entry((s, t)).or_default().push(id);
585            }
586            for ((s, t), mut want) in model {
587                let mut got = self.hash.find(v(s), v(t)).to_vec();
588                want.sort_unstable();
589                got.sort_unstable();
590                assert_eq!(got, want, "EHash disagrees about ({s},{t})");
591            }
592            assert_eq!(self.hash.map.len(), self.hash_keys());
593            check_ehpos(&self.hash);
594        }
595
596        fn hash_keys(&self) -> usize {
597            let mut ks: Vec<(usize, usize)> = self.live.iter().map(|&(s, t, _)| (s, t)).collect();
598            ks.sort_unstable();
599            ks.dedup();
600            ks.len()
601        }

656        let mut g = Mini::new(3);
657        let kept = g.add(0, 1);
658        let gone = g.add(1, 2);
659        g.remove(gone);
660
661        // A released id: the slot is a tombstone, and the block position it
662        // used to name is now occupied by nothing at all.
663        assert_eq!(g.slots.locate(&g.blocks, gone, End::Out), None);
664        assert_eq!(g.slots.locate(&g.blocks, gone, End::In), None);
665        assert_eq!(g.slots.endpoints(gone), None);
666        // An id past the end of the table.
667        assert_eq!(g.slots.locate(&g.blocks, e(50), End::Out), None);
668        assert_eq!(g.slots.locate(&g.blocks, e(50), End::In), None);
669        // A block entry overwritten out from under the slot: the graph
670        // truncated to one vertex, or the entry replaced by another edge's.
671        assert!(g.slots.locate(&g.blocks, kept, End::Out).is_some());
672        assert_eq!(g.slots.locate(&g.blocks[..1], kept, End::In), None);
673        assert_eq!(g.slots.locate(&[], kept, End::Out), None);
674
675        // …and the case a caller-side guard gets wrong. Vertex 0 holds one
676        // out-entry and one in-entry, so positions 0 and 1 both exist; a slot
677        // whose halves agree on a number must still not match the other half.
678        g.add(2, 0);
679        let mut t = EdgeSlots::new();
680        t.on_insert(kept, slot(0, 0, 0, 0));
681        assert!(
682            t.locate(&g.blocks, kept, End::Out).is_some(),
683            "position 0 of vertex 0's out-half really is `kept`"
684        );
685        assert_eq!(
686            t.locate(&g.blocks, kept, End::In),
687            None,
688            "the same number read against the in-half must not match"

771    /// `pos[id]` is the index of `id` in its own bucket, for every live edge,
772    /// and no bucket outlives its last edge.
773    fn check_ehpos(h: &EHash) {
774        for (key, bucket) in &h.map {
775            assert!(
776                !bucket.is_empty(),
777                "empty bucket for {key:?} was not erased"
778            );
779            for (i, id) in bucket.iter().enumerate() {
780                assert_eq!(
781                    h.pos[id.index()] as usize,
782                    i,
783                    "`pos` disagrees with the bucket for {id:?}"
784                );
785            }
786        }
787    }

789    #[test]
790    fn ehash_swap_removes_and_erases_empty_buckets() {
791        let mut h = EHash::default();
792        for i in 0..3 {
793            h.on_link(v(1), v(2), e(i));
794        }
795        h.on_link(v(2), v(1), e(3));
796        check_ehpos(&h);
797        assert_eq!(h.find(v(1), v(2)).len(), 3);
798        assert_eq!(h.find(v(2), v(1)), &[e(3)]);
799        assert_eq!(h.find(v(2), v(2)), &[], "an absent pair is an empty slice");
800
801        // From the middle: the survivor swapped into the hole must have its
802        // recorded position rewritten, or the next unlink reads the wrong one.
803        h.on_unlink(v(1), v(2), e(0));
804        check_ehpos(&h);
805        let mut got: Vec<_> = h.find(v(1), v(2)).to_vec();
806        got.sort_unstable();
807        assert_eq!(got, vec![e(1), e(2)]);
808
809        h.on_unlink(v(1), v(2), e(1));
810        h.on_unlink(v(1), v(2), e(2));
811        check_ehpos(&h);
812        assert_eq!(h.find(v(1), v(2)), &[]);
813        assert!(
814            !h.map.contains_key(&(v(1), v(2))),
815            "the key outlived its last edge"
816        );
817        h.on_unlink(v(2), v(1), e(3));
818        assert!(h.map.is_empty());
819    }
