1//! Monomorphising dispatch from a runtime [`ValueKind`] to a static type.
2//!
3//! ## The two failure modes this closes
4//!
5//! `dispatcher` (`dispatch.hh:186-286`) walks a **linear scan** over a Hana
6//! product guarded by a mutable `bool found`, doing up to three `any_cast`
7//! probes per candidate (`:248-267`) plus a `try`/`catch(bad_extract)` per
8//! candidate on the `python::object` path. If nothing matches it throws
9//! `DispatchNotFound`, whose message is "This is a graph_tool bug. :-("
10//! (`:86-88`) -- so a user type error and a missing instantiation are
11//! indistinguishable.
12//!
13//! [`value_subset!`] generates the member list, the accepted-kind array, the
14//! narrowing, the kernel trait *with its bound*, and the exhaustive match,
15//! **from one token list**. Two consequences:
16//!
17//! 1. A member whose type the kernel cannot handle does not compile. Putting
18//!    `Str => String` into a subset bounded by [`Scalar`](super::Scalar) is
19//!    `error[E0277] ... required by a bound in <Kernel>::call`, pointed at the
20//!    macro invocation.
21//! 2. A row whose [`ValueKind`] and type disagree does not compile. That
22//!    matters: transposing two rows otherwise builds cleanly and produces a
23//!    *runtime* error naming the wrong type as offered and listing it as
24//!    accepted -- strictly worse than `DispatchNotFound`, which at least
25//!    prints honestly.
26//!
27//! ## What remains possible ([DESIGN](crate::design) defect table, row 34)
28//!
29//! The narrowing ends in a `_ => Err` arm, because a subset is by definition
30//! partial. So adding a 16th [`ValueKind`] does not break the build at every
31//! call site; it makes that member unsupported at runtime wherever no subset
32//! lists it. The mitigation is a coverage test asserting that every
33//! `ValueKind::ALL` member appears in at least one shipped subset. This is the
34//! one place the "no `DispatchNotFound`" claim is only partial, and it is
35//! stated rather than hidden.
36
37use crate::ids::IdTag;
38use crate::prop::dense::DenseProp;
39use crate::prop::value::{PropValue, ValueKind};
40use std::any::Any;
41use std::marker::PhantomData;

43/// A type-erased property map at the dispatch boundary.
44///
45/// One `TypeId` comparison, where `std::any` needs `any_cast<T>` then
46/// `any_cast<reference_wrapper<T>>` then `any_cast<shared_ptr<T>>` per
47/// candidate (`dispatch.hh:248-268`): Rust's ownership model collapses that
48/// fan-out, because the boundary always hands out `&'a mut`.
49pub struct AnyMap<'a, K: IdTag> {
50    kind: ValueKind,
51    cell: &'a mut dyn Any,
52    _k: PhantomData<fn() -> K>,
53}
54
55impl<'a, K: IdTag> AnyMap<'a, K> {
56    /// Erase a concrete map.
57    pub fn new<V: PropValue>(m: &'a mut DenseProp<V, K>) -> Self {
58        AnyMap {
59            kind: V::KIND,
60            cell: m,
61            _k: PhantomData,
62        }
63    }
64
65    /// Which member this map stores.
66    #[inline]
67    pub const fn kind(&self) -> ValueKind {
68        self.kind
69    }
70
71    /// Recover the concrete map. Crate-visible: callers go through a
72    /// generated `dispatch`, which performs this once per call and discharges
73    /// it against the const assertion.
74    #[inline]
75    pub fn downcast<V: PropValue>(&mut self) -> Option<&mut DenseProp<V, K>> {
76        self.cell.downcast_mut::<DenseProp<V, K>>()
77    }
78}

80/// A generic operation over the whole 15-member universe.
81///
82/// Replaces `hana::for_each(value_types, f)`. Object-unsafe by construction,
83/// therefore always monomorphised -- the same property as the C++ generic
84/// lambda, which is what makes this a like-for-like port rather than a
85/// regression to dynamic dispatch.
86pub trait KindVisitor {
87    /// What the operation produces.
88    type Out;
89    /// Run at one concrete member.
90    fn visit<V: PropValue>(self) -> Self::Out;
91}
92
93/// Declare a subset of the value universe together with its dispatcher.
94///
95/// ```ignore
96/// value_subset!(
97///     /// The arithmetic scalars.
98///     pub ScalarV, ScalarKernel, "value", gt_core::prop::Scalar,
99///     { Bool => u8, I16 => i16, I32 => i32, I64 => i64, F64 => f64 }
100/// );
101/// ```
102#[macro_export]
103macro_rules! value_subset {
104    (
105        $(#[$meta:meta])*
106        $vis:vis $name:ident, $kernel:ident, $axis:literal, $bound:path,
107        { $($variant:ident => $ty:ty),+ $(,)? }
108    ) => {
109        $(#[$meta])*
110        #[derive(Clone, Copy, PartialEq, Eq, Debug)]
111        $vis enum $name { $(
112            #[doc = concat!("The `", stringify!($variant), "` member.")]
113            $variant
114        ),+ }
115
116        // The row-agreement proof. Without it, `$variant => $ty` is an
117        // unchecked assertion and transposing two rows compiles.
118        $(
119            const _: () = assert!(
120                matches!(<$ty as $crate::prop::PropValue>::KIND, $crate::prop::ValueKind::$variant),
121                concat!(
122                    "value_subset! ", stringify!($name), ": variant ", stringify!($variant),
123                    " is mapped to ", stringify!($ty), ", whose KIND differs"
124                )
125            );
126        )+
127
128        #[doc = concat!("A kernel monomorphised over every member of [`", stringify!($name), "`].")]
129        $vis trait $kernel<K: $crate::ids::IdTag> {
130            /// What the kernel produces.
131            type Out;
132            /// Run at one concrete member, with the map already downcast.
133            fn call<V: $bound>(self, map: &mut $crate::prop::DenseProp<V, K>) -> Self::Out;
134        }
135
136        impl $name {
137            /// Every member this subset accepts, for the error message.
138            $vis const ACCEPTED: &'static [$crate::prop::ValueKind] =
139                &[$($crate::prop::ValueKind::$variant),+];
140
141            /// Narrow a runtime kind into this subset.
142            ///
143            /// The **only** fallible step in a dispatch. Everything after it
144            /// is total.
145            $vis fn narrow(k: $crate::prop::ValueKind)
146                -> ::core::result::Result<Self, $crate::error::DispatchError>
147            {
148                match k {
149                    $($crate::prop::ValueKind::$variant => ::core::result::Result::Ok($name::$variant),)+
150                    other => ::core::result::Result::Err($crate::error::DispatchError {
151                        axis: $axis,
152                        offered: other,
153                        accepted: Self::ACCEPTED,
154                    }),
155                }
156            }
157
158            /// Run `kernel` at the concrete member, handing it the downcast map.
159            ///
160            /// Exhaustive: no `bool found`, no linear scan, no second failure
161            /// point inside the kernel body.
162            #[inline]
163            $vis fn dispatch<K: $crate::ids::IdTag, T: $kernel<K>>(
164                self,
165                map: &mut $crate::prop::dispatch::AnyMap<'_, K>,
166                kernel: T,
167            ) -> T::Out {
168                match self {
169                    $($name::$variant => kernel.call::<$ty>(
170                        map.downcast::<$ty>()
171                            .expect("kind/type agreement is proved by value_subset!'s const assertion"),
172                    ),)+
173                }
174            }
175        }
176    };
177}
