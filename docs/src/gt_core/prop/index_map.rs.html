1//! The identity property map.
2//!
3//! `vertex_index_map_t` and `edge_index_map_t` are appended to every
4//! non-`writable_` property axis (`graph_properties.hh:166-231`), so they are
5//! present in `vertex_properties`, `scalar_vertex_properties`,
6//! `integer_vertex_properties` and their edge counterparts -- the axis names
7//! that occur in roughly three quarters of the 324 dispatch call sites.
8//!
9//! They are *storage-free*: the map returns the descriptor's own index. No
10//! vector-owning or slice-holding property-map type can represent them, which
11//! is why the kernel-facing map is a trait
12//! ([`ReadProp`]) rather than a struct.
13
14use crate::bound::Bound;
15use crate::ids::{Id, IdTag};
16
17use super::map::{Owned, ReadProp};
18
19/// Reads back the identifier's own index. Carries no storage.
20#[derive(Clone, Copy, Debug)]
21pub struct IndexProp<K: IdTag> {
22    bound: Bound<K>,
23}
24
25impl<K: IdTag> IndexProp<K> {
26    /// The identity map over an index space.
27    ///
28    /// Takes the bound so that the map still knows which graph it belongs to
29    /// and how large the space is, for the same checks a
30    /// [`DenseProp`](super::DenseProp) makes.
31    #[inline]
32    pub const fn new(bound: Bound<K>) -> Self {
33        IndexProp { bound }
34    }
35
36    /// The index space this map covers.
37    #[inline]
38    pub const fn bound(&self) -> Bound<K> {
39        self.bound
40    }
41}
42
43impl<K: IdTag> ReadProp<K> for IndexProp<K> {
44    type Value = i64;
45    type Ref<'s>
46        = Owned<i64>
47    where
48        Self: 's;
49    #[inline(always)]
50    fn get_ref(&self, k: Id<K>) -> Owned<i64> {
51        Owned(k.index() as i64)
52    }
53}
54
55// NOTE: deliberately no `WriteProp`. graph-tool's `writable_*` axes exist
56// precisely because the index map cannot be written, and it distinguishes them
57// with a `has_unwritable` flag threaded through `get_seq_type_names`
58// (`graph_properties.hh:278`). Here the distinction is the trait bound.
59
60#[cfg(test)]
61mod tests {
62    use super::*;
63    use crate::ids::{EdgeTag, GraphId, MAX_INDEX, VertexTag};
64
65    /// The whole behaviour: `get(vertex_index, v) == v`, storage-free.
66    #[test]
67    fn the_identity_map_reads_back_the_index() {
68        let g = GraphId::fresh();
69        let p: IndexProp<VertexTag> = IndexProp::new(Bound::new(g, 16));
70        for i in [0usize, 1, 2, 15, 4096, MAX_INDEX] {
71            let k = Id::<VertexTag>::from_index(i);
72            assert_eq!(*p.get_ref(k), i as i64);
73            assert_eq!(p.get(k), k.index() as i64);
74        }
75    }
76
77    /// It is an *index* map, not a bounds-checked map: `vertex_index_map_t`
78    /// answers for any descriptor, and the bound it carries is there for the
79    /// same sizing check a `DenseProp` makes, not to gate reads.
80    #[test]
81    fn a_read_outside_the_bound_is_still_the_index() {
82        let g = GraphId::fresh();
83        let p: IndexProp<VertexTag> = IndexProp::new(Bound::new(g, 2));
84        assert_eq!(*p.get_ref(Id::<VertexTag>::from_index(99)), 99);
85        assert_eq!(p.bound(), Bound::new(g, 2));
86        assert_eq!(p.bound().graph(), g);
87    }
88
89    /// Storage-free is the reason the kernel-facing map is a trait: no
90    /// vector-owning struct can represent this member of the axes at
91    /// `graph_properties.hh:166-231`.
92    #[test]
93    fn the_index_map_carries_nothing_but_its_bound() {
94        assert_eq!(
95            size_of::<IndexProp<VertexTag>>(),
96            size_of::<Bound<VertexTag>>()
97        );
98        assert_eq!(
99            [
100                <IndexProp<VertexTag> as ReadProp<VertexTag>>::IS_UNITY,
101                <IndexProp<VertexTag> as ReadProp<VertexTag>>::IS_CONSTANT,
102            ],
103            [false; 2]
104        );
105    }
106
107    /// The two spaces are separate maps over one identity.
108    #[test]
109    fn both_index_spaces_have_their_own_map() {
110        let g = GraphId::fresh();
111        let v: IndexProp<VertexTag> = IndexProp::new(Bound::new(g, 4));
112        let e: IndexProp<EdgeTag> = IndexProp::new(Bound::new(g, 9));
113        assert_eq!(*v.get_ref(Id::<VertexTag>::from_index(3)), 3);
114        assert_eq!(*e.get_ref(Id::<EdgeTag>::from_index(3)), 3);
115        // `v.get_ref(Id::<EdgeTag>::from_index(3))` does not compile.
116    }
117}
