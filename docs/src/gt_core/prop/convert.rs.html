1//! Conversion between members of the value universe.
2//!
3//! ## Why `From` cannot serve ([DESIGN](crate::design) D7)
4//!
5//! `value_convert.hh:135` handles `is_vector_v<To> && is_vector_v<From>`, and
6//! `prop_map_as` narrows as well as widens. Both
7//! `impl From<Vec<i32>> for Vec<i64>` and `impl From<f64> for i32` are
8//! `error[E0117]` from *any* crate, since `Vec` and the primitives are all
9//! foreign. A design bounded on `U: From<P::Value>` therefore buys exactly
10//! std's widening scalar conversions and nothing else. [`ConvertFrom`] is
11//! crate-local, so every pair in the real lattice is implementable.
12//!
13//! ## The N+M collapse
14//!
15//! `prop_map_as` (`graph_properties.cc:69-87`) crosses two 30-wide
16//! property-map ranges: 900 instantiations, which then run a
17//! `std::is_same_v<decltype(map), decltype(tgt)>` check at runtime to discover
18//! that 30 of them are the identity. Here the 15 diagonal cases stay
19//! monomorphic and fast, and the 210 off-diagonal ones route through
20//! [`AnyValue`]: 15 `to_any` plus 15 `from_any`, i.e. 30 monomorphisations in
21//! place of 225 pairwise `convert<S, T>`. The cost is an intermediate
22//! allocation on the off-diagonal path, which is cold.
23//!
24//! The single blanket `impl<S: ToAny, T: FromAny> ConvertFrom<S> for T` is what
25//! makes that true: `convert_from` is `#[inline]` and its body is two
26//! const-folded guards over `S::KIND` and `T::KIND` around a call into the two
27//! erasure functions, so nothing pairwise survives codegen. Measured on the
28//! release build of `tests/u33_convert.rs`, which instantiates all 196
29//! non-Python ordered pairs: **0** `to_any`, **7** `from_any` and **0**
30//! `convert_from` symbols survive, against the 900 leaves `prop_map_as` emits.
31//!
32//! ## Where this deliberately does *not* follow the C++
33//!
34//! `convert` (`value_convert.hh:73-200`) tries, in order: identity;
35//! `To == python::object`; `From == python::object`; `From == string &&
36//! is_scalar_v<To>`; `To == string && is_scalar_v<From>`;
37//! `is_convertible_v<From, To>`; elementwise for vector-to-vector; then
38//! `convert_dispatch`, which throws. Two rows of that lattice are answered
39//! differently here, both on purpose:
40//!
41//! * **`long double`.** `std::is_scalar_v<long double>` is true, so the C++
42//!   converts it to and from every scalar, to `std::string`, and elementwise
43//!   inside `vector<long double>`. [`LongDouble`] is an opaque 16-byte payload
44//!   with no arithmetic ([DESIGN](crate::design) D7) precisely so that a value graph-tool
45//!   wrote is handed back byte-for-byte rather than routed through an `f64`
46//!   that cannot hold it. So `long double` and `vector<long double>` convert
47//!   only to themselves; every other pair is
48//!   [`NoConversion`](PropError::NoConversion). [`convertible`] says so
49//!   statically, at compile time, which the C++ `is_convertible_v` also does —
50//!   it simply answers `true`.
51//! * **Out-of-range float-to-integer.** `is_convertible_v<double, int16_t>`
52//!   selects `To(v)` at `value_convert.hh:127`, i.e. a `static_cast`, and a
53//!   `double` outside the destination's range makes that **undefined
54//!   behaviour**. Rust's `as` saturates, and this port keeps the saturation:
55//!   `1e30_f64` becomes `i16::MAX`, not whatever `cvttsd2si` happened to
56//!   leave behind. Integer-to-integer narrowing is *not* a divergence — both
57//!   languages wrap modulo 2^N.
58//!
59//! Everything else is matched pair by pair, including the two `uint8_t`
60//! detours at `value_convert.hh:112-116` and `:120-124`, which route the
61//! "bool" member through `int` so that it prints as `"1"` rather than as a
62//! control character.

70/// A normalised carrier for cross-type property conversion.
71#[derive(Clone, Debug, PartialEq)]
72pub enum AnyValue {
73    /// Any integral member, widened.
74    Int(i64),
75    /// `double`.
76    Float(f64),
77    /// `long double`, opaque.
78    Long(LongDouble),
79    /// `string`.
80    Str(String),
81    /// Any integral vector, widened.
82    IntVec(Vec<i64>),
83    /// `vector<double>`.
84    FloatVec(Vec<f64>),
85    /// `vector<long double>`, opaque.
86    LongVec(Vec<LongDouble>),
87    /// `vector<string>`.
88    StrVec(Vec<String>),
89    /// A Python handle. Never leaves the thread that made it, because the
90    /// only value that can occupy it is `!Send`.
91    Py(PyCell),
92}
93
94impl AnyValue {
95    /// The member this carrier presents itself as.
96    ///
97    /// The carrier is *normalised*, so this is not in general the member the
98    /// value came from: every integral member widens into
99    /// [`Int`](AnyValue::Int) and reports [`ValueKind::I64`]. It is the kind
100    /// named in a [`PropError::NoConversion`] raised by
101    /// [`FromAny::from_any`] called directly; [`ConvertFrom::convert_from`]
102    /// knows the true source member statically and names *that* instead.
103    pub const fn kind(&self) -> ValueKind {
104        match self {
105            AnyValue::Int(_) => ValueKind::I64,
106            AnyValue::Float(_) => ValueKind::F64,
107            AnyValue::Long(_) => ValueKind::LongDouble,
108            AnyValue::Str(_) => ValueKind::Str,
109            AnyValue::IntVec(_) => ValueKind::VecI64,
110            AnyValue::FloatVec(_) => ValueKind::VecF64,
111            AnyValue::LongVec(_) => ValueKind::VecLongDouble,
112            AnyValue::StrVec(_) => ValueKind::VecStr,
113            AnyValue::Py(_) => ValueKind::PyObject,
114        }
115    }
116}

118/// Opaque slot for the Python member inside [`AnyValue`].
119///
120/// A distinct type so that `AnyValue` stays nameable in crates built without
121/// the `python` feature, while remaining impossible to construct there: with
122/// the feature off the payload is [`Infallible`](std::convert::Infallible), so
123/// `AnyValue::Py` is an **uninhabited** variant rather than a reachable one
124/// carrying nothing.
125///
126/// With the feature on it holds the handle itself, which is what makes
127/// `python::object` a full member of the conversion lattice rather than a
128/// 15th member that can only be named. The consequence is deliberate and
129/// load-bearing: [`PyValue`](crate::prop::value::PyValue) is `!Send`, so
130/// `AnyValue` is `!Send` too in a Python build, and the carrier cannot be the
131/// hole through which a handle reaches rayon ([DESIGN](crate::design) §7).
132#[cfg(not(feature = "python"))]
133#[derive(Clone, Debug, PartialEq, Eq)]
134pub struct PyCell(#[doc(hidden)] pub(crate) std::convert::Infallible);
135
136/// Opaque slot for the Python member inside [`AnyValue`].
137#[cfg(feature = "python")]
138pub struct PyCell(#[doc(hidden)] pub(crate) crate::prop::value::PyValue);

140#[cfg(feature = "python")]
141impl Clone for PyCell {
142    /// `Py_INCREF` needs a token, and `PyValue` deliberately has no `Clone`.
143    /// Acquiring one here is sound and cannot deadlock a worker: the handle is
144    /// `!Send`, so this runs on the thread that made it, and [DESIGN](crate::design) §7
145    /// states explicitly that `Python::with_gil` inside a worker is the
146    /// supported shape.
147    fn clone(&self) -> Self {
148        PyCell(pyo3::Python::with_gil(|py| self.0.clone_ref(py)))
149    }
150}

152#[cfg(feature = "python")]
153impl PartialEq for PyCell {
154    /// Object identity, matching `operator==` on a `boost::python::object`
155    /// handle rather than `__eq__` on the object: comparing two carriers must
156    /// not be able to run arbitrary Python.
157    fn eq(&self, other: &Self) -> bool {
158        use pyo3::types::PyAnyMethods;
159        pyo3::Python::with_gil(|py| self.0.bind(py).is(&other.0.bind(py)))
160    }
161}

173/// Lossy or lossless conversion within the value universe.
174pub trait ConvertFrom<S>: Sized {
175    /// Convert, or explain why not.
176    fn convert_from(s: &S) -> Result<Self, PropError>;
177}
178
179/// Erase a value into the carrier.
180pub trait ToAny: PropValue {
181    /// Normalise.
182    fn to_any(&self) -> AnyValue;
183}
184
185/// Recover a value from the carrier.
186pub trait FromAny: PropValue + Sized {
187    /// Denormalise, or explain why not.
188    fn from_any(a: &AnyValue) -> Result<Self, PropError>;
189
190    /// The diagonal of the lattice: duplicate a value of this same member.
191    ///
192    /// Not `Clone`. Nine of the fifteen members are `Clone` and the
193    /// fifteenth deliberately is not — `Py_INCREF` needs a `Python<'py>`
194    /// token, which is exactly why `PyValue` has `clone_ref` and no `Clone`
195    /// impl ([DESIGN](crate::design) D7, §7). Bounding [`ConvertFrom`]'s blanket impl on
196    /// `T: Clone` would therefore drop `python::object` out of the lattice,
197    /// and `convert<python::object, python::object>` is `value_convert.hh:79`,
198    /// the very first branch the C++ takes.
199    fn from_same(s: &Self) -> Self;
200}

206/// `long double` carries its bytes and nothing else, so it is on the diagonal
207/// and nowhere else. See the module docs.
208const fn is_opaque(k: ValueKind) -> bool {
209    matches!(k, ValueKind::LongDouble | ValueKind::VecLongDouble)
210}
211
212/// Whether a conversion between these two members exists at all.
213///
214/// The analogue of `is_convertible_v<To, From>` (`value_convert.hh:220-226`),
215/// which the C++ computes by instantiating `convert<To, From, true>` and
216/// asking whether the result type is `void`. Here it is an ordinary `const fn`
217/// over the two [`ValueKind`]s, so [`ConvertFrom::convert_from`] folds it away
218/// and an impossible pair compiles to a constant `Err` that never reaches
219/// [`ToAny::to_any`] at all.
220///
221/// **Type level, not value level.** `string -> int32_t` is convertible; the
222/// string `"abc"` still fails, exactly as `boost::lexical_cast` still throws.
223/// The rules, read off `value_convert.hh:73-165`:
224///
225/// * the diagonal always converts (`:79`);
226/// * `long double` and `vector<long double>` convert only on the diagonal —
227///   this port's divergence, see the module docs;
228/// * `python::object` converts to and from everything else (`:82`, `:86-107`),
229///   including the vector members, whose element-wise `extract` fallback lives
230///   at `:96-106`;
231/// * otherwise a scalar-or-string converts to a scalar-or-string (`:110`,
232///   `:118`, `:126`) and a vector converts to a vector (`:130-144`), and the
233///   two groups do not mix: `convert_dispatch`'s primary template throws
234///   (`:204-210`).
235pub const fn convertible(from: ValueKind, to: ValueKind) -> bool {
236    if from as u8 == to as u8 {
237        return true;
238    }
239    if is_opaque(from) || is_opaque(to) {
240        return false;
241    }
242    if matches!(from, ValueKind::PyObject) || matches!(to, ValueKind::PyObject) {
243        return true;
244    }
245    from.is_vector() == to.is_vector()
246}

248/// A pair that does not exist in the lattice, or a value that did not survive
249/// one that does.
250///
251/// `#[cold]`: every caller is an error path, and keeping the `PropError`
252/// construction out of line leaves the conversion bodies branch-free.
253#[cold]
254fn no_conversion(from: ValueKind, to: ValueKind) -> PropError {
255    PropError::NoConversion { from, to }
256}
257
258impl<S: ToAny, T: FromAny> ConvertFrom<S> for T {
259    #[inline]
260    fn convert_from(s: &S) -> Result<T, PropError> {
261        // The diagonal. `ValueKind` is injective over the universe -- each of
262        // the fifteen types is the sole inhabitant of its member -- so equal
263        // kinds mean `S == T` and the downcast cannot fail; it is here because
264        // proving that to the compiler is what makes this safe rather than a
265        // `transmute`. The guard is a comparison of two associated consts, so
266        // the whole block folds away on every off-diagonal leaf and the
267        // `TypeId` compare never reaches a real build.
268        if S::KIND as u8 == T::KIND as u8
269            && let Some(same) = (s as &dyn Any).downcast_ref::<T>()
270        {
271            return Ok(T::from_same(same));
272        }
273        // Also const, because `convertible` is a `const fn` over two
274        // associated consts: an impossible pair folds to a constant `Err` and
275        // never touches the carrier, so `to_any` is not instantiated for it.
276        if !convertible(S::KIND, T::KIND) {
277            return Err(no_conversion(S::KIND, T::KIND));
278        }
279        T::from_any(&s.to_any()).map_err(|e| match e {
280            // `from_any` only sees the *normalised* member, so it names
281            // `int64_t` where the caller passed a `uint8_t` map. The static
282            // half of the pair is known here; say the truth.
283            PropError::NoConversion { .. } => no_conversion(S::KIND, T::KIND),
284            other => other,
285        })
286    }
287}

293/// `boost::lexical_cast<P>(s)`: strict, whole-string, no surrounding
294/// whitespace. `str::parse` agrees on all of that, and also on rejecting
295/// `"1.5"` for an integral `P` and `"70000"` for `int16_t`.
296fn parse_lex<P: FromStr>(s: &str) -> Option<P> {
297    s.parse::<P>().ok()
298}
299
300/// `boost::lexical_cast<std::string>(double)`.
301///
302/// Boost writes through a stream at `lcast_precision<double>::value == 17`
303/// significant digits in the default (`%g`-like) float format. Rust's
304/// `Display` writes the *shortest* decimal that round-trips. Both round-trip;
305/// this one is shorter, and `0.1` prints as `"0.1"` rather than as
306/// `"0.10000000000000001"`. The non-finite spellings are matched exactly,
307/// which `Display` alone does not do: it writes `NaN`, and every `.gt` and
308/// GraphML file in existence says `nan`.
309fn fmt_f64(x: f64) -> String {
310    if x.is_nan() {
311        "nan".to_owned()
312    } else {
313        // `inf` and `-inf` already agree with boost.
314        x.to_string()
315    }
316}

318#[cfg(feature = "python")]
319fn py_extract<T>(c: &PyCell, to: ValueKind) -> Result<T, PropError>
320where
321    T: for<'py> pyo3::FromPyObject<'py>,
322{
323    use pyo3::types::PyAnyMethods;
324    pyo3::Python::with_gil(|py| {
325        let bound = c.0.bind(py);
326        bound
327            .extract::<T>()
328            .map_err(|_| no_conversion(ValueKind::PyObject, to))
329    })
330}

332/// The four integral members. `$lex` is the type `boost::lexical_cast` is
333/// asked for when the source is a `std::string`, which is **not** always
334/// `$t`: `value_convert.hh:112-116` runs `uint8_t` through `int` first, so
335/// `"300"` parses as the `int` 300 and only then narrows to 44 — whereas
336/// `"5000000000"` is rejected by `lexical_cast<int>` rather than silently
337/// wrapping through `int64_t`.
338macro_rules! impl_integral_member {
339    ($($t:ty => $lex:ty),+ $(,)?) => {$(
340        impl ToAny for $t {
341            #[inline]
342            fn to_any(&self) -> AnyValue {
343                AnyValue::Int(i64::from(*self))
344            }
345        }
346
347        impl FromAny for $t {
348            fn from_any(a: &AnyValue) -> Result<Self, PropError> {
349                let from = a.kind();
350                match a {
351                    // `To(v)` (`:127`): modulo 2^N in both languages.
352                    AnyValue::Int(i) => Ok(*i as $t),
353                    // `To(v)` again, but out of range is UB there and
354                    // saturating here. See the module docs.
355                    AnyValue::Float(f) => Ok(*f as $t),
356                    AnyValue::Str(s) => parse_lex::<$lex>(s)
357                        .map(|v| v as $t)
358                        .ok_or_else(|| no_conversion(from, Self::KIND)),
359                    #[cfg(feature = "python")]
360                    AnyValue::Py(c) => py_extract::<$t>(c, Self::KIND),
361                    _ => Err(no_conversion(from, Self::KIND)),
362                }
363            }
364
365            #[inline]
366            fn from_same(s: &Self) -> Self {
367                *s
368            }
369        }
370    )+};
371}

373impl_integral_member!(u8 => i32, i16 => i16, i32 => i32, i64 => i64);

382impl FromAny for f64 {
383    fn from_any(a: &AnyValue) -> Result<Self, PropError> {
384        let from = a.kind();
385        match a {
386            // `is_convertible_v<int64_t, double>` (`:126`): widening, and

395            _ => Err(no_conversion(from, Self::KIND)),
396        }
397    }
398
399    #[inline]
400    fn from_same(s: &Self) -> Self {
401        *s
402    }
403}

455// ---------------------------------------------------------------------------
456// Vectors
457//
458// `value_convert.hh:130-144`: vector-to-vector is elementwise and only when
459// the *elements* convert, and one failed element fails the whole conversion
460// (the `try` block at `:75` catches nothing until it unwinds out of the loop).
461// `collect::<Result<Vec<_>, _>>()` is the same short-circuit.
462// ---------------------------------------------------------------------------
463
464macro_rules! impl_integral_vector_member {
465    ($($t:ty => $lex:ty),+ $(,)?) => {$(
466        impl ToAny for Vec<$t> {
467            #[inline]
468            fn to_any(&self) -> AnyValue {
469                AnyValue::IntVec(self.iter().map(|&x| i64::from(x)).collect())
470            }
471        }
472
473        impl FromAny for Vec<$t> {
474            fn from_any(a: &AnyValue) -> Result<Self, PropError> {
475                let from = a.kind();
476                match a {
477                    AnyValue::IntVec(v) => Ok(v.iter().map(|&x| x as $t).collect()),
478                    AnyValue::FloatVec(v) => Ok(v.iter().map(|&x| x as $t).collect()),
479                    AnyValue::StrVec(v) => v
480                        .iter()
481                        .map(|s| {
482                            parse_lex::<$lex>(s)
483                                .map(|x| x as $t)
484                                .ok_or_else(|| no_conversion(from, Self::KIND))
485                        })
486                        .collect(),
487                    #[cfg(feature = "python")]
488                    AnyValue::Py(c) => py_extract::<Vec<$t>>(c, Self::KIND),
489                    _ => Err(no_conversion(from, Self::KIND)),
490                }
491            }
492
493            #[inline]
494            fn from_same(s: &Self) -> Self {
495                s.clone()
496            }
497        }
498    )+};
499}

590#[cfg(feature = "python")]
591impl FromAny for crate::prop::value::PyValue {
592    /// `convert<python::object, From>` is `boost::python::object(v)`
593    /// (`value_convert.hh:81-84`) and cannot fail. It cannot fail here either,
594    /// with the one exception the module docs record: `long double` has no
595    /// Python spelling that round-trips its bytes, so the opaque carrier
596    /// refuses rather than handing out an `f64` that lost the low bits.
597    fn from_any(a: &AnyValue) -> Result<Self, PropError> {
598        use pyo3::ToPyObject;
599        let from = a.kind();
600        pyo3::Python::with_gil(|py| {
601            let obj = match a {
602                AnyValue::Int(i) => i.to_object(py),
603                AnyValue::Float(f) => f.to_object(py),
604                AnyValue::Str(s) => s.to_object(py),
605                AnyValue::IntVec(v) => v.to_object(py),
606                AnyValue::FloatVec(v) => v.to_object(py),
607                AnyValue::StrVec(v) => v.to_object(py),
608                AnyValue::Py(c) => return Ok(Self::from_same(&c.0)),
609                AnyValue::Long(_) | AnyValue::LongVec(_) => {
610                    return Err(no_conversion(from, Self::KIND));
611                }
612            };
613            Ok(crate::prop::value::PyValue::new(obj))
614        })
615    }
616
617    fn from_same(s: &Self) -> Self {
618        pyo3::Python::with_gil(|py| s.clone_ref(py))
619    }
620}

645            let opaque = matches!(k, ValueKind::LongDouble | ValueKind::VecLongDouble);
646            assert_eq!(

648                opaque && k == ValueKind::LongDouble

650            assert_eq!(
651                convertible(k, ValueKind::VecLongDouble),
652                opaque && k == ValueKind::VecLongDouble

654        }

662        assert!(!convertible(ValueKind::Str, ValueKind::VecStr));
663        assert!(!convertible(ValueKind::F64, ValueKind::VecF64));
664        assert!(!convertible(ValueKind::VecF64, ValueKind::I64));
