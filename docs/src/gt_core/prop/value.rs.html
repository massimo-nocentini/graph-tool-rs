1//! The closed 15-member value universe.
2//!
3//! ## What is on the base trait, and why it matters ([DESIGN](crate::design) D7)
4//!
5//! Two of the source designs wrote
6//! `trait PropValue: Sealed + Default + Send + Sync + 'static`. Both bounds
7//! are wrong, and each breaks a different headline claim:
8//!
9//! * **`Default`** is unimplementable for the 15th member. pyo3 has no
10//!   `impl Default for Py<T>` -- the default `python::object` is `Py_None`,
11//!   which needs a live interpreter and an incref. So
12//!   `data.resize_with(n, T::default)`, the body of the central sizing
13//!   chokepoint, cannot exist for the Python type. Here growth takes a
14//!   `FnMut() -> T` closure, which the PyO3 boundary fills by capturing its
15//!   `Python<'py>` token; the context-free case is the [`Zeroed`] marker.
16//! * **`Send + Sync`** is exactly what defeats the GIL guarantee, because
17//!   rayon checks `Send`, not a positive marker such as `GilFree`. With them
18//!   on the base trait, `store.as_mut_slice().par_iter_mut()` over a
19//!   `python::object` map compiles with no `unsafe` and no diagnostic. They
20//!   live on [`GilFree`] instead, and -- belt and braces -- [`PyValue`] is
21//!   `!Send` *by construction*, so a kernel that forgets the marker still
22//!   cannot reach rayon.
23//!
24//! `long double` and `vector<long double>` have no Rust equivalent (`f128` is
25//! unstable). They are carried as [`LongDouble`], an opaque 16-byte payload
26//! with **no arithmetic**: `.gt` files and `PropertyMap.value_type()` round-trip
27//! unchanged, and the [`Scalar`] bound excludes the type from every arithmetic
28//! kernel at compile time. That is what the subset-bound mechanism is for.
29
30mod sealed {
31    /// Seals the value universe. Private, so no downstream crate can name it
32    /// and therefore none can add a 16th member -- in particular, none can
33    /// re-introduce a `Send + Sync` Python handle under a different newtype.
34    pub trait Sealed {}
35}
36
37/// The 15 members, as a closed enum.
38///
39/// Replaces the positionally-coupled pair `value_types`
40/// (`graph_properties.hh:61-69`) and `type_names[]` (`:72-76`), which are kept
41/// in lockstep by hand and by nothing else.
42#[repr(u8)]
43#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
44pub enum ValueKind {
45    /// `uint8_t`, named "bool" on the Python side.
46    Bool = 0,
47    /// `int16_t`.
48    I16,
49    /// `int32_t`.
50    I32,
51    /// `int64_t`.
52    I64,
53    /// `double`.
54    F64,
55    /// `long double`, carried opaquely as [`LongDouble`].
56    LongDouble,
57    /// `std::string`.
58    Str,
59    /// `std::vector<uint8_t>`.
60    VecBool,
61    /// `std::vector<int16_t>`.
62    VecI16,
63    /// `std::vector<int32_t>`.
64    VecI32,
65    /// `std::vector<int64_t>`.
66    VecI64,
67    /// `std::vector<double>`.
68    VecF64,
69    /// `std::vector<long double>`.
70    VecLongDouble,
71    /// `std::vector<std::string>`.
72    VecStr,
73    /// `boost::python::object`.
74    PyObject,
75}

77impl ValueKind {
78    /// Every member, in graph-tool's declaration order.
79    pub const ALL: [ValueKind; 15] = [
80        ValueKind::Bool,
81        ValueKind::I16,
82        ValueKind::I32,
83        ValueKind::I64,
84        ValueKind::F64,
85        ValueKind::LongDouble,
86        ValueKind::Str,
87        ValueKind::VecBool,

97    /// The Python-visible name, byte-for-byte `type_names[]`
98    /// (`graph_properties.hh:72-76`). An exhaustive match, so a 16th member
99    /// breaks the build here rather than silently mis-naming.
100    pub const fn name(self) -> &'static str {
101        match self {
102            ValueKind::Bool => "bool",
103            ValueKind::I16 => "int16_t",
104            ValueKind::I32 => "int32_t",
105            ValueKind::I64 => "int64_t",
106            ValueKind::F64 => "double",
107            ValueKind::LongDouble => "long double",
108            ValueKind::Str => "string",
109            ValueKind::VecBool => "vector<bool>",
110            ValueKind::VecI16 => "vector<int16_t>",
111            ValueKind::VecI32 => "vector<int32_t>",
112            ValueKind::VecI64 => "vector<int64_t>",
113            ValueKind::VecF64 => "vector<double>",
114            ValueKind::VecLongDouble => "vector<long double>",
115            ValueKind::VecStr => "vector<string>",
116            ValueKind::PyObject => "python::object",
117        }
118    }
119
120    /// Parse a Python-visible name.
121    ///
122    /// The inverse of [`name`](Self::name), and **exact**: `new_property`
123    /// (`graph_python_interface.hh:674-691`) drives a `hana::for_each` over
124    /// `value_types` and accepts the member whose `type_name == type_names[i]`
125    /// (`:659`), with no normalisation of case, whitespace or spelling. A name
126    /// that matches nothing there raises `ValueException("Invalid property
127    /// type: " + type)` (`:689`); here it is `None`.
128    ///
129    /// The alias table -- `"int" -> "int32_t"`, `"float" -> "double"`,
130    /// `"object" -> "python::object"`, and the `vector<...>` rewrite --
131    /// belongs to `_type_alias` (`graph_tool/__init__.py:210-229`), which runs
132    /// *above* the C++ boundary and never reaches `type_names[]`. Folding it
133    /// in here would make `from_name(name(k))` no longer the only way to
134    /// obtain a kind from a string, and would silently accept spellings
135    /// graph-tool's own C++ rejects. It stays at the Python boundary.
136    ///
137    /// `const` so the round-trip against [`name`](Self::name) is proved at
138    /// compile time rather than asserted in a test; see the `const` block
139    /// below. It matches on the bytes because `str` patterns are not yet
140    /// allowed in a `const fn`, which changes nothing about the generated
141    /// code: either spelling lowers to a length-bucketed comparison chain,
142    /// not to the C++ linear scan.
143    pub const fn from_name(s: &str) -> Option<ValueKind> {
144        Some(match s.as_bytes() {
145            b"bool" => ValueKind::Bool,
146            b"int16_t" => ValueKind::I16,
147            b"int32_t" => ValueKind::I32,
148            b"int64_t" => ValueKind::I64,
149            b"double" => ValueKind::F64,
150            b"long double" => ValueKind::LongDouble,
151            b"string" => ValueKind::Str,
152            b"vector<bool>" => ValueKind::VecBool,
153            b"vector<int16_t>" => ValueKind::VecI16,
154            b"vector<int32_t>" => ValueKind::VecI32,
155            b"vector<int64_t>" => ValueKind::VecI64,
156            b"vector<double>" => ValueKind::VecF64,
157            b"vector<long double>" => ValueKind::VecLongDouble,
158            b"vector<string>" => ValueKind::VecStr,
159            b"python::object" => ValueKind::PyObject,
160            _ => return None,
161        })
162    }
163
164    /// `hana::filter(value_types, is_scalar)` (`:78-80`), minus
165    /// `long double`, which has no arithmetic here. See the module docs.
166    pub const fn is_scalar(self) -> bool {
167        matches!(
168            self,
169            ValueKind::Bool | ValueKind::I16 | ValueKind::I32 | ValueKind::I64 | ValueKind::F64
170        )
171    }
172
173    /// `integer_types` (`:83`).
174    pub const fn is_integer(self) -> bool {
175        matches!(
176            self,
177            ValueKind::Bool | ValueKind::I16 | ValueKind::I32 | ValueKind::I64
178        )
179    }
180
181    /// `floating_types` (`:87`).
182    pub const fn is_floating(self) -> bool {
183        matches!(self, ValueKind::F64 | ValueKind::LongDouble)
184    }
185
186    /// `vector_types` (`:91`).
187    pub const fn is_vector(self) -> bool {
188        self.elem().is_some()
189    }
190
191    /// The element kind of a vector member.
192    pub const fn elem(self) -> Option<ValueKind> {
193        match self {
194            ValueKind::VecBool => Some(ValueKind::Bool),
195            ValueKind::VecI16 => Some(ValueKind::I16),
196            ValueKind::VecI32 => Some(ValueKind::I32),
197            ValueKind::VecI64 => Some(ValueKind::I64),
198            ValueKind::VecF64 => Some(ValueKind::F64),
199            ValueKind::VecLongDouble => Some(ValueKind::LongDouble),
200            ValueKind::VecStr => Some(ValueKind::Str),
201            _ => None,
202        }
203    }
204}

206// `from_name(name(k)) == Some(k)` for all fifteen members, proved at compile
207// time. `const _` rather than a named item: it is a proof obligation, not a
208// value, and nothing should be able to "use" it.
209//
210// The pair this replaces -- `value_types` (`graph_properties.hh:61-69`) and
211// `type_names[]` (`:72-76`) -- is held in correspondence by **position in two
212// separate declarations** and by nothing else; `hana::index_if` (`:656`)
213// happily indexes `type_names[]` with whatever index the type tuple yields,
214// so inserting a member into one list and not the other mis-names every
215// member after it, silently, in `PropertyMap.value_type()` and in every `.gt`
216// header written thereafter.
217//
218// Here the two directions are independent matches, so the same mistake is
219// still *writable* -- but it is not *compilable*: a member whose `name` arm
220// and `from_name` arm disagree fails this evaluation, and a member missing
221// from `from_name` falls into its `_` arm and fails it too. There is no
222// runtime path to the inconsistent state, which is why this is a `const`
223// block rather than a `#[test]`.
224const _: () = {
225    let mut i = 0;
226    while i < ValueKind::ALL.len() {
227        let k = ValueKind::ALL[i];
228        // `as u8` because `PartialEq::eq` is not usable in a const context;
229        // the discriminants are explicit (`#[repr(u8)]`, `Bool = 0`), so the
230        // comparison is exact rather than approximate.
231        assert!(
232            matches!(ValueKind::from_name(k.name()), Some(g) if g as u8 == k as u8),
233            "ValueKind::name and ValueKind::from_name disagree: some member \
234             does not round-trip through its type_names[] spelling"
235        );
236        i += 1;
237    }
238};

240/// A member of the value universe.
241///
242/// `Elem` replaces the whole `last_type_func_t` family (`dispatch.hh:113`,
243/// `:301-385`) by *projection* rather than by search: `velem_dprop_t` and its
244/// siblings collapse a dispatch axis to one element, but `hana::for_each`
245/// still runs over that element, still performs the three-way `any_cast`
246/// probe, and can still reach the `DispatchNotFound` throw. An associated
247/// type adds no arm, no probe and no failure mode.
248pub trait PropValue: sealed::Sealed + 'static {
249    /// This type's member.
250    const KIND: ValueKind;
251    /// `Self` for scalars and the Python member, the element type for vectors.
252    type Elem: PropValue;
253}

255/// Values that can be constructed with no context: the 14 non-Python members.
256pub trait Zeroed: PropValue {
257    /// The value a freshly grown property-map slot holds.
258    fn zero() -> Self;
259}

261/// Values that carry no CPython reference and may therefore cross threads.
262///
263/// Every parallel entry point in the port is bounded by this. Because the
264/// supertrait chain reaches the private seal, no crate can add an impl.
265pub trait GilFree: PropValue + Clone + Send + Sync {}
266
267/// Widening to `f64`.
268///
269/// `Into<f64>` cannot serve: there is no `impl From<i64> for f64` (it is
270/// lossy), and `int64_t` is graph-tool's most-used scalar property type. A
271/// design bounded on `Into<f64>` silently ships a scalar axis with `i64`
272/// missing.
273pub trait ToF64: Copy {
274    /// Widen, accepting the precision loss that `i64 -> f64` can incur.
275    fn to_f64(self) -> f64;
276}

278/// Arithmetic-capable scalar members: `{uint8_t, int16_t, int32_t, int64_t, double}`.
279pub trait Scalar: PropValue + GilFree + Copy + ToF64 + PartialOrd {}

281/// Opaque carrier for `long double` and the element of `vector<long double>`.
282///
283/// Storage and round-trip only: no arithmetic, no [`ToF64`], no [`Scalar`].
284/// The x86-64 `long double` is 80-bit extended stored in 16 bytes; this
285/// preserves the bytes so `.gt` files written by graph-tool load and re-save
286/// unchanged.
287#[repr(transparent)]
288#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Default)]
289pub struct LongDouble(pub [u8; 16]);
290
291// `sizeof(long double) == 16` on every ABI graph-tool builds `.gt` files on
292// (x86-64 System V: 80-bit extended, padded to 16; AArch64: true binary128,
293// also 16). The payload is the *storage*, so a narrower carrier would
294// truncate a value on load and write a different one back out.
295//
296// Alignment deliberately is **not** matched: C++ gives `long double` a
297// 16-byte alignment, `[u8; 16]` has 1. Nothing here reinterprets the bytes as
298// a float -- that is the whole point of the opaque carrier -- and the size
299// equality is what keeps `Vec<LongDouble>`'s element stride equal to the
300// C++ array's.
301const _: () = assert!(size_of::<LongDouble>() == 16);
302
303macro_rules! impl_scalar_value {
304    ($($t:ty => $k:ident),+ $(,)?) => {$(
305        impl sealed::Sealed for $t {}
306        impl PropValue for $t { const KIND: ValueKind = ValueKind::$k; type Elem = $t; }
307        impl Zeroed for $t { #[inline] fn zero() -> Self { 0 as $t } }
308        impl GilFree for $t {}
309        impl ToF64 for $t { #[inline] fn to_f64(self) -> f64 { self as f64 } }
310        impl Scalar for $t {}
311    )+};
312}
313impl_scalar_value!(u8 => Bool, i16 => I16, i32 => I32, i64 => I64, f64 => F64);

359/// The Python member of the universe.
360///
361/// `!Send` and `!Sync` **by construction**. pyo3's `Py<T>` is unconditionally
362/// `Send + Sync` (`pyo3-0.22.6/src/instance.rs:943-944`), and both `Clone`
363/// and `Drop` do refcount work without requiring a `Python<'py>` token, so a
364/// bare handle in a property map reaches rayon with nothing to stop it. The
365/// `PhantomData<*mut ()>` is what turns
366/// `graph_properties_copy.cc:35-42`'s inverted `is_python` predicate -- which
367/// drops the GIL and enables `#pragma omp parallel` in exactly the
368/// object-to-object case -- into a compile error.
369#[cfg(feature = "python")]
370pub struct PyValue {
371    obj: pyo3::Py<pyo3::PyAny>,
372    _not_send: std::marker::PhantomData<*mut ()>,
373}
374
375#[cfg(feature = "python")]
376impl PyValue {
377    /// Wrap an owned handle.
378    pub fn new(obj: pyo3::Py<pyo3::PyAny>) -> Self {
379        PyValue {
380            obj,
381            _not_send: std::marker::PhantomData,
382        }
383    }
384    /// Borrow under a GIL token.
385    pub fn bind<'py>(&self, py: pyo3::Python<'py>) -> pyo3::Bound<'py, pyo3::PyAny> {
386        self.obj.bind(py).clone()
387    }
388    /// Duplicate the handle. Requires the token, so the incref is sound.
389    pub fn clone_ref(&self, py: pyo3::Python<'_>) -> Self {
390        PyValue::new(self.obj.clone_ref(py))
391    }
392    /// `Py_None`. This is the only way to make a "default" Python value, and
393    /// it needs the interpreter -- which is why [`Zeroed`] is not implemented.
394    pub fn none(py: pyo3::Python<'_>) -> Self {
395        PyValue::new(py.None())
396    }
397}
