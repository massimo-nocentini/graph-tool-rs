14/// Whether an operation may leave the calling thread.
15pub trait Mode: sealed::Sealed {
16    /// Whether work in this mode may leave the calling thread.
17    ///
18    /// This is the *witness*: the resolution of the lattice is a `const` that
19    /// a `const` block can assert, so "`f64 -> f64` is parallel" is proved at
20    /// compile time rather than inferred by reading the source. See the
21    /// `const` block below [`copy_prop`].
22    const PARALLEL: bool;
23    /// The mode's own name, for diagnostics.
24    const NAME: &'static str;
25}

27/// Parallel: no CPython reference is touched, the GIL is released.
28pub struct Par;
29/// Serial: a CPython reference is touched, the GIL is held throughout.
30pub struct Seq;
31
32impl sealed::Sealed for Par {}
33impl sealed::Sealed for Seq {}
34impl Mode for Par {
35    const PARALLEL: bool = true;
36    const NAME: &'static str = "Par";
37}
38impl Mode for Seq {
39    const PARALLEL: bool = false;
40    const NAME: &'static str = "Seq";
41}

43/// The greatest lower bound of two modes. `Seq` absorbs.
44///
45/// There is no `||` here for a human to invert.
46pub trait Meet<Rhs: Mode>: Mode {
47    /// The meet.
48    type Out: Mode;
49}
50impl Meet<Par> for Par {
51    type Out = Par;
52}
53impl Meet<Seq> for Par {
54    type Out = Seq;
55}
56impl Meet<Par> for Seq {
57    type Out = Seq;
58}
59impl Meet<Seq> for Seq {
60    type Out = Seq;
61}

63/// A mode that may be given to `V`.
64///
65/// This is the bound on [`ModeOf::Mode`], and it is what makes the policy
66/// table below *unable* to be mis-edited: `Par` permits only a value that can
67/// cross a thread, so writing `impl ModeOf for PyValue { type Mode = Par; }`
68/// is `error[E0277]: the trait bound `PyValue: Sync` is not satisfied` rather
69/// than a silently faster and unsound build. `tests/ui/u30_par_mode_for_py_value.rs`
70/// pins that diagnostic.
71///
72/// The C++ has no analogue because its predicate is a `bool` computed inside
73/// the kernel body (`graph_properties_copy.cc:35-42`): there is nothing for a
74/// compiler to check, and in fact the expression is wrong in both directions.
75pub trait Allows<V: ?Sized>: Mode {}
76impl<V: ?Sized + Sync + Send> Allows<V> for Par {}
77impl<V: ?Sized> Allows<V> for Seq {}

79/// Which mode a member of the value universe forces.
80///
81/// Note this is *not* an open policy table: [`PropValue`] is sealed in
82/// gt-core, so the only types that can appear here are the fifteen. A newtype
83/// over a raw `pyo3::Py<PyAny>` cannot be given `Mode = Par`, because it
84/// cannot be a property value at all.
85pub trait ModeOf: PropValue {
86    /// The mode this member forces.
87    ///
88    /// `Allows<Self>` is not decoration: it is the second of [DESIGN](gt_core::design) §7's
89    /// three mechanisms, moved from the strategy impls up onto the table
90    /// itself, so that the table cannot be edited into unsoundness even by
91    /// someone who never reads the strategy bounds.
92    type Mode: Mode + Allows<Self>;
93}

95/// The fourteen members whose value can be moved with no token in hand.
96///
97/// A **local** marker, deliberately, rather than a use of gt-core's
98/// [`GilFree`](gt_core::prop::GilFree). The two say the same thing about the
99/// same fourteen types, but only a trait defined *here* lets the two
100/// [`CopyValue`] impls below be proved disjoint: for an upstream trait rustc
101/// must assume "upstream crates may add a new impl of `GilFree` for `PyValue`
102/// in future versions" and rejects the pair as overlapping (`E0119`), whereas
103/// nothing outside gt-py can ever implement this one.
104///
105/// It is minted by the same macro as [`ModeOf`], from the same token list, so
106/// the two tables cannot drift apart -- which is the failure mode
107/// `value_types` / `type_names[]` (`graph_properties.hh:61-76`) has by
108/// construction.
109pub trait TokenFree: PropValue {}

114        impl TokenFree for $t {}
115    )+};
116}

131    Vec<String>,

144/// Copying one value of the universe into another, **with the token**.
145///
146/// ## Why [`ConvertFrom`](gt_core::prop::ConvertFrom) does not serve the
147/// serial half
148///
149/// `convert<To, From>` (`value_convert.hh:73-105`) has three arms that touch
150/// CPython: `is_same_v<To, From>` on `python::object`, `is_same_v<To, object>`
151/// and `is_same_v<From, object>`. All three need a live interpreter, and
152/// `ConvertFrom::convert_from(&S) -> Result<T, _>` has no token in its
153/// signature to give them -- it cannot, because gt-core builds without pyo3 at
154/// all. gt-core's diagonal therefore reaches the interpreter by *re-acquiring*
155/// the token (`Python::with_gil` inside `PyCell`'s `Clone`), once per value.
156///
157/// That is sound, and [DESIGN](gt_core::design) §7 says so explicitly. It is also exactly the
158/// work `Seq` exists to avoid: `Seq::copy` was handed a `Python<'_>` and holds
159/// it for the whole loop, so re-deriving one per element is a TLS probe and a
160/// guard construction per property-map entry, paid for nothing. The
161/// token-carrying conversion therefore lives here, next to the mode that
162/// guarantees the token exists, and its two impl families are precisely the
163/// lattice's two halves:
164///
165/// * [`TokenFree`] -> [`TokenFree`] delegates to gt-core's `ConvertFrom` and
166///   ignores the token, because no CPython reference is touched;
167/// * `PyValue -> PyValue` is `Py::clone_ref`, which *takes* the token.
168///
169/// Note what this trait cannot be used for: [`Par::copy`](Par) does **not**
170/// bound on it, and could not. Its body is the inside of
171/// [`detach`](crate::gil::detach), where `Python<'_>` -- being `!Send` -- cannot be
172/// captured. The parallel path is therefore bounded on the tokenless
173/// `ConvertFrom` by construction, not by convention.
174pub trait CopyValue<S: PropValue>: PropValue + Sized {
175    /// Produce the target value for one key.
176    fn copy_value(py: Python<'_>, s: &S) -> Result<Self, PropError>;
177}
178
179/// Every conversion whose **target** is GIL-free, delegating to gt-core's
180/// tokenless `ConvertFrom`. Disjoint from the two impls below because
181/// [`TokenFree`] is local to this crate and [`PyValue`] is not one of its
182/// members; see its docs for why that distinction is the whole reason it
183/// exists.
184///
185/// `S` is deliberately **not** bounded on [`TokenFree`], only on `PropValue`.
186/// That is what admits `S = PyValue`, i.e. `convert`'s
187/// `is_same_v<From, boost::python::object>` arm (`value_convert.hh:86-107`):
188/// `extract<To>` with an elementwise fallback for vector targets. gt-core
189/// carries that arm (`prop/convert.rs`, `feature = "python"`), so the column
190/// is reachable here the moment the target is one of the fourteen. The token
191/// is ignored on this path because gt-core re-derives its own; see the note on
192/// the `PyValue` target impl below for what that costs and why it is still the
193/// right trade for now.
194impl<S, T> CopyValue<S> for T
195where
196    S: PropValue,
197    T: TokenFree + gt_core::prop::ConvertFrom<S>,
198{
199    #[inline]
200    fn copy_value(_py: Python<'_>, s: &S) -> Result<T, PropError> {
201        T::convert_from(s)
202    }
203}

205/// `convert`'s `is_same_v<To, boost::python::object>` arm
206/// (`value_convert.hh:82-85`): `boost::python::object(v)` never fails, and
207/// neither does its counterpart here.
208///
209/// **Cost, recorded rather than hidden.** gt-core's `ConvertFrom` has no token
210/// in its signature -- it cannot, because gt-core builds without pyo3 -- so it
211/// re-acquires one with `Python::with_gil` per value, which on this path is a
212/// TLS probe and a guard construction per property-map entry that `Seq` was
213/// already handed a token to avoid. The re-entrant acquisition is sound
214/// ([DESIGN](gt_core::design) section 7), and correctness does not depend on removing it. The
215/// saving needs a token parameter on gt-core's `FromAny`, which is gt-core's
216/// call to make; when it exists, this body becomes the direct
217/// `ToPyObject::to_object(py, s)` and this note goes away.
218impl<S> CopyValue<S> for PyValue
219where
220    S: TokenFree,
221    PyValue: gt_core::prop::ConvertFrom<S>,
222{
223    #[inline]
224    fn copy_value(_py: Python<'_>, s: &S) -> Result<PyValue, PropError> {
225        <PyValue as gt_core::prop::ConvertFrom<S>>::convert_from(s)
226    }
227}

229/// `convert`'s `is_same_v<To, From>` arm (`value_convert.hh:78-81`) for the
230/// Python member: the C++ returns `(const To&)v`, i.e. a new reference to the
231/// *same* object, and so does this. The incref is sound because the token is
232/// a parameter.
233impl CopyValue<PyValue> for PyValue {
234    #[inline]
235    fn copy_value(py: Python<'_>, s: &PyValue) -> Result<PyValue, PropError> {
236        Ok(s.clone_ref(py))
237    }

240/// How to copy one property map into another.
241///
242/// The `Python<'_>` token is a **parameter of the trait method**, not
243/// something the call site decides to pass. So `Par` cannot be implemented
244/// without `allow_threads` and `Seq` cannot run without the token held.
245///
246/// This closes the half of the C++ truth table that a serial-versus-parallel
247/// switch alone leaves open: for `int -> int`, `graph_properties_copy.cc`
248/// *holds* the GIL across a long serial loop, blocking every other Python
249/// thread for the duration.
250pub trait CopyStrategy<S, T, K: IdTag> {
251    /// Copy `src` into `dst`.
252    fn copy(
253        py: Python<'_>,
254        bound: Bound<K>,
255        src: &DenseProp<S, K>,
256        dst: &mut DenseProp<T, K>,
257    ) -> Result<(), PropError>;
258}
259
260/// graph-tool's parallel cut-off: `__openmp_min_thresh = 300`
261/// (`openmp.cc:20`), read by `graph_properties_copy.cc:42` as
262/// `num_vertices(g) > get_openmp_min_thresh()`.
263///
264/// The *sense* of the port's use differs from the C++'s in one respect worth
265/// stating: here the threshold governs only whether rayon is entered. The GIL
266/// is released either way, because releasing it is a property of the value
267/// types and not of the workload size. In the C++ the two decisions are welded
268/// to one another through `is_python`, which is how defect #31 -- the GIL held
269/// across a long serial `int -> int` loop -- comes about.
270const MIN_PAR_ITEMS: usize = 300;
271
272/// Items per parallel chunk.
273///
274/// A **constant**, deliberately: [DESIGN](gt_core::design) §8 forbids any partition that is a
275/// function of the pool size, so that a run with one worker and a run with
276/// sixteen perform the same work in the same grouping. Nothing here folds
277/// floating point, so the grouping cannot move a result; what it fixes is
278/// *which* error a failing copy reports (see [`Par::copy`](Par)).
279const PAR_GRAIN: usize = 256;
280
281/// The two runs a copy operates on, each checked against the bound.
282///
283/// Both maps must already be sized: [DESIGN](gt_core::design) §5 and U32 put the sizing at the
284/// Python entry point, *before* `detach`, which is the only place it can
285/// happen for a map of [`PyValue`] (the member has no context-free default, so
286/// `DenseProp::sized_for` is not available to it). Handing back the short run
287/// instead -- `get_unchecked(size = 0)` (`dispatch.hh:171-177`) -- is the
288/// defect `Bound` exists to prevent.
289fn runs<'s, 'd, S, T, K: IdTag>(
290    bound: Bound<K>,
291    src: &'s DenseProp<S, K>,
292    dst: &'d mut DenseProp<T, K>,
293) -> Result<(&'s [S], &'d mut [T]), PropError> {
294    let n = bound.len();
295    check(dst.graph(), dst.len(), bound)?;
296    check(src.graph(), src.len(), bound)?;
297    Ok((&src.as_slice()[..n], &mut dst.as_mut_slice()[..n]))
298}
299
300/// One map's half of [`runs`]' check: identity first, then length.
301///
302/// Identity first because it is the one the C++ never makes: the Python guard
303/// at `__init__.py:3200` compares *filtered counts*, which agree in exactly
304/// the case that makes `graph_copy.cc:72`'s out-of-bounds write silent.
305#[inline]
306fn check<K: IdTag>(
307    owner: gt_core::ids::GraphId,
308    have: usize,
309    bound: Bound<K>,
310) -> Result<(), PropError> {
311    if owner != bound.graph() {
312        return Err(PropError::WrongGraph {
313            owner: owner.get(),
314            expected: bound.graph().get(),
315        });
316    }
317    if have < bound.len() {
318        return Err(PropError::Undersized {
319            have,
320            need: bound.len(),
321        });
322    }
323    Ok(())
324}

326/// Convert one contiguous run, keeping the **lowest-indexed** failure.
327///
328/// The whole run is executed even after a failure. `parallel_loop_no_spawn<true>`
329/// (`parallel_util.hh:399-437`) instead sets a thread-private `skip` flag, so
330/// which iterations are abandoned is the OpenMP schedule and the set of work
331/// actually performed is not reproducible. Here it is a function of the inputs
332/// alone -- and, because `Seq` runs the full cover too, it does not depend on
333/// which arm of the lattice was selected either.
334#[inline]
335fn convert_run<S, T>(base: usize, s: &[S], d: &mut [T]) -> Option<(usize, PropError)>
336where
337    T: gt_core::prop::ConvertFrom<S>,
338{
339    let mut first: Option<(usize, PropError)> = None;
340    for (i, (slot, v)) in d.iter_mut().zip(s).enumerate() {
341        match T::convert_from(v) {
342            Ok(c) => *slot = c,
343            Err(e) => {
344                if first.is_none() {
345                    first = Some((base + i, e));
346                }
347            }
348        }
349    }
350    first
351}

353/// Lowest index wins. Commutative and associative, so rayon's tree shape --
354/// which *is* a function of the pool size -- cannot be observed in the result.
355#[inline]
356fn earlier(
357    a: Option<(usize, PropError)>,
358    b: Option<(usize, PropError)>,
359) -> Option<(usize, PropError)> {
360    match (a, b) {
361        (None, other) | (other, None) => other,
362        (Some(x), Some(y)) => Some(if x.0 <= y.0 { x } else { y }),
363    }
364}

366impl<S, T, K: IdTag> CopyStrategy<S, T, K> for Seq
367where
368    S: PropValue,
369    T: CopyValue<S>,
370{
371    /// Holds the token for the whole loop.
372    ///
373    /// `py` is threaded through every single conversion, so there is no
374    /// formulation of this body that quietly drops the GIL -- and equally, no
375    /// formulation that reaches rayon, since the token cannot cross a thread.
376    ///
377    /// **Deviation from the skeleton, recorded here because it is load
378    /// bearing.** The skeleton bounded this impl on
379    /// `S: PropValue + Clone, T: PropValue + ConvertFrom<S>`.
380    ///
381    /// `S: Clone` is not satisfiable by the one member this impl exists to
382    /// serve. [`PyValue`] is `!Clone` *by construction* — `Py_INCREF` needs a
383    /// token, so duplication is `clone_ref(py)` and nothing else — which means
384    /// `copy_prop::<PyValue, PyValue, _>` could not even be **named** under
385    /// the skeleton's bounds, and the entire `Seq` half of the lattice would
386    /// have been dead code. It is dropped, and nothing here needs it: a
387    /// conversion reads `&S`.
388    ///
389    /// `ConvertFrom<PyValue>` *is* satisfiable, but only by re-acquiring a
390    /// token per value; [`CopyValue`] is the same conversion using the one
391    /// already in hand. See its docs.
392    fn copy(
393        py: Python<'_>,
394        bound: Bound<K>,
395        src: &DenseProp<S, K>,
396        dst: &mut DenseProp<T, K>,
397    ) -> Result<(), PropError> {
398        let (s, d) = runs(bound, src, dst)?;
399        let mut first: Option<(usize, PropError)> = None;
400        for (i, (slot, v)) in d.iter_mut().zip(s).enumerate() {
401            match T::copy_value(py, v) {
402                Ok(c) => *slot = c,
403                Err(e) => {
404                    if first.is_none() {
405                        first = Some((i, e));
406                    }
407                }
408            }
409        }
410        match first {
411            None => Ok(()),
412            Some((_, e)) => Err(e),
413        }
414    }
415}

417impl<S, T, K: IdTag> CopyStrategy<S, T, K> for Par
418where
419    S: PropValue + Clone + Sync,
420    T: PropValue + gt_core::prop::ConvertFrom<S> + Send,
421    K: Send + Sync,
422{
423    /// Body is `py.allow_threads(|| ... par_iter_mut() ...)`.
424    ///
425    /// The token is *consumed* by [`detach`] and cannot enter the closure, so
426    /// the conversion this body can reach is the tokenless `ConvertFrom` and
427    /// nothing else. That is why the `Par` half needs no discipline from its
428    /// author: the alternative does not typecheck.
429    ///
430    /// Note the release is unconditional, including on the short path below
431    /// [`MIN_PAR_ITEMS`]. Defect #31 is the opposite choice -- `int -> int`
432    /// keeps the GIL for the whole loop (`graph_properties_copy.cc:38`)
433    /// because `is_python` is `true` there, so every other Python thread in
434    /// the process stalls for the duration of a copy that touches no Python
435    /// object at all.
436    fn copy(
437        py: Python<'_>,
438        bound: Bound<K>,
439        src: &DenseProp<S, K>,
440        dst: &mut DenseProp<T, K>,
441    ) -> Result<(), PropError> {
442        let (s, d) = runs(bound, src, dst)?;
443        let failure = detach(py, move || {
444            if d.len() < MIN_PAR_ITEMS {
445                return convert_run(0, s, d);
446            }
447            d.par_chunks_mut(PAR_GRAIN)
448                .zip(s.par_chunks(PAR_GRAIN))
449                .enumerate()
450                .map(|(c, (dc, sc))| convert_run(c * PAR_GRAIN, sc, dc))
451                .reduce(|| None, earlier)
452        });
453        match failure {
454            None => Ok(()),
455            Some((_, e)) => Err(e),
456        }
457    }
458}

497/// Run a GIL-free region.
498///
499/// `F: Send` means nothing carrying a `Python<'py>` or a `Bound<'py, _>` can
500/// be captured in. This is [`GILRelease`](https://git.skewed.de) with the
501/// bound attached, and the bound is the part graph-tool cannot express:
502/// `GILRelease(bool)` (`gil_release.hh:31-35`) saves the thread state only if
503/// `PyGILState_Check()`, so it is a **silent no-op on any thread that does not
504/// already hold the GIL** -- which is every OpenMP worker.
505#[inline]
506pub fn detach<R, F>(py: Python<'_>, f: F) -> R
507where
508    F: Send + FnOnce() -> R,
509    R: Send,
510{
511    py.allow_threads(f)
512}

579        assert_eq!(chunks(PAR_GRAIN), 1);

584    /// Lowest index wins, whichever order the tree folds in.
585    #[test]
586    fn the_reported_failure_is_the_lowest_indexed_one() {
587        let a = Some((7usize, PropError::NotReadable));
588        let b = Some((3usize, PropError::NotWritable));
589        assert_eq!(earlier(a.clone(), b.clone()), b);
590        assert_eq!(earlier(b.clone(), a.clone()), b);
591        assert_eq!(earlier(None, a.clone()), a);
592        assert_eq!(earlier(a.clone(), None), a);
593        assert_eq!(earlier(None, None), None);
594        // A tie keeps the left operand, so the fold is total rather than
595        // merely "usually deterministic".
596        let c = Some((3usize, PropError::NotReadable));
597        assert_eq!(earlier(c.clone(), b.clone()), c);
598    }
