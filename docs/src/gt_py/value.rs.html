1//! The shipped value subsets, and the coverage guarantee.
2//!
3//! Each invocation of
4//! [`value_subset!`](gt_core::value_subset) is the single source of truth for
5//! one dispatch axis: its member list, its accepted-kind array, its
6//! narrowing, its kernel trait *with the bound the kernel body needs*, and its
7//! exhaustive match, all from one token list.
8//!
9//! The axes mirror `graph_properties.hh:78-104` and `:166-231`. The one
10//! deliberate divergence is that `long double` is not in
11//! [`ScalarV`](ScalarV): it has no arithmetic in this port, so the
12//! [`Scalar`](gt_core::prop::Scalar) bound excludes it -- which is the
13//! mechanism working as intended rather than a gap.
14
15use gt_core::prop::{GilFree, PropValue, Scalar, ValueKind};
16use gt_core::value_subset;
17
18value_subset!(
19    /// Arithmetic scalars: `{uint8_t, int16_t, int32_t, int64_t, double}`.
20    pub ScalarV, ScalarKernel, "value", Scalar,
21    { Bool => u8, I16 => i16, I32 => i32, I64 => i64, F64 => f64 }
22);
23
24value_subset!(
25    /// Integral scalars: `integer_types` (`graph_properties.hh:83`).
26    pub IntegerV, IntegerKernel, "value", Scalar,
27    { Bool => u8, I16 => i16, I32 => i32, I64 => i64 }
28);
29
30value_subset!(
31    /// Floating scalars actually usable in arithmetic. `long double` is
32    /// excluded; see the module docs.
33    pub FloatingV, FloatingKernel, "value", Scalar,
34    { F64 => f64 }
35);
36
37value_subset!(
38    /// Every member except the Python one: the 14 that may cross a thread.
39    pub GilFreeV, GilFreeKernel, "value", GilFree,
40    {
41        Bool => u8, I16 => i16, I32 => i32, I64 => i64, F64 => f64,
42        LongDouble => gt_core::prop::LongDouble,
43        Str => String,
44        VecBool => Vec<u8>, VecI16 => Vec<i16>, VecI32 => Vec<i32>,
45        VecI64 => Vec<i64>, VecF64 => Vec<f64>,
46        VecLongDouble => Vec<gt_core::prop::LongDouble>,
47        VecStr => Vec<String>,
48    }
49);
50
51value_subset!(
52    /// The whole universe, including the Python member. Kernels over this
53    /// axis are serial: [`PyValue`](gt_core::prop::PyValue) is `!Send`.
54    pub AnyV, AnyKernel, "value", PropValue,
55    {
56        Bool => u8, I16 => i16, I32 => i32, I64 => i64, F64 => f64,
57        LongDouble => gt_core::prop::LongDouble,
58        Str => String,
59        VecBool => Vec<u8>, VecI16 => Vec<i16>, VecI32 => Vec<i32>,
60        VecI64 => Vec<i64>, VecF64 => Vec<f64>,
61        VecLongDouble => Vec<gt_core::prop::LongDouble>,
62        VecStr => Vec<String>,
63        PyObject => gt_core::prop::PyValue,
64    }
65);
66
67/// Every member reachable from at least one shipped subset.
68///
69/// This is the mitigation for the one residual hole in the dispatch story.
70/// A subset's `narrow` necessarily ends in a `_ => Err` arm, so adding a
71/// sixteenth [`ValueKind`] would not break the build at every call site; it
72/// would make that member unsupported at runtime wherever no subset lists it.
73/// The test below turns that into a build failure at the one place it can be.
74pub const fn covered(kind: ValueKind) -> bool {
75    matches!(
76        kind,
77        ValueKind::Bool
78            | ValueKind::I16
79            | ValueKind::I32
80            | ValueKind::I64
81            | ValueKind::F64
82            | ValueKind::LongDouble
83            | ValueKind::Str
84            | ValueKind::VecBool
85            | ValueKind::VecI16
86            | ValueKind::VecI32
87            | ValueKind::VecI64
88            | ValueKind::VecF64
89            | ValueKind::VecLongDouble
90            | ValueKind::VecStr
91            | ValueKind::PyObject
92    )
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    #[test]
100    fn every_value_kind_is_reachable_from_some_subset() {
101        for k in ValueKind::ALL {
102            assert!(covered(k), "{k:?} ({}) is in no shipped subset", k.name());
103            assert!(AnyV::narrow(k).is_ok(), "{k:?} is not in AnyV");
104        }
105    }
106
107    #[test]
108    fn narrowing_outside_a_subset_names_the_accepted_set() {
109        let e = ScalarV::narrow(ValueKind::VecStr).unwrap_err();
110        assert_eq!(e.offered, ValueKind::VecStr);
111        assert_eq!(e.accepted, ScalarV::ACCEPTED);
112    }
113}
