1//! The importable extension module.
2//!
3//! Every entry point follows the same three-step shape, and the order matters:
4//!
5//! 1. take the `Python<'py>` token and borrow the graph and the property maps;
6//! 2. size every map with
7//!    [`DenseProp::sized_for`](gt_core::prop::DenseProp::sized_for) -- read-only
8//!    maps too, because graph-tool's checked map auto-grows on read
9//!    (`fast_vector_property_map.hh:132-136`; the plan's `:129` is three lines
10//!    above the `soft_reserve(i + 1)` that actually does it), so `g.vp.x[v]`
11//!    on a never-written map must keep returning the default;
12//! 3. hand the sized views to a kernel inside [`detach`](crate::gil::detach),
13//!    whose `F: Send` bound is what keeps the token out of the region.
14//!
15//! Step 2 is the two-phase reserve-then-hand-out that `_get_any`
16//! (`graph_tool/__init__.py:363-373`) implements in nine lines of Python. It
17//! does not disappear; it moves from Python into one place in Rust, where the
18//! type system makes forgetting it impossible rather than merely unlikely.
19//!
20//! ## Where the fifteenth member went
21//!
22//! `#[pyclass]` requires `Send`, and [`PyValue`] is `!Send` by construction
23//! ([DESIGN](gt_core::design) section 7). That is not an obstacle to work around; it is the
24//! guarantee arriving at the boundary, and the only honest way to spell it is
25//! **two classes**:
26//!
27//! * [`PyPropertyMap`] carries the fourteen GIL-free members. It is `Send`,
28//!   so CPython may hand it to any thread, and its kernels run inside
29//!   [`detach`](crate::gil::detach).
30//! * [`PyObjectPropertyMap`] carries `python::object`. It is
31//!   `#[pyclass(unsendable)]`, because the store it holds is `!Send` and the
32//!   compiler will not be persuaded otherwise, and every one of its loops
33//!   holds the token.
34//!
35//! Folding both into one class would have forced `unsendable` onto the
36//! fourteen members that do not need it, which is the pessimisation
37//! `graph_properties_copy.cc:35-42` makes in the other direction -- and one
38//! `unsendable` class is a smaller lie than fourteen.
39//!
40//! ## Errors
41//!
42//! No entry point here panics on a caller error. `GraphError`, `PropError` and
43//! `DispatchError` become a `PyErr` at the boundary and nothing above relies
44//! on `catch_unwind`, which is what [DESIGN](gt_core::design) section 10 promises when
45//! `[profile.release] panic = "abort"` removes unwinding entirely. The
46//! exception classes are graph-tool's own translation table
47//! (`graph_bind.cc:63-76`): `ValueException -> ValueError`,
48//! `GraphException -> RuntimeError`.
49

70/// `GraphException -> RuntimeError`, `ValueException -> ValueError`
71/// (`graph_bind.cc:67-73`).
72///
73/// A descriptor that does not name a live element is the condition
74/// `graph_tool/__init__.py:2149` raises `ValueError("Invalid vertex index")`
75/// for; the two exhaustion cases and the invariant breach have no graph-tool
76/// analogue at all -- `Vertex` is also the index type there and a graph grown
77/// past `null_vertex()` silently collides with it (`graph_adjacency.hh:924`).
78fn graph_err(e: GraphError) -> PyErr {
79    match e {
80        GraphError::NoSuchVertex(_) | GraphError::NoSuchEdge(_) => {
81            PyValueError::new_err(e.to_string())
82        }
83        GraphError::EdgeIdSpaceExhausted { .. }
84        | GraphError::VertexIdSpaceExhausted { .. }
85        | GraphError::Invariant(_) => PyRuntimeError::new_err(e.to_string()),
86    }
87}

89/// Every `PropError` is a `ValueException` in graph-tool's vocabulary: the six
90/// sites that can raise one (`graph_properties.hh:266, :319, :447, :457`,
91/// `graph_copy.cc:99, :181`) all throw that type.
92fn prop_err(e: PropError) -> PyErr {
93    PyValueError::new_err(e.to_string())
94}

96/// The narrowing failure. graph-tool's counterpart is `DispatchNotFound`,
97/// whose message is "This is a graph_tool bug. :-(" (`dispatch.hh:86-88`);
98/// this one names the offered member and the accepted set.
99fn dispatch_err(e: DispatchError) -> PyErr {
100    PyValueError::new_err(e.to_string())
101}

103/// A key outside the index space of its own graph.
104///
105/// graph-tool has no such check on this path: the checked map's `operator[]`
106/// does `soft_reserve(i + 1)` (`fast_vector_property_map.hh:132-136`) and
107/// happily allocates `i + 1` slots for any `i`, so `g.vp.x[1 << 40]` on a
108/// five-vertex graph is an allocation, not an error. **Deliberate
109/// divergence:** a [`Bound`](gt_core::bound::Bound) is minted only by a graph,
110/// so the only length a map here can grow to is that graph's index bound, and
111/// a key past it is an `IndexError`. The auto-grow that matters -- a
112/// never-written key *inside* the bound reading back as the default -- is
113/// preserved exactly, by sizing before every read.
114fn key_error<K: IdTag>(key: usize, bound: usize) -> PyErr {
115    PyIndexError::new_err(format!(
116        "{} index {key} is outside this graph's index bound of {bound}",
117        K::NAME
118    ))
119}

125/// How one member of the value universe crosses the boundary.
126///
127/// Twelve of the fourteen GIL-free members delegate to pyo3's own conversions.
128/// [`LongDouble`] and `Vec<LongDouble>` cannot: the type is an *opaque
129/// 16-byte payload* with no arithmetic ([DESIGN](gt_core::design) D7), so the only lossless
130/// Python spelling is `bytes`. Exposing it as a `float` would round-trip an
131/// 80-bit extended value through a `double` and write a different one back to
132/// the `.gt` file, which is the whole failure the opaque carrier exists to
133/// prevent.
134trait PyBridge: Sized {
135    /// Hand the value to Python.
136    fn to_py(&self, py: Python<'_>) -> PyObject;
137    /// Take a value from Python, or fail with a `PyErr`.
138    fn from_py(ob: &Bound<'_, PyAny>) -> PyResult<Self>;
139}
140
141macro_rules! bridge_native {
142    ($($t:ty),+ $(,)?) => {$(
143        impl PyBridge for $t {
144            #[inline]
145            fn to_py(&self, py: Python<'_>) -> PyObject {
146                self.clone().into_py(py)
147            }
148            #[inline]
149            fn from_py(ob: &Bound<'_, PyAny>) -> PyResult<Self> {
150                ob.extract::<$t>()
151            }
152        }
153    )+};
154}
155
156bridge_native!(
157    u8,
158    i16,
159    i32,
160    i64,
161    f64,
162    String,
163    Vec<u8>,
164    Vec<i16>,
165    Vec<i32>,
166    Vec<i64>,
167    Vec<f64>,
168    Vec<String>,
169);

171impl PyBridge for LongDouble {
172    fn to_py(&self, py: Python<'_>) -> PyObject {
173        PyBytes::new_bound(py, &self.0).unbind().into_any()
174    }
175
176    fn from_py(ob: &Bound<'_, PyAny>) -> PyResult<Self> {
177        let raw = ob.downcast::<PyBytes>()?.as_bytes();
178        // `sizeof(long double) == 16` on every ABI graph-tool writes `.gt`
179        // files on; a shorter or longer payload is not a `long double`.
180        let exact: [u8; 16] = raw.try_into().map_err(|_| {
181            PyValueError::new_err(format!(
182                "long double is an opaque 16-byte payload, got {} bytes",
183                raw.len()
184            ))
185        })?;
186        Ok(LongDouble(exact))
187    }
188}
189
190impl PyBridge for Vec<LongDouble> {
191    fn to_py(&self, py: Python<'_>) -> PyObject {
192        let items: Vec<PyObject> = self.iter().map(|v| v.to_py(py)).collect();
193        PyList::new_bound(py, items).unbind().into_any()
194    }
195
196    fn from_py(ob: &Bound<'_, PyAny>) -> PyResult<Self> {
197        let mut out = Vec::new();
198        for item in ob.iter()? {
199            out.push(LongDouble::from_py(&item?)?);
200        }
201        Ok(out)
202    }
203}

209/// Generates the fourteen-armed store from one token list, in the same shape
210/// -- and for the same reason -- as
211/// [`value_subset!`](gt_core::value_subset): the member list, the
212/// construction, the key accessors and the erasure all come from one place, so
213/// a member cannot be added to one and forgotten in another.
214macro_rules! gil_free_store {
215    ($($variant:ident => $ty:ty),+ $(,)?) => {
216        /// One property map of one key space, with its member chosen at
217        /// runtime.
218        enum Store<K: IdTag> {
219            $( $variant(DenseProp<$ty, K>), )+
220        }
221
222        impl<K: IdTag> Store<K> {
223            /// An empty map of `kind` belonging to `graph`, or `None` for the
224            /// one member this store cannot hold.
225            fn new(graph: GraphId, kind: ValueKind) -> Option<Self> {
226                match kind {
227                    $( ValueKind::$variant => Some(Store::$variant(DenseProp::new(graph))), )+
228                    ValueKind::PyObject => None,
229                }
230            }
231
232            /// Which member this map stores.
233            fn kind(&self) -> ValueKind {
234                match self { $( Store::$variant(_) => ValueKind::$variant, )+ }
235            }
236
237            /// Slots currently allocated. Not the graph's size: it is the
238            /// number `sized_for` has grown the store to so far.
239            fn allocated(&self) -> usize {
240                match self { $( Store::$variant(m) => m.len(), )+ }
241            }
242
243            /// Read one key, **sizing first**.
244            ///
245            /// This is the auto-grow-on-read of
246            /// `fast_vector_property_map.hh:132-136`, kept honest: a key that
247            /// was never written reads back as `Zeroed::zero`, which is what
248            /// `(*_store)[i]` returns after `soft_reserve(i + 1)` there.
249            fn get(
250                &mut self,
251                py: Python<'_>,
252                bound: IndexBound<K>,
253                key: usize,
254            ) -> PyResult<PyObject> {
255                match self { $( Store::$variant(m) => {
256                    let view = m.sized_for(bound).map_err(prop_err)?;
257                    let slot = view
258                        .as_slice()
259                        .get(key)
260                        .ok_or_else(|| key_error::<K>(key, bound.len()))?;
261                    Ok(PyBridge::to_py(slot, py))
262                } )+ }
263            }
264
265            /// Write one key.
266            ///
267            /// The conversion runs *before* the store is sized, so a rejected
268            /// value leaves the map exactly as it was.
269            fn set(
270                &mut self,
271                bound: IndexBound<K>,
272                key: usize,
273                value: &Bound<'_, PyAny>,
274            ) -> PyResult<()> {
275                match self { $( Store::$variant(m) => {
276                    let v = <$ty as PyBridge>::from_py(value)?;
277                    let mut view = m.sized_for(bound).map_err(prop_err)?;
278                    let slot = view
279                        .as_mut_slice()
280                        .get_mut(key)
281                        .ok_or_else(|| key_error::<K>(key, bound.len()))?;
282                    *slot = v;
283                    Ok(())
284                } )+ }
285            }
286
287            /// Grow the store to `bound`, filling new slots with the member's
288            /// default.
289            ///
290            /// **This is step 2, and it is an entry-point obligation rather
291            /// than a kernel one.** It has to be: a kernel is bounded by its
292            /// axis' trait -- [`Scalar`] for this one -- and none of the axis
293            /// traits implies [`Zeroed`](gt_core::prop::Zeroed), because
294            /// `Zeroed` is exactly the property the fifteenth member does not
295            /// have. Only here, where the concrete type is still in hand, is
296            /// the default constructible at all. `gt_dispatch::pmap`
297            /// (`dispatch.hh:171-177`) has the same information available and
298            /// throws it away by calling `get_unchecked()` with no size.
299            fn size(&mut self, bound: IndexBound<K>) -> Result<(), PropError> {
300                match self { $( Store::$variant(m) => { m.sized_for(bound)?; } )+ }
301                Ok(())
302            }
303
304            /// Erase to the dispatch boundary.
305            ///
306            /// One `TypeId` comparison downstream, where `dispatch.hh:248-268`
307            /// needs up to three `any_cast` probes per candidate.
308            fn erase(&mut self) -> AnyMap<'_, K> {
309                match self { $( Store::$variant(m) => AnyMap::new(m), )+ }
310            }
311        }
312    };
313}

332/// Which index space a map is keyed on. graph-tool's `key_type()` answers
333/// `"v"`, `"e"` or `"g"`; the graph-wide case is not a property *map* here,
334/// so there are two.
335enum KeyStore {
336    /// Keyed on vertices.
337    Vertex(Store<VertexTag>),
338    /// Keyed on edges.
339    Edge(Store<EdgeTag>),
340}

345    Vertex(DenseProp<PyValue, VertexTag>),

351// The scalar kernel
352// ---------------------------------------------------------------------------

354/// Sum a scalar-valued map, in a GIL-free region.
355///
356/// Step 3 of the shape, and only step 3. The map reached this point already
357/// sized by [`Store::size`], so `view()` -- which is fallible precisely
358/// because `&self` cannot grow -- succeeds. Skipping the sizing would make
359/// this kernel fail on a map the caller has never written, where graph-tool
360/// returns the default (`fast_vector_property_map.hh:132-136`); doing it here
361/// instead is not possible, because [`Scalar`] does not imply
362/// [`Zeroed`](gt_core::prop::Zeroed).
363///
364/// The `Python<'py>` token lives in the kernel *struct*, never in the closure:
365/// [`detach`]'s `F: Send` bound is what makes that distinction enforceable,
366/// and the slice is `Send` only because `V: Scalar` implies
367/// [`GilFree`](gt_core::prop::GilFree).
368struct SumScalar<'py, K: IdTag> {
369    py: Python<'py>,
370    bound: IndexBound<K>,
371}
372
373impl<K: IdTag> ScalarKernel<K> for SumScalar<'_, K> {
374    type Out = Result<f64, PropError>;
375
376    fn call<V: Scalar>(self, map: &mut DenseProp<V, K>) -> Self::Out {
377        let view = map.view(self.bound)?;
378        let data: &[V] = view.as_slice();
379        // The Python member cannot reach this line: it is not a `Scalar`, and
380        // `&[PyValue]` is not `Send` in any case.
381        Ok(detach(self.py, move || {
382            data.iter().map(|v| v.to_f64()).sum::<f64>()
383        }))
384    }
385}

391/// A graph, owned by Python.
392#[pyclass(name = "Graph", module = "graph_tool._rust")]
393pub struct PyGraph {
394    inner: gt_core::adj::Graph,
395}
396
397#[pymethods]
398impl PyGraph {
399    /// An empty graph.
400    #[new]
401    fn new() -> Self {
402        PyGraph {
403            inner: gt_core::adj::Graph::new(),
404        }
405    }
406
407    /// Number of vertices.
408    fn num_vertices(&self) -> usize {
409        self.inner.num_vertices()
410    }
411
412    /// Number of edges.
413    fn num_edges(&self) -> usize {
414        self.inner.num_edges()
415    }
416
417    /// Size of the edge index space, which exceeds
418    /// [`num_edges`](Self::num_edges) once an edge has been removed.
419    ///
420    /// This is `g.edge_index_range` (`graph_tool/__init__.py:369`), and it is
421    /// the number an edge property map must be sized to -- not the count.
422    fn edge_index_range(&self) -> usize {
423        self.inner.edge_bound().len()
424    }
425
426    /// Append an isolated vertex.
427    fn add_vertex(&mut self) -> PyResult<usize> {
428        self.inner
429            .add_vertex()
430            .map(|v| v.index())
431            .map_err(graph_err)
432    }
433
434    /// Add an edge, returning its dense index.
435    fn add_edge(&mut self, s: usize, t: usize) -> PyResult<usize> {
436        let sv = VertexId::new(s).ok_or_else(|| PyValueError::new_err(invalid_vertex(s)))?;
437        let tv = VertexId::new(t).ok_or_else(|| PyValueError::new_err(invalid_vertex(t)))?;
438        self.inner
439            .add_edge(sv, tv)
440            .map(|e| e.id().index())
441            .map_err(graph_err)
442    }
443
444    /// Remove an edge by index.
445    ///
446    /// Takes the index, never a descriptor: the endpoints come from the slot
447    /// table, so there is no caller-supplied orientation that can disagree
448    /// with storage. `remove_edge(e, g)` (`graph_adjacency.hh:1310-1312`)
449    /// trusts the descriptor's own `s`/`t` and decrements `_n_edges` at the
450    /// call site, which is the mechanism behind `clear_vertex`'s double
451    /// decrement (`:1403-1413`).
452    fn remove_edge(&mut self, e: usize) -> PyResult<()> {
453        let id = EdgeId::new(e).ok_or_else(|| PyValueError::new_err(invalid_edge(e)))?;
454        self.inner.remove_edge(id).map_err(graph_err)
455    }
456
457    /// A new vertex property map of the named type.
458    ///
459    /// `type` is `type_names[]` exactly (`graph_properties.hh:72-76`), and an
460    /// unknown spelling raises `ValueError("Invalid property type: " + type)`
461    /// as `new_property` does (`graph_python_interface.hh:689`). The alias
462    /// table -- `"int"`, `"float"`, `"object"` -- belongs to `_type_alias`
463    /// (`graph_tool/__init__.py:210-229`), which runs *above* this boundary
464    /// and never reaches `type_names[]`.
465    fn new_vertex_property(slf: &Bound<'_, PyGraph>, value_type: &str) -> PyResult<PyObject> {
466        new_property(slf, value_type, false)
467    }
468
469    /// A new edge property map of the named type.
470    fn new_edge_property(slf: &Bound<'_, PyGraph>, value_type: &str) -> PyResult<PyObject> {
471        new_property(slf, value_type, true)
472    }
473
474    /// Sum a scalar-valued property map over this graph's index space.
475    ///
476    /// The worked example of the three-step shape. Note that the map is
477    /// read-only to the kernel and is **sized anyway** ([DESIGN](gt_core::design) section 5),
478    /// and that the bound comes from the map's own key type exactly as
479    /// `_get_any` chooses between `num_vertices` and `edge_index_range`
480    /// (`graph_tool/__init__.py:363-373`).
481    fn sum_property(&self, py: Python<'_>, prop: &mut PyPropertyMap) -> PyResult<f64> {
482        // The only fallible step of the dispatch, and it happens before any
483        // map is touched.
484        let member = ScalarV::narrow(prop.kind).map_err(dispatch_err)?;
485        match &mut prop.store {
486            KeyStore::Vertex(s) => {
487                let bound = self.inner.vertex_bound();
488                // Step 2, for a map the kernel only reads.
489                s.size(bound).map_err(prop_err)?;
490                let mut erased = s.erase();
491                member
492                    .dispatch(&mut erased, SumScalar { py, bound })
493                    .map_err(prop_err)
494            }
495            KeyStore::Edge(s) => {
496                let bound = self.inner.edge_bound();
497                s.size(bound).map_err(prop_err)?;
498                let mut erased = s.erase();
499                member
500                    .dispatch(&mut erased, SumScalar { py, bound })
501                    .map_err(prop_err)
502            }
503        }
504    }
505}

518/// Shared body of the two `new_*_property` methods.
519///
520/// The fifteenth member gets a different class, for the reason in the module
521/// docs; `value_type()` and `key_type()` read the same on both, so Python code
522/// that only asks those two questions does not have to know.
523fn new_property(slf: &Bound<'_, PyGraph>, value_type: &str, edge: bool) -> PyResult<PyObject> {
524    let py = slf.py();
525    let kind = ValueKind::from_name(value_type)
526        .ok_or_else(|| PyValueError::new_err(format!("Invalid property type: {value_type}")))?;
527    let graph_id = slf.try_borrow()?.inner.graph_id();
528    let handle: Py<PyGraph> = slf.clone().unbind();
529
530    if kind == ValueKind::PyObject {
531        let store = if edge {
532            ObjStore::Edge(DenseProp::new(graph_id))
533        } else {
534            ObjStore::Vertex(DenseProp::new(graph_id))
535        };
536        return Ok(Py::new(
537            py,
538            PyObjectPropertyMap {
539                graph: handle,
540                store,
541            },
542        )?
543        .into_any());
544    }
545
546    // `Store::new` is generic in `K`, and which one is fixed here and nowhere
547    // else -- so a vertex map can never later be sized from an edge bound.
548    // It returns `None` only for `PyObject`, which the branch above took;
549    // spelled as an error rather than an `expect` so that a sixteenth member
550    // turns a coverage gap into a `ValueError` and not a panic at the
551    // boundary (`gt_core::design` defect table, row 34).
552    let missing = || PyValueError::new_err(format!("Invalid property type: {value_type}"));
553    let store = if edge {
554        KeyStore::Edge(Store::new(graph_id, kind).ok_or_else(missing)?)
555    } else {
556        KeyStore::Vertex(Store::new(graph_id, kind).ok_or_else(missing)?)
557    };
558    Ok(Py::new(
559        py,
560        PyPropertyMap {
561            kind,
562            graph: handle,
563            store,
564        },
565    )?
566    .into_any())
567}

573/// A property map, owned by Python.
574///
575/// Carries any of the fourteen GIL-free members. The fifteenth lives in
576/// [`PyObjectPropertyMap`]; see the module docs for why that is two classes
577/// and not one.
578#[pyclass(name = "PropertyMap", module = "graph_tool._rust")]
579pub struct PyPropertyMap {
580    kind: gt_core::prop::ValueKind,
581    graph: Py<PyGraph>,
582    store: KeyStore,
583}
584
585impl PyPropertyMap {
586    /// The graph's two index bounds, read and released before anything else
587    /// happens.
588    ///
589    /// A [`Bound`](gt_core::bound::Bound) is two `Copy` words carrying the
590    /// graph's identity, so holding the `PyRef` across the rest of the call is
591    /// unnecessary -- and holding it would turn any re-entrant mutation from
592    /// Python into a borrow error rather than letting it proceed.
593    fn bounds(&self, py: Python<'_>) -> PyResult<(IndexBound<VertexTag>, IndexBound<EdgeTag>)> {
594        let g = self.graph.bind(py).try_borrow()?;
595        Ok((g.inner.vertex_bound(), g.inner.edge_bound()))
596    }
597}

649/// A `python::object` property map, owned by Python.
650///
651/// `unsendable` is not a concession: [`PyValue`] is `!Send` by construction
652/// ([DESIGN](gt_core::design) section 7, mechanism 3), so a store of them is `!Send` and
653/// `#[pyclass]` will not accept it without the marker. That is the inverted
654/// `is_python` predicate of `graph_properties_copy.cc:35-42` -- which releases
655/// the GIL and runs `#pragma omp parallel` in exactly the object-to-object
656/// case -- showing up as a type error instead of a data race.
657#[pyclass(unsendable, name = "ObjectPropertyMap", module = "graph_tool._rust")]
658pub struct PyObjectPropertyMap {
659    graph: Py<PyGraph>,
660    store: ObjStore,
661}
662
663impl PyObjectPropertyMap {
664    /// As [`PyPropertyMap::bounds`].
665    fn bounds(&self, py: Python<'_>) -> PyResult<(IndexBound<VertexTag>, IndexBound<EdgeTag>)> {
666        let g = self.graph.bind(py).try_borrow()?;
667        Ok((g.inner.vertex_bound(), g.inner.edge_bound()))
668    }
669}
670
671/// Size a Python-valued map and read one key.
672///
673/// `sized_for_with`, not `sized_for`: `Py<PyAny>` has no `Default` and the
674/// default `python::object` is `Py_None`, which needs the interpreter. That is
675/// why [`Zeroed`](gt_core::prop::Zeroed) is not a supertrait of
676/// [`PropValue`](gt_core::prop::PropValue) -- see [DESIGN](gt_core::design) section 13.1.
677fn obj_get<K: IdTag>(
678    py: Python<'_>,
679    map: &mut DenseProp<PyValue, K>,
680    bound: IndexBound<K>,
681    key: usize,
682) -> PyResult<PyObject> {
683    let view = map
684        .sized_for_with(bound, || PyValue::none(py))
685        .map_err(prop_err)?;
686    let slot = view
687        .as_slice()
688        .get(key)
689        .ok_or_else(|| key_error::<K>(key, bound.len()))?;
690    Ok(slot.bind(py).unbind())
691}
692
693/// Size a Python-valued map and write one key.
694fn obj_set<K: IdTag>(
695    py: Python<'_>,
696    map: &mut DenseProp<PyValue, K>,
697    bound: IndexBound<K>,
698    key: usize,
699    value: &Bound<'_, PyAny>,
700) -> PyResult<()> {
701    let mut view = map
702        .sized_for_with(bound, || PyValue::none(py))
703        .map_err(prop_err)?;
704    let slot = view
705        .as_mut_slice()
706        .get_mut(key)
707        .ok_or_else(|| key_error::<K>(key, bound.len()))?;
708    *slot = PyValue::new(value.clone().unbind());
709    Ok(())
710}
711
712/// Apply a Python callable to every slot, serially, with the token held.
713///
714/// This is the `Seq` arm of the lattice arriving at the boundary, and it is
715/// the *correct* behaviour that `graph_properties_copy.cc:35-42` intended
716/// before the predicate was inverted. There is no `detach` here and there
717/// cannot be one: the closure captures `f`, a `Bound<'py, PyAny>`, and
718/// [`detach`](crate::gil::detach)'s `F: Send` bound rejects it.
719///
720/// An exception from `f` propagates as a `PyErr` through `?`. Nothing
721/// unwinds, so `[profile.release] panic = "abort"` changes nothing about this
722/// path -- which is the claim [DESIGN](gt_core::design) section 10 makes for the whole
723/// boundary.
724fn obj_map<K: IdTag>(
725    py: Python<'_>,
726    map: &mut DenseProp<PyValue, K>,
727    bound: IndexBound<K>,
728    f: &Bound<'_, PyAny>,
729) -> PyResult<()> {
730    let mut view = map
731        .sized_for_with(bound, || PyValue::none(py))
732        .map_err(prop_err)?;
733    for slot in view.as_mut_slice() {
734        let out = f.call1((slot.bind(py),))?;
735        *slot = PyValue::new(out.unbind());
736    }
737    Ok(())
738}

740#[pymethods]
741impl PyObjectPropertyMap {
742    /// Always `"python::object"`.
743    fn value_type(&self) -> &'static str {
744        ValueKind::PyObject.name()
745    }
746
747    /// `"v"` or `"e"`.
748    fn key_type(&self) -> &'static str {
749        match self.store {
750            ObjStore::Vertex(_) => "v",
751            ObjStore::Edge(_) => "e",
752        }
753    }
754
755    /// Slots the store has grown to so far.
756    fn allocated(&self) -> usize {
757        match &self.store {
758            ObjStore::Vertex(m) => m.len(),
759            ObjStore::Edge(m) => m.len(),
760        }
761    }
762
763    /// Read one key. A never-written key inside the bound reads back as
764    /// `None`.
765    fn __getitem__(&mut self, py: Python<'_>, key: usize) -> PyResult<PyObject> {
766        let (vb, eb) = self.bounds(py)?;
767        match &mut self.store {
768            ObjStore::Vertex(m) => obj_get(py, m, vb, key),
769            ObjStore::Edge(m) => obj_get(py, m, eb, key),
770        }
771    }
772
773    /// Write one key.
774    fn __setitem__(
775        &mut self,
776        py: Python<'_>,
777        key: usize,
778        value: &Bound<'_, PyAny>,
779    ) -> PyResult<()> {
780        let (vb, eb) = self.bounds(py)?;
781        match &mut self.store {
782            ObjStore::Vertex(m) => obj_set(py, m, vb, key, value),
783            ObjStore::Edge(m) => obj_set(py, m, eb, key, value),
784        }
785    }
786
787    /// Replace every slot with `f(slot)`, serially.
788    fn map_values(&mut self, py: Python<'_>, f: &Bound<'_, PyAny>) -> PyResult<()> {
789        let (vb, eb) = self.bounds(py)?;
790        match &mut self.store {
791            ObjStore::Vertex(m) => obj_map(py, m, vb, f),
792            ObjStore::Edge(m) => obj_map(py, m, eb, f),
793        }
794    }
795}

832/// Python-level smoke test.
