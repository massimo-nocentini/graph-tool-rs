1//! Tabulated `lgamma` and `log`.
2//!
3//! `support/cache.hh:55-57` declares **three global mutable**
4//! `std::vector<double>` (`__safelog_cache`, `__xlogx_cache`,
5//! `__lgamma_cache`); `init_cache` (`:83-95`) resizes them at `:90` and fills
6//! them inside a `#pragma omp parallel for` at `:91-93` while `get_cached`
7//! (`:73-81`) reads them at `:80`. Worse, `get_cached` (`:74`),
8//! `safelog_fast` (`:106-111`) and `lgamma_fast` (`:139-144`) are annotated
9//! `[[gnu::const]]`, which promises the optimiser that they read no memory --
10//! licensing hoisting across the very resize that races with them. That is
11//! defect #34 in [DESIGN](gt_core::design) §17.
12//!
13//! (The skeleton's original text cited `:56-58`, `:83-91` and `:73`, and the
14//! bench header cited `:55-57`, `:84-92` and `:72-73`. The constructs are the
15//! same three; the offsets above were re-read out of the 3.8 tree and are the
16//! ones this module is written against. U22 owns the reconciliation, so this
17//! is it.)
18//!
19//! Here the tables are one immutable, `Sync` value, threaded explicitly. There
20//! is no global to race on and no attribute to lie with.
21//!
22//! # What is tabulated, and what is not
23//!
24//! Two tables, not three. `__xlogx_cache` is dropped: [`Cache::xlogx`] is
25//! `x * safelog(x)`, one multiply on top of a table this cache already holds,
26//! and a third array of the same length buys nothing but a second stream of
27//! cache misses. The values are bit-identical to the C++'s, because
28//! `xlogx(y) = y * safelog(y)` is exactly how `cache.hh:121-124` computes the
29//! entries it stores.
30//!
31//! The restricted-partition table `__q_cache` (`int_part.cc:30-54`) *is* kept,
32//! but triangular and capped -- see [`Cache::log_q`].
33//!
34//! # The slow paths are on the hot path
35//!
36//! graph-tool deliberately feeds `lgamma` out-of-domain values and installs an
37//! `ignore_error` policy (`cache.hh:37-49`) so that they come back as NaN
38//! instead of throwing. Everything below returns NaN or an infinity where
39//! `boost::math` with that policy would, and **nothing here panics** -- there
40//! is no `assert`, no unwrap, and every table read goes through
41//! `slice::get`, so no `panic_bounds_check` survives into the kernels.
42

47/// Ceiling on the dimension of the restricted-partition table.
48///
49/// `state.hh:242` and `planted_partition/spec.hh:127` both call
50/// `init_q_cache(std::min(std::max(2 * int(_E), 100), 10000))`, and
51/// `__q_cache` is a **square** `boost::multi_array`: at that ceiling the C++
52/// allocates `10001 * 10001` doubles, 800 MB, to store a table that is zero
53/// above the diagonal.
54///
55/// Here the table is triangular (`k <= n`) and this constant caps `n`, so the
56/// worst case is `1025 * 1026 / 2` doubles, 4.2 MB. Above it, `log_q` falls
57/// through to the same asymptotic `log_q_approx` (`int_part.cc:129-139`) the
58/// C++ uses whenever `n` exceeds whatever `init_q_cache` was last called with.

147/// Tabulated special functions, immutable after construction.
148///
149/// `Send + Sync` by construction -- every field is a `Box<[f64]>` or a
150/// `usize`, and no method takes `&mut self`. One `Cache` is shared by every
151/// thread in a sweep; see `tests/u22_cache.rs::cache_is_sync_and_threads_agree`.
152#[derive(Clone, Debug)]
153pub struct Cache {
154    /// `lgamma[i] == ln Gamma(i + 1) == ln(i!)`. Length is a power of two.
155    lgamma: Box<[f64]>,
156    /// `safelog[i] == ln(i)`, and `safelog[0] == 0` (`cache.hh:99-104`).
157    safelog: Box<[f64]>,
158    /// Triangular `log q(n, k)` for `n <= q_max`, `k <= n`; row `n` starts at
159    /// `n * (n + 1) / 2`.
160    q: Box<[f64]>,
161    /// Largest `n` for which `q` holds an exact value.
162    q_max: usize,
163}
164
165impl Cache {
166    /// Tabulate up to `n`.
167    ///
168    /// The tables are sized `get_next_size(n + 1)` (`cache.hh:64-71`), i.e.
169    /// the smallest power of two that makes index `n` valid, exactly as
170    /// `init_cache` (`:86-90`) sizes them. Unlike the C++ this happens once,
171    /// in a constructor, and never again: there is no `init_*_fast` call to
172    /// forget and no resize to race with a reader.
173    ///
174    /// The fill is parallel, as `cache.hh:91` is; it is a pure `map` over a
175    /// disjoint index range, so the result does not depend on the thread
176    /// count.
177    pub fn build(n: usize) -> Self {
178        let len = n
179            .saturating_add(1)
180            .checked_next_power_of_two()
181            .unwrap_or(usize::MAX);
182
183        let lgamma: Vec<f64> = (0..len)
184            .into_par_iter()
185            .map(|i| lgamma1p_slow(i as f64))
186            .collect();
187        let safelog: Vec<f64> = (0..len)
188            .into_par_iter()
189            .map(|i| safelog_slow(i as f64))
190            .collect();
191
192        let q_max = n.min(Q_MAX);
193        Self {
194            lgamma: lgamma.into_boxed_slice(),
195            safelog: safelog.into_boxed_slice(),
196            q: build_q(q_max),
197            q_max,
198        }
199    }
200
201    /// The number of tabulated entries; index `i < len()` is a table hit.
202    #[inline]
203    #[must_use]
204    pub fn len(&self) -> usize {
205        self.lgamma.len()
206    }
207
208    /// Always false -- `build` allocates at least one entry.
209    #[inline]
210    #[must_use]
211    pub fn is_empty(&self) -> bool {
212        self.lgamma.is_empty()
213    }
214
215    /// `lgamma(x + 1)`, falling back to a real `lgamma` out of range.
216    ///
217    /// graph-tool installs an `ignore_error` policy on `boost::math::lgamma`
218    /// (`cache.hh:37-49`) because the SBM deliberately feeds it out-of-domain
219    /// values; this matches that behaviour rather than panicking:
220    ///
221    /// * a pole (`x + 1` a non-positive integer, so `x` a negative integer)
222    ///   gives NaN, as `raise_pole_error` under `ignore_error` does;
223    /// * `-inf` gives NaN (domain error), `+inf` gives `+inf`;
224    /// * NaN in, NaN out.
225    ///
226    /// `get_cached` (`cache.hh:73-81`) indexes the table with an *integral*
227    /// `x`. This takes `f64`, so it checks integrality before indexing: a
228    /// truncating index would answer `lgamma1p(2.5)` with `ln(2!)`, which is
229    /// not what any C++ call site asks for, because every C++ call site passes
230    /// an integer.
231    #[inline]
232    pub fn lgamma1p(&self, x: f64) -> f64 {
233        if x >= 0.0 {
234            let i = x as usize;
235            if i as f64 == x
236                && let Some(&v) = self.lgamma.get(i)
237            {
238                return v;
239            }
240        }
241        lgamma1p_slow(x)
242    }
243
244    /// `log(x)`, returning 0 at `x == 0` (`safelog_fast`, `cache.hh:106-111`).
245    #[inline]
246    pub fn safelog(&self, x: f64) -> f64 {
247        if x >= 0.0 {
248            let i = x as usize;
249            if i as f64 == x
250                && let Some(&v) = self.safelog.get(i)
251            {
252                return v;
253            }
254        }
255        safelog_slow(x)
256    }
257
258    /// `x * log(x)`, zero at `x == 0` (`xlogx_fast`, `cache.hh:126-131`).
259    ///
260    /// The C++ keeps a third table for this. Here it is one multiply over the
261    /// `safelog` table, which is how the C++ computes the entries it stores
262    /// (`cache.hh:119-124`), so the values agree bit for bit.
263    #[inline]
264    pub fn xlogx(&self, x: f64) -> f64 {
265        x * self.safelog(x)
266    }
267
268    /// `log(binom(n, k))`. `support/util.hh:42-47` (`lbinom_fast`).
269    ///
270    /// The three guard cases and the association of the two subtractions are
271    /// copied from `:44-46`; the association is load-bearing because these are
272    /// large cancelling terms and regrouping them changes the last bits.
273    #[inline]
274    pub fn lbinom(&self, n: f64, k: f64) -> f64 {
275        if n == 0.0 || k == 0.0 || k >= n {
276            return 0.0;
277        }
278        (self.lgamma1p(n) - self.lgamma1p(k)) - self.lgamma1p(n - k)
279    }
280
281    /// `log q(n, k)`, the restricted-partition count. `support/int_part.hh:39-52`.
282    ///
283    /// `q(n, k)` is the number of partitions of `n` into at most `k` parts
284    /// (`q_rec`, `int_part.cc:61-72`). Exact from the table while
285    /// `n <= Q_MAX`, and the asymptotic `log_q_approx` (`:129-139`) above it --
286    /// the same two-armed structure as `int_part.hh:49-51`, with a triangular
287    /// and capped table in place of a square unbounded one.
288    #[inline]
289    pub fn log_q(&self, n: usize, k: usize) -> f64 {
290        // `int_part.hh:43-48`, minus the `n < 0` arm: `n` is unsigned here.
291        let k = k.min(n);
292        if n == 0 && k == 0 {
293            return 0.0;
294        }
295        if k == 0 {
296            return f64::NEG_INFINITY;
297        }
298        if n <= self.q_max
299            && let Some(&v) = self.q.get(n * (n + 1) / 2 + k)
300        {
301            return v;
302        }
303        self.log_q_approx(n, k)
304    }
305
306    /// `log_q_approx`, `int_part.cc:129-139`.
307    fn log_q_approx(&self, n: usize, k: usize) -> f64 {
308        let nf = n as f64;
309        let kf = k as f64;
310        if kf < nf.powf(0.25) {
311            // `log_q_approx_small`, `int_part.cc:110-113`.
312            return self.lbinom(nf - 1.0, kf - 1.0) - self.lgamma1p(kf);
313        }
314        let u = kf / nf.sqrt();
315        let v = get_v(u);
316        let lf = v.ln()
317            - (-(-v).exp() * (1.0 + u * u / 2.0)).ln_1p() / 2.0
318            - LN_2 * 3.0 / 2.0
319            - u.ln()
320            - PI.ln();
321        let g = 2.0 * v / u - u * (-(-v).exp()).ln_1p();
322        lf - nf.ln() + nf.sqrt() * g
323    }
324}

330/// `safelog`, `cache.hh:97-104`: `log(x)`, and 0 at `x == 0`.
331#[inline]
332fn safelog_slow(x: f64) -> f64 {
333    if x == 0.0 { 0.0 } else { x.ln() }
334}

336/// `ln Gamma(x + 1)`, computed rather than looked up.
337///
338/// Three regimes, chosen so that the *relative* error is bounded everywhere
339/// and not merely the absolute one. `lgamma1p` has a zero at `x = 0` and
340/// another at `x = 1`; a plain `ln_gamma(x + 1.0)` is accurate to a few
341/// units in the last place of a quantity that is itself near zero there, so
342/// its relative error blows up. Inside `SERIES_WINDOW` of each zero the
343/// Taylor series is used instead: its leading term is `-EULER * x`, there is
344/// no cancellation, and the relative error stays at rounding.
345fn lgamma1p_slow(x: f64) -> f64 {
346    // Exact, and the two values the series and `ln_gamma` would only round to.
347    if x == 0.0 || x == 1.0 {
348        return 0.0;
349    }
350    if x.abs() <= SERIES_WINDOW {
351        return lgamma1p_series(x);
352    }
353    let t = x - 1.0;
354    if t.abs() <= SERIES_WINDOW {
355        // ln Gamma(2 + t) = ln(1 + t) + ln Gamma(1 + t); both terms have the
356        // sign of t, so the sum does not cancel.
357        return t.ln_1p() + lgamma1p_series(t);
358    }
359    ln_gamma(x + 1.0)
360}

362/// `ln Gamma(1 + x) = -gamma*x + sum_{k>=2} (-1)^k zeta(k) x^k / k`.
363///
364/// Only called for `|x| <= SERIES_WINDOW`, where it is converged to well
365/// below rounding.
366#[inline]
367fn lgamma1p_series(x: f64) -> f64 {
368    let mut p = 0.0;
369    for &c in ZETA_SERIES.iter().rev() {
370        p = p * x + c;
371    }
372    (-EULER) * x + x * x * p
373}

375/// `ln|Gamma(z)|` for real `z`, with graph-tool's `ignore_error` policy
376/// (`cache.hh:37-49`) at the poles and at the domain edge.
377///
378/// `boost::math::lgamma` under `pole_error<ignore_error>` returns NaN at the
379/// non-positive integers and under `domain_error<ignore_error>` returns NaN
380/// for `-inf`; both are reproduced here, and neither panics.
381fn ln_gamma(z: f64) -> f64 {
382    if z.is_nan() {
383        return z;
384    }
385    if z == f64::INFINITY {
386        return f64::INFINITY;
387    }
388    if z <= 0.0 {
389        // `-inf` is a domain error; a non-positive integer is a pole. Both are
390        // NaN under `ignore_error`.
391        if z == f64::NEG_INFINITY || z == z.floor() {
392            return f64::NAN;
393        }
394        // Reflection: ln|Gamma(z)| = ln(pi) - ln|sin(pi z)| - ln Gamma(1 - z).
395        // `sin(PI * z)` loses every bit of its argument for large |z|, so the
396        // range reduction is done first: |sin(pi z)| = sin(pi * frac(z)) with
397        // frac(z) in (0, 1).
398        let frac = z - z.floor();
399        let s = (PI * frac).sin();
400        return PI.ln() - s.ln() - ln_gamma(1.0 - z);
401    }
402    ln_gamma_lanczos(z)
403}
404
405/// Lanczos, `g = 607/128`, 15 terms, for `z > 0`.
406///
407/// `Gamma(z) = sqrt(2 pi) * base^(z - 1/2) * e^-base * A(z)` with
408/// `base = z + g - 1/2` and `A(z) = c0 + sum_k c_k / (z - 1 + k)`; taken in
409/// logs so that neither `base^(z - 1/2)` nor `e^-base` ever overflows.
410#[inline]
411fn ln_gamma_lanczos(z: f64) -> f64 {
412    let zm1 = z - 1.0;
413    let mut sum = LANCZOS[0];
414    for (i, &c) in LANCZOS.iter().enumerate().skip(1) {
415        sum += c / (zm1 + i as f64);
416    }
417    let base = zm1 + LANCZOS_G + 0.5;
418    LN_SQRT_2PI + (zm1 + 0.5) * base.ln() - base + sum.ln()
419}

421/// `log_sum_exp`, `support/util.hh:77-90`, with `handle_inf = true`.
422///
423/// The `a == b` arm is what makes `-inf + -inf` come out as `-inf` rather than
424/// NaN, which the `q` recurrence below depends on: every cell starts at
425/// `-inf`.
426#[inline]
427fn log_sum_exp(a: f64, b: f64) -> f64 {
428    if a == b {
429        return a + LN_2;
430    }
431    if a > b {
432        a + (b - a).exp().ln_1p()
433    } else {
434        b + (a - b).exp().ln_1p()
435    }
436}

438/// `get_v`, `int_part.cc:115-127`.
439///
440/// Fixed-point iteration on `v = u * sqrt(spence(exp(-v)))`. The C++ loop has
441/// no iteration bound; this one does, so that a `log_q` on a pathological `u`
442/// cannot hang a sweep. In practice it converges in under twenty.
443fn get_v(u: f64) -> f64 {
444    const EPSILON: f64 = 1e-8;
445    const MAX_ITER: usize = 1_000;
446    let mut v = u;
447    let mut delta = 1.0;
448    let mut iter = 0;
449    while delta > EPSILON && iter < MAX_ITER {
450        let n_v = u * spence((-v).exp()).sqrt();
451        delta = (n_v - v).abs();
452        v = n_v;
453        iter += 1;
454    }
455    v
456}

488/// `polevl`, `spence.cc:91-105`.
489#[inline]
490fn polevl(x: f64, coef: &[f64; 8]) -> f64 {
491    let mut ans = coef[0];
492    for &c in &coef[1..] {
493        ans = ans * x + c;
494    }

498/// Cephes dilogarithm, `spence.cc:107-150`, transcribed.
499///
500/// `spence(x) = -integral_1^x ln(t)/(t - 1) dt`, for `x >= 0`; NaN below zero,
501/// which is what the C++ returns too (`:113-114`).
502fn spence(x: f64) -> f64 {
503    let mut x = x;
504    if x < 0.0 {
505        return f64::NAN;
506    }
507    if x == 1.0 {
508        return 0.0;
509    }
510    if x == 0.0 {
511        return PI * PI / 6.0;
512    }
513
514    let mut flag = 0u32;
515    if x > 2.0 {
516        x = 1.0 / x;
517        flag |= 2;
518    }
519
520    let w = if x > 1.5 {
521        flag |= 2;
522        (1.0 / x) - 1.0
523    } else if x < 0.5 {
524        flag |= 1;
525        -x
526    } else {
527        x - 1.0
528    };
529
530    let mut y = -w * polevl(w, &SPENCE_A) / polevl(w, &SPENCE_B);
531
532    if flag & 1 != 0 {
533        y = (PI * PI) / 6.0 - x.ln() * (-x).ln_1p() - y;
534    }
535    if flag & 2 != 0 {
536        let z = x.ln();
537        y = -0.5 * z * z - y;
538    }
539    y
540}

542/// The triangular `log q(n, k)` table, `int_part.cc:32-54`.
543///
544/// The C++ builds a square `(n_max + 1)^2` array and fills the whole thing
545/// with `-inf` first; the half above the diagonal is never written and never
546/// read, because `log_q` clamps `k` to `n` before indexing. Storing only
547/// `k <= n` halves the memory and, more usefully, keeps a row contiguous.
548fn build_q(q_max: usize) -> Box<[f64]> {
549    let len = (q_max + 1) * (q_max + 2) / 2;
550    let mut q = vec![f64::NEG_INFINITY; len];
551    // `int_part.cc:42`.
552    q[0] = 0.0;
553    for n in 1..=q_max {
554        let row = n * (n + 1) / 2;
555        // `int_part.cc:46`: a partition of n into at most one part.
556        q[row + 1] = 0.0;
557        for k in 2..=n {
558            // `:49` -- q[n][k] is still -inf, so this is q[n][k-1].
559            let mut v = log_sum_exp(q[row + k], q[row + k - 1]);
560            // `:50-51`. `n >= k` holds for the whole loop.
561            let m = k.min(n - k);
562            let prev = (n - k) * (n - k + 1) / 2 + m;
563            v = log_sum_exp(v, q[prev]);
564            q[row + k] = v;
565        }
566    }
567    q.into_boxed_slice()
568}
