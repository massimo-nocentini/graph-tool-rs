1//! The Metropolis loop.
2//!
3//! Every member `MetropolisStateBase` (`loops/mcmc_loop.hh:36-75`) labels
4//! `// required` and then gives a *working* body is genuinely required here:
5//! `get_vlist()` returns an empty array (`:39`), `node_state()` returns 0
6//! (`:43`), `move_proposal()` returns 0 (`:46`), `virtual_move()` returns
7//! `{0,0}` (`:52-53`), `perform_move()` is a no-op (`:56`). Nothing is
8//! `= delete`d, so a state that misspells `virtual_move` compiles and reports
9//! `dS = 0, mP = 0` for every move -- and `metropolis_accept` (`:78-96`) then
10//! accepts unconditionally.
11

15/// The two numbers a move is judged by.
16///
17/// `base/mcmc.hh:112-128` returns `std::tuple<double, double>` and
18/// `mcmc_loop.hh:174` destructures it **positionally**. Transposing them now
19/// requires writing the wrong field name.
20#[derive(Clone, Copy, PartialEq, Debug)]
21pub struct MoveScore {
22    /// Change in the entropy functional.
23    pub d_entropy: f64,
24    /// Log of the Hastings ratio.
25    pub log_hastings: f64,
26}
27
28/// A proposal, with how many elementary steps it represents.
29///
30/// `mcmc_loop.hh:112-118` detects the multi-step case with a `constexpr`
31/// probe on `move_proposal`'s *return type*: `merge_split.hh:1201-1203`
32/// returns `std::tuple<size_t, size_t>` and the loop does
33/// `nattempts += nsteps`. A design whose `propose` returns a bare move cannot
34/// express merge-split, multiflip or multilevel at all. Single-step states
35/// write `Step { mv, nsteps: 1 }`, and nothing is read positionally.
36#[derive(Clone, Copy, PartialEq, Debug)]
37pub struct Step<M> {
38    /// The proposed move.
39    pub mv: M,
40    /// How many elementary steps it stands for.
41    pub nsteps: usize,
42}
43
44/// How a sweep orders its nodes.
45///
46/// Collapses `is_sequential()` x `is_deterministic()` (`mcmc_loop.hh:121`,
47/// `:196`) into one required method with three legal values. This deletes the
48/// misspelled `is_determinisitic` (`:60`) *and* the fourth, unrepresentable
49/// C++ combination -- `!sequential && deterministic`, which silently means
50/// "random order" because `:196` is guarded by `is_sequential()`.
51#[derive(Clone, Copy, PartialEq, Eq, Debug)]
52pub enum Schedule {
53    /// Sample nodes independently with replacement.
54    Random,
55    /// Visit every node once, in a fresh random order each sweep.
56    Shuffled,
57    /// Visit every node once, in a fixed order.
58    Alternating,
59}
60
61/// What a sweep did.
62#[derive(Clone, Copy, PartialEq, Debug, Default)]
63pub struct SweepResult {
64    /// Total entropy change.
65    pub d_entropy: f64,
66    /// Elementary steps attempted.
67    pub nattempts: usize,
68    /// Elementary steps accepted.
69    pub nmoves: usize,
70}

72/// A state the Metropolis loop can drive.
73pub trait MetropolisState: Send {
74    /// What a sweep iterates over.
75    type Node: Copy + Send + Sync;
76    /// What a move is.
77    type Move: Copy + PartialEq + Send + Sync;
78    /// The value meaning "no move".
79    const NULL_MOVE: Self::Move;
80
81    /// The node list, owned by the state so a sweep can shuffle it in place.
82    fn nodes(&mut self) -> &mut Vec<Self::Node>;
83    /// Sweeps to run.
84    fn n_iter(&self) -> usize;
85    /// Inverse temperature.
86    fn beta(&self) -> f64;
87    /// Node ordering.
88    fn schedule(&self) -> Schedule;
89    /// Elementary steps per sweep, for the [`Schedule::Random`] case.
90    /// `merge_split.hh:1470`'s `get_N()`.
91    fn steps_per_iter(&mut self) -> usize;
92    /// The node's current state, for the null-move test.
93    fn node_state(&self, v: Self::Node) -> Self::Move;
94    /// Propose.
95    fn propose<R: Rng + ?Sized>(&mut self, v: Self::Node, rng: &mut R) -> Step<Self::Move>;
96    /// Price.
97    fn score(&mut self, v: Self::Node, s: Self::Move) -> MoveScore;
98    /// Apply.
99    fn perform(&mut self, v: Self::Node, s: Self::Move);
100
101    // -- legitimate defaults: pure no-ops, no policy ------------------------
102
103    /// Skip a node entirely.
104    #[inline]
105    fn skip_node(&self, v: Self::Node) -> bool {
106        false
107    }
108    /// Per-sweep setup; returns an entropy offset.
109    #[inline]
110    fn init_iter<R: Rng + ?Sized>(&mut self, rng: &mut R) -> f64 {
111        0.0
112    }
113    /// Per-step bookkeeping.
114    #[inline]
115    fn after_step(&mut self, v: Self::Node, s: Self::Move) {}
116}

118/// The Metropolis acceptance test. `mcmc_loop.hh:78-96`.
119///
120/// Transcribed term for term:
121///
122/// ```text
123/// if (std::isinf(beta))          ->  beta.is_infinite()
124///     return dS < 0;             ->  score.d_entropy < 0.0
125/// double a = -dS * beta + mP;    ->  -score.d_entropy * beta + score.log_hastings
126/// if (a > 0) return true;        ->  a > 0.0
127/// std::uniform_real_distribution<> sample;   // [0, 1)
128/// return sample(rng) < exp(a);   ->  rng.random::<f64>() < a.exp()
129/// ```
130///
131/// `rand`'s `StandardUniform` for `f64` is `[0, 1)` (`distr/float.rs`,
132/// "Multiply-based method; 53 random bits; [0, 1) interval"), which is
133/// `std::uniform_real_distribution<double>`'s default range, so the zero
134/// measure endpoint convention agrees as well.
135///
136/// The generator is touched **only** on the `a <= 0` path: an infinite `beta`
137/// and an uphill-free move both return without drawing. A sweep's stream is
138/// therefore a function of which moves were proposed, exactly as in C++, and
139/// two implementations that disagree about when to draw desynchronise
140/// visibly rather than subtly.
141///
142/// `beta.is_infinite()` -- not `beta == f64::INFINITY` -- is `std::isinf`,
143/// which is also true at `-inf`. That is faithful and is *not* on the defect
144/// list: graph-tool's `beta` is an inverse temperature and the Python layer
145/// never passes a negative one, so the branch has no reachable disagreement
146/// to fix. Naming a different rule here would be inventing semantics.
147#[inline]
148pub fn accept<R: Rng + ?Sized>(score: MoveScore, beta: f64, rng: &mut R) -> bool {
149    if beta.is_infinite() {
150        // `mcmc_loop.hh:82-85`: the zero-temperature limit is a downhill test,
151        // and the Hastings ratio does not enter it.
152        return score.d_entropy < 0.0;
153    }
154
155    let a = -score.d_entropy * beta + score.log_hastings;
156    if a > 0.0 {
157        return true;
158    }
159    rng.random::<f64>() < a.exp()
160}

162/// Run `state.n_iter()` sweeps.
163///
164/// No `Python` token is in scope inside this function, so
165/// `mcmc_loop.hh:101`'s unconditional `GILRelease gil;` has no analogue to get
166/// wrong: the GIL is released once, outside, by the PyO3 boundary, and the
167/// `F: Send` bound on `allow_threads` means nothing holding a token can be
168/// captured into the region.
169pub fn mcmc_sweep<S, R>(state: &mut S, rng: &mut R) -> SweepResult
170where
171    S: MetropolisState,
172    R: Rng + ?Sized,
173{
174    let mut out = SweepResult::default();
175
176    // `:108`. Read once, as C++ does; `beta` is not a per-sweep quantity.
177    let beta = state.beta();
178    let n_iter = state.n_iter();
179
180    for _ in 0..n_iter {
181        let schedule = state.schedule();
182
183        // `:121-122`. The C++ guard is `is_sequential() && !is_deterministic()`
184        // -- and `is_deterministic()` is *not* the member `MetropolisStateBase`
185        // declares (`:60` spells it `is_determinisitic`), which is defect #20.
186        // One enum, three arms, nothing to misspell.
187        if schedule == Schedule::Shuffled {
188            state.nodes().shuffle(rng);
189        }
190
191        // `:134`. After the shuffle, so the generator is consumed in C++'s
192        // order and a fixed seed reproduces C++'s node permutation.
193        out.d_entropy += state.init_iter(rng);
194
195        // `:125-132`'s `get_N`, with the `constexpr` return-type probe
196        // (defect #25) replaced by the schedule. A sequential sweep visits the
197        // node list; a random sweep takes the state's own step count. A
198        // multi-step state that C++ drives through `get_N()` over a
199        // one-element `_vlist` (`merge_split.hh:1464-1472`) ports as a node
200        // list of that many entries, which is why this needs no third arm.
201        let n = match schedule {
202            Schedule::Random => state.steps_per_iter(),
203            Schedule::Shuffled | Schedule::Alternating => state.nodes().len(),
204        };
205
206        for vi in 0..n {
207            // `:138-139`. `nodes()` is re-borrowed per step, never hoisted:
208            // `perform` may push or pop nodes, and C++ re-evaluates
209            // `vlist[vi]` and `get_N()` at every step for the same reason.
210            let v = {
211                let nodes = state.nodes();
212                if nodes.is_empty() {
213                    break;
214                }
215                match schedule {
216                    // `uniform_sample(vlist, rng)`: with replacement.
217                    Schedule::Random => {
218                        let k = rng.random_range(0..nodes.len());
219                        nodes[k]
220                    }
221                    Schedule::Shuffled | Schedule::Alternating => {
222                        if vi >= nodes.len() {
223                            break;
224                        }
225                        nodes[vi]
226                    }
227                }
228            };
229
230            // `:141-142`.
231            if state.skip_node(v) {
232                continue;
233            }
234
235            let step = state.propose(v, rng);
236            let s = step.mv;
237
238            // `:168-173`, plus `base/mcmc.hh:105-106`. The second test is
239            // hoisted out of the state: every concrete C++ state ends its
240            // `move_proposal` with `if (s == node_state(v)) return _null_move;`
241            // and every one that forgot would price a self-move that
242            // `mcmc.hh:112-114` then has to special-case a second time. One
243            // place, and it is the reason `node_state` is a required member
244            // here instead of the verbose-print accessor it is at `:146`.
245            if s == S::NULL_MOVE || s == state.node_state(v) {
246                continue;
247            }
248
249            // `:176`. Named fields, so `d_entropy` and `log_hastings` cannot
250            // arrive transposed the way `std::tie(dS, mP)` can (defect #23).
251            let score = state.score(v, s);
252
253            // `:178`. `nsteps`, not 1 -- defect #25. In C++ this is a variable
254            // declared at `:124` and assigned only inside a `constexpr`
255            // branch of a lambda, so a state whose `move_proposal` returns a
256            // bare move silently reports one attempt for a batch of moves.
257            out.nattempts += step.nsteps;
258
259            // `:181-187`. `_force_accept` has no analogue: it is a compile
260            // time member of the state in C++ and belongs, if anywhere, to the
261            // state's own `score`.
262            if accept(score, beta, rng) {
263                state.perform(v, s);
264                out.nmoves += step.nsteps;
265                out.d_entropy += score.d_entropy;
266            }
267
268            // `:189`. Runs for accepted and rejected moves alike, but not for
269            // skipped or null ones.
270            state.after_step(v, s);
271        }
272
273        // `:196-197`. The reversal is what makes this schedule *alternating*:
274        // sweep 0 runs the list forwards, sweep 1 backwards, and a vertex at
275        // the end of one sweep is at the start of the next.
276        if schedule == Schedule::Alternating {
277            state.nodes().reverse();
278        }
279    }
280
281    out
282}
