1//! The state/loop layering.
2//!
3//! ## The rule that makes Potts-style bugs impossible ([DESIGN](gt_core::design) D8)
4//!
5//! graph-tool's layering is *mixin-from-below*: `SpecBase::Imp`
6//! (`base/spec_base.hh:58-130`) `= delete`s exactly three members --
7//! `move_vertex` (`:69`), `virtual_move` (`:70`), `entropy` (`:71`) -- and
8//! **defaults fourteen others with working bodies**. A misnamed deleted
9//! member is a compile error; a misnamed defaulted one silently inherits the
10//! base's behaviour.
11//!
12//! That is a live bug, verified: `potts/spec.hh:113-117` overrides the
13//! proposal with `uniform_int_distribution<group_t> sample(0, _q-1)`, then
14//! `:124` defines `get_move_lprob`. `grep -rn get_move_lprob src/` finds
15//! exactly that one line, called from nowhere; the loop calls
16//! `get_move_prob` (`base/mcmc.hh:123-124`), which Potts does not define, so
17//! it inherits `SpecBase::Imp::get_move_prob` (`spec_base.hh:88-109`) --
18//! `log(1. - d) - safelog_fast(B)`, the density of the *base's* sampler.
19//! The Hastings ratio does not correspond to the kernel being sampled.
20//!
21//! There is a **second instance one layer up**, not previously reported:
22//! `loops/mcmc_loop.hh:60` declares `constexpr bool is_determinisitic()` --
23//! misspelled -- while the loop calls `is_deterministic()` at `:121` and
24//! `:196`. The default is dead; every concrete state happens to define the
25//! correct spelling, so it survives only because nothing relies on it.
26//!
27//! The rule this module applies:
28//!
29//! > **A trait method may carry a default body only if that body is either a
30//! > pure no-op or derivable from other *required* methods of the same
31//! > trait.** A default that encodes policy is not a default; it is a required
32//! > method.
33//!
34//! And, stronger than the rule: [`GroupProposal::propose`] **returns its own
35//! forward density**, so there is no second place for the density to live.
36//! Requiring `log_move_prob` merely makes the misnaming a compile error;
37//! returning `log_fwd` from the sampler means a copy-pasted density in the
38//! same impl block cannot silently disagree with a hand-written sampler.
39

49/// A proposed move, carrying the density of the kernel that produced it.
50#[derive(Clone, Copy, PartialEq, Debug)]
51pub struct Proposal {
52    /// The group proposed.
53    pub to: Group,
54    /// `log q(r -> to | v)` under the kernel that sampled it.
55    pub log_fwd: f64,
56}

58/// The three members graph-tool `= delete`s, plus the scratch they need.
59///
60/// `Scratch` is the fix for a signature that cannot be implemented as written.
61/// `virtual_move(&self, ...)` looks right, but `blockmodel/state.hh:304` is
62/// `get_move_entries(v, r, nr, _m_entries); move_vertex(v, r, nr, _m_entries);`
63/// -- virtual_move **mutates** a reusable scratch `EntrySet`, and every one of
64/// those C++ members is non-const. `&self` plus the `Sync` supertrait leaves
65/// only three options: allocate per Metropolis step (on the hottest path in
66/// the library), use a `thread_local` (reintroducing exactly the thread-keyed
67/// hidden global that `parallel_rng.hh:56-73` is condemned for, and
68/// destroying reproducibility), or take `&mut self` (deleting the lock-free
69/// proposal phase). Threading `&mut Self::Scratch` is the fourth, and it makes
70/// `_m_entries_pool.resize(get_num_threads())` (`:451`) an explicit per-chunk
71/// value instead of a thread-count-indexed global.
72pub trait SpecCore: Sync {
73    /// Entropy parameters.
74    type Args: EntropyArgs;
75    /// Per-caller scratch. [`Workspace`](crate::delta::Workspace) for the
76    /// block model.
77    type Scratch: Default + Send;
78    /// Edge/vertex weight type.
79    type W: Weight;
80    /// Directedness.
81    type D: Dir;
82
83    /// Apply a move. `spec_base.hh:69`.
84    fn move_vertex(&mut self, v: VertexId, r: Group, nr: Group, sc: &mut Self::Scratch);
85
86    /// Price a move without applying it. `spec_base.hh:70`.
87    fn virtual_move(
88        &self,
89        v: VertexId,
90        r: Group,
91        nr: Group,
92        ea: &Self::Args,
93        sc: &mut Self::Scratch,
94    ) -> f64;
95
96    /// The absolute entropy. `spec_base.hh:71`.
97    fn entropy(&self, ea: &Self::Args) -> f64;
98}

100/// The proposal kernel **and** its density, in one impl block.
101///
102/// Bundling them is the substantive fix. In C++ the sampler and the density
103/// can come from *different classes* in the inheritance chain; in Rust trait
104/// items resolve per-`impl`, so one type supplies both and a disagreement is a
105/// local, visible inconsistency rather than an inheritance accident.
106///
107/// The generator is a **type parameter**, not a fixed concrete type: a
108/// concrete `Rng` baked into the trait would make swapping the generator, or
109/// testing with a deterministic stub, an edit to every signature.
110pub trait GroupProposal: SpecCore {
111    /// Sample a move, returning its own forward log-density.
112    fn propose<R: Rng + ?Sized>(
113        &self,
114        v: VertexId,
115        r: Group,
116        c: f64,
117        d: f64,
118        rng: &mut R,
119    ) -> Option<Proposal>;
120
121    /// `log q(s -> r | v)` for the reverse move, which is never sampled.
122    fn log_reverse(&self, v: VertexId, r: Group, s: Group, c: f64, d: f64) -> f64;
123
124    /// The Hastings ratio. Derived, so the loop never hand-swaps the
125    /// endpoints.
126    ///
127    /// `base/mcmc.hh:123-124` is
128    /// `pf = get_move_prob(v, r, nr, c, d, false);`
129    /// `pb = get_move_prob(v, nr, r, c, d, true);` -- the caller must exchange
130    /// `r` and `nr` **and** flip the flag, in agreement, on two adjacent
131    /// lines. Here there is nothing to exchange.
132    #[inline]
133    fn log_hastings(&self, v: VertexId, r: Group, p: &Proposal, c: f64, d: f64) -> f64 {
134        self.log_reverse(v, r, p.to, c, d) - p.log_fwd
135    }
136
137    /// A local proposal. Derived from [`propose`](Self::propose), so the
138    /// default is legitimate (`spec_base.hh:83-86`).
139    #[inline]
140    fn propose_local<R: Rng + ?Sized>(
141        &self,
142        v: VertexId,
143        r: Group,
144        rng: &mut R,
145    ) -> Option<Proposal> {
146        self.propose(v, r, 0.0, 0.0, rng)
147    }
148}

163/// Group merging. Capability as **trait presence**.
164///
165/// Replaces the duplicated pair `has_merge` on the spec (`spec_base.hh:50`)
166/// and `_has_merge` on `Imp` (`:115`) and again on the loop states
167/// (`merge_split.hh:99`, `multilevel.hh:114`) -- two independent booleans for
168/// one capability, at two scopes, which a derived spec can set
169/// inconsistently. `if constexpr (_has_merge)` becomes a where-clause, and
170/// there is no second flag to disagree with the first.
171pub trait Mergeable: SpecCore {
172    /// Merge `s` into `t`.
173    fn merge_groups(&mut self, s: Group, t: Group, sc: &mut Self::Scratch);
174    /// Price a merge.
175    fn virtual_merge(&self, s: Group, t: Group, ea: &Self::Args, sc: &mut Self::Scratch) -> f64;
176}
