8/// The empty field cell.
9///
10/// `EntrySet::_null` (`entries.hh:198`) is `numeric_limits<uint32_t>::max()`
11/// even though the field vectors are `uint32_t` and the sentinel is compared
12/// against a `size_t`; here the sentinel and the cell have one type.
13const NULL: u32 = u32::MAX;
14
15/// Field-table slots, in `Field<D>`'s order. `const`-checked against it below,
16/// because this module indexes the table arithmetically (`fi & 1` selects the
17/// endpoint, `fi >= 2` the half) and that arithmetic is only meaningful if the
18/// two orders agree.
19const F_R_OUT: usize = 0;
20const F_NR_OUT: usize = 1;
21const F_R_IN: usize = 2;
22const F_NR_IN: usize = 3;
23
24const _: () = {
25    assert!(Field::<Directed>::R_OUT.index() == F_R_OUT);
26    assert!(Field::<Directed>::NR_OUT.index() == F_NR_OUT);
27    assert!(Field::<Directed>::R_IN.index() == F_R_IN);
28    assert!(Field::<Directed>::NR_IN.index() == F_NR_IN);
29    // `fi & 1` picks the endpoint and `fi >= 2` the half only if the four
30    // constants are laid out `{r,nr} x {out,in}` in exactly that nesting.
31    assert!(F_R_OUT + 2 == F_R_IN);
32    assert!(F_NR_OUT + 2 == F_NR_IN);
33};

35/// Which move a buffer is currently recording.
36///
37/// `EntrySet::_rnr` (`entries.hh:200`), with `null_group` spelled `None`.
38/// `modify_entries` (`:322-332`) never records `from == to`: the `r == nr`
39/// case dispatches to `single`, which sets `to` to the null group and folds
40/// both endpoints onto the `r` half-fields. A `MoveKey` whose two endpoints
41/// are equal and present is therefore not a value this buffer expects, and
42/// the crate-private `DeltaBuf::touch` behind [`DeltaBuf::touch_dyn`]
43/// `debug_assert`s so.
44#[derive(Clone, Copy, PartialEq, Eq, Debug, Default)]
45pub struct MoveKey {
46    /// Source group.
47    pub from: Option<Group>,
48    /// Target group.
49    pub to: Option<Group>,
50}
51
52/// Back-pointer from an entry to the field slot that indexes it.
53///
54/// Storing one per entry is what makes `begin` branch-free **and independent
55/// of the previous move key**. `EntrySet::clear` (`entries.hh:169-176`)
56/// re-derives each field address through `get_field`, which reads `_rnr`, so
57/// `set_move` must call `clear()` *before* assigning `_rnr` -- a temporal
58/// coupling stated nowhere.
59///
60/// Eight bytes, not a packed `u32`. A `field:2 | other:30` packing saves four
61/// bytes per entry in a loop that runs once per cleared slot, and caps the
62/// group count at 2^30 -- a capability the C++ has and there is no reason to
63/// give up for that.
64#[repr(C)]
65#[derive(Clone, Copy, PartialEq, Eq, Debug)]
66pub struct SlotRef {
67    /// The other endpoint keying the slot.
68    pub other: u32,
69    /// Which half-field.
70    pub field: u8,
71}
72
73/// Interns the block-graph edge and its current weight for a `(r, s)` pair.
74///
75/// A closure rather than a trait method, so that this module never needs to
76/// know what a block state is: the reified transition is *data*, and saying so
77/// structurally means the delta types do not depend on the state types.
78pub trait Resolve<W: Weight>: FnMut(Group, Group) -> (Option<BEdge>, W) {}
79impl<W: Weight, F: FnMut(Group, Group) -> (Option<BEdge>, W)> Resolve<W> for F {}

81/// A block pair touching neither endpoint of the current move.
82///
83/// `get_field` (`entries.hh:108-119`) falls through to `return _dummy;` -- a
84/// single `uint32_t` member shared by *every* out-of-plane pair (`:223`) --
85/// and `insert_delta_dispatch` then keys an entry on it, so two distinct pairs
86/// silently accumulate into one. The path is reachable only from
87/// `propagate_entries`, where the projected pair provably contains an
88/// endpoint, so the C++ bug is latent and guarded by an invariant written
89/// nowhere. Here there is no dummy cell to alias into.
90#[derive(Clone, Copy, PartialEq, Eq, Debug, thiserror::Error)]
91#[error("block pair ({0:?}, {1:?}) touches neither endpoint of the recorded move")]
92pub struct OutOfPlane(pub Group, pub Group);
93
94/// One level's sparse record of "which block pairs change, and by how much".
95///
96/// The field table is [`D::Fields`](Dir::Fields), the associated *array*
97/// `[Vec<u32>; D::N_FIELDS]`, so an undirected buffer holds **two**
98/// half-fields and a directed one four -- the port of `entries.hh:52`'s
99/// `if constexpr (directed)`. Because the table is an array and not a
100/// `Vec<Vec<u32>>`, the undirected buffer does not merely *allocate* less: it
101/// is a strictly smaller type, which `size_of` asserts. Lookup is O(1) with
102/// no hashing, which is valid because every changed pair contains `r` or
103/// `nr`.
104#[derive(Clone, Debug)]
105pub struct DeltaBuf<D: Dir, W: Weight> {
106    mv: MoveKey,
107    fields: D::Fields,
108    entries: Vec<Entry<W>>,
109    slots: Vec<SlotRef>,
110    /// Cumulative count of field cells written by [`DeltaBuf::begin`].
111    ///
112    /// Not instrumentation for its own sake: "`begin` is O(#entries)" is the
113    /// one claim of this type that a timing test cannot establish and a
114    /// counter can. One increment per store, in a loop that already stores.
115    field_writes: u64,
116    _d: std::marker::PhantomData<fn() -> D>,
117}
118
119impl<D: Dir, W: Weight> Default for DeltaBuf<D, W> {
120    fn default() -> Self {
121        DeltaBuf {
122            mv: MoveKey::default(),
123            fields: D::Fields::default(),
124            entries: Vec::new(),
125            slots: Vec::new(),
126            field_writes: 0,
127            _d: std::marker::PhantomData,
128        }
129    }
130}

132/// The endpoint a half-field names is absent from the move key.
133///
134/// Out of line and `#[cold]`: `modify_entries_dispatch` guards every
135/// `insert_delta_rnr` with `if (r != null_group)` / `if (nr != null_group)`
136/// (`entries.hh:245-255`), so this is a recorder bug, not a data case.
137#[cold]
138#[inline(never)]
139fn absent_endpoint(fi: usize) -> ! {
140    panic!("touch on half-field {fi} whose move-key endpoint is absent");
141}

143impl<D: Dir, W: Weight> DeltaBuf<D, W> {
144    /// Start recording a new move, clearing in O(#entries).
145    ///
146    /// Only ever grows the field tables, matching `entries.hh:48`'s
147    /// amortisation. Correct in any order relative to the move key, unlike
148    /// `set_move`/`clear`.
149    pub fn begin(&mut self, mv: MoveKey, n_groups: usize) {
150        // `clear()` (`entries.hh:169-176`) walks `_entries` and re-derives
151        // each address with `get_field(r, s)`, i.e. through `_rnr`. Walking
152        // the recorded addresses instead is the same O(#entries) and drops
153        // the dependency on the key entirely -- so assigning `mv` first,
154        // last, or not at all cannot leave a live cell behind.
155        for slot in self.slots.drain(..) {
156            self.fields.as_mut()[slot.field as usize][slot.other as usize] = NULL;
157            self.field_writes += 1;
158        }
159        self.entries.clear();
160        self.mv = mv;
161
162        // `set_move` (`:59-65`): grow only, null-filled.
163        if n_groups > self.fields.as_ref()[0].len() {
164            for f in self.fields.as_mut() {
165                f.resize(n_groups, NULL);
166            }
167        }
168    }

173        self.mv

182    /// Number of groups the field table currently covers.
183    ///
184    /// `_r_out_field.size()` (`entries.hh:63`). Grows, never shrinks.
185    #[inline]
186    pub fn table_len(&self) -> usize {
187        self.fields.as_ref()[0].len()
188    }

190    /// Cumulative number of field cells [`begin`](Self::begin) has written.
191    ///
192    /// Exactly one per entry cleared, and nothing else touches the table on
193    /// the reset path. A test asserts this against `entries().len()` to pin
194    /// the "O(#entries), not O(B)" claim without measuring a clock.
195    #[inline]
196    pub const fn field_writes(&self) -> u64 {
197        self.field_writes
198    }

200    /// Whether every cell of the field table is null.
201    ///
202    /// O(`D::N_FIELDS` * `table_len()`), i.e. the cost `begin` deliberately
203    /// does *not* pay. For the audit and for tests; never on a hot path.
204    pub fn table_is_clear(&self) -> bool {
205        self.fields.as_ref().iter().flatten().all(|&c| c == NULL)
206    }

208    /// The half-field the pair `(s, t)` is keyed by, or `None` when it
209    /// touches neither endpoint.
210    ///
211    /// `get_field` (`entries.hh:108-119`) branch for branch and **in order**:
212    /// the order is load-bearing, because the pair `(r, nr)` matches two of
213    /// the four tests and the first one wins. The fallthrough returns `None`
214    /// where the C++ returns `_dummy`.
215    #[inline]
216    fn slot_of(&self, s: Group, t: Group) -> Option<(usize, usize)> {
217        if self.mv.from == Some(s) {
218            return Some((F_R_OUT, t.index()));
219        }
220        if self.mv.from == Some(t) {
221            return Some((Self::sink_half(F_R_OUT, s, t), s.index()));
222        }
223        if self.mv.to == Some(s) {
224            return Some((F_NR_OUT, t.index()));
225        }
226        if self.mv.to == Some(t) {
227            return Some((Self::sink_half(F_NR_OUT, s, t), s.index()));
228        }
229        None
230    }

232    /// `get_field_rnr<First, Source = false>` (`entries.hh:90-103`).
233    ///
234    /// Directed: `(s == t) ? out_field[t] : in_field[s]` -- an endpoint's own
235    /// self-pair has no in-half, so it lives in the out-half whichever call
236    /// site reaches it. Undirected: `out_field[s]`, there being no in-half at
237    /// all, which `Field<Undirected>` states in the type.
238    #[inline]
239    fn sink_half(out: usize, s: Group, t: Group) -> usize {
240        if D::DIRECTED && s != t { out + 2 } else { out }
241    }

243    /// Record a weight change on one half-field.
244    ///
245    /// The static path, and the only one the recorder uses. `Field<D>` is
246    /// total and `ADD` is a const generic, so this is
247    /// `insert_delta_rnr<First, Source, Add>` (`entries.hh:142`) one for one,
248    /// with all three parameters resolved at monomorphisation and no
249    /// fallthrough to reach.
250    ///
251    /// `other` is the *other* endpoint of the pair: `t` for an out-half and
252    /// `s` for an in-half, matching `get_field_rnr`'s indexing. The caller
253    /// owes the same routing decision `modify_entries_dispatch` makes at
254    /// `entries.hh:258-261` and `:311-314` -- a pair containing both `r` and
255    /// `nr` belongs to `r`'s half-fields, because `get_field` tests `r`
256    /// first. Violating that would key one pair under two cells;
257    /// `debug_assert` catches it.
258    ///
259    /// # Panics
260    ///
261    /// If the endpoint `f` names is absent from the move key, or if `other`
262    /// is beyond the group count given to [`begin`](Self::begin).
263    pub(crate) fn touch<const ADD: bool>(
264        &mut self,
265        f: Field<D>,
266        other: Group,
267        w: W,
268        resolve: &mut impl super::Resolve<W>,
269    ) {
270        let fi = f.index();
271        // `fi & 1`: out/in halves of the same endpoint are two apart.
272        let own = match if fi & 1 == 0 { self.mv.from } else { self.mv.to } {
273            Some(g) => g,
274            None => absent_endpoint(fi),
275        };
276        let is_in = fi >= 2;
277        let (s, t) = if is_in { (other, own) } else { (own, other) };
278        // The `s == t` arm of `get_field_rnr`, hoisted: an in-half touch on
279        // the endpoint's own self-pair is an out-half touch.
280        let fi = if is_in && other == own { fi - 2 } else { fi };
281
282        debug_assert_eq!(
283            self.slot_of(s, t),
284            Some((fi, other.index())),
285            "half-field {fi} is not where `get_field` routes ({s:?}, {t:?}) \
286             under {:?}",
287            self.mv
288        );
289        self.insert::<ADD>(fi, other.index(), s, t, w, resolve);
290    }

292    /// Record a weight change on an arbitrary pair.
293    ///
294    /// `insert_delta` (`entries.hh:149`), made **total**. Its only caller is
295    /// [`propagate`](crate::blockmodel::propagate), which is exactly where the
296    /// C++'s latent `_dummy` aliasing lives. The error is provably unreachable
297    /// given the hierarchy invariant, and is returned rather than asserted
298    /// because the projection is a caller-supplied closure.
299    ///
300    /// On `Err` the buffer is **unchanged**: nothing is resolved, no entry is
301    /// created, no cell is written. That is defect #4 -- the C++ creates an
302    /// entry keyed on `_dummy`, which the next out-of-plane pair then finds
303    /// already occupied and adds to.
304    pub fn touch_dyn(
305        &mut self,
306        s: Group,
307        t: Group,
308        w: W,
309        resolve: &mut impl super::Resolve<W>,
310    ) -> Result<(), OutOfPlane> {
311        let Some((fi, oi)) = self.slot_of(s, t) else {
312            return Err(OutOfPlane(s, t));
313        };
314        // `insert_delta` is only ever instantiated with `Add = true`
315        // (`state.hh:1113`); a sign change is expressed in the delta.
316        self.insert::<true>(fi, oi, s, t, w, resolve);
317        Ok(())
318    }

320    /// `insert_delta_dispatch` (`entries.hh:122-135`), plus the interning the
321    /// C++ defers to `get_mes` (`:181-191`).
322    ///
323    /// The before-image is read exactly once per *entry*, not once per touch:
324    /// the resolve closure runs only on the cell that was null.
325    #[inline]
326    fn insert<const ADD: bool>(
327        &mut self,
328        fi: usize,
329        oi: usize,
330        s: Group,
331        t: Group,
332        w: W,
333        resolve: &mut impl super::Resolve<W>,
334    ) {
335        let cell = self.fields.as_ref()[fi][oi];
336        let i = if cell == NULL {
337            let i = self.entries.len();
338            debug_assert!(i < NULL as usize, "entry index collides with the null cell");
339            let (me, mrs_before) = resolve(s, t);
340            self.entries.push(Entry {
341                r: s,
342                s: t,
343                delta: W::ZERO,
344                me,
345                mrs_before,
346            });
347            self.slots.push(SlotRef {
348                other: oi as u32,
349                field: fi as u8,
350            });
351            self.fields.as_mut()[fi][oi] = i as u32;
352            i
353        } else {
354            cell as usize
355        };
356        let e = &mut self.entries[i];
357        e.delta = if ADD { e.delta + w } else { e.delta - w };
358    }

360    /// The entry index for a pair, if it has one.
361    #[inline]
362    fn lookup(&self, s: Group, t: Group) -> Option<usize> {
363        let (fi, oi) = self.slot_of(s, t)?;
364        // `get_delta` indexes `_r_out_field[t]` unchecked (`entries.hh:156`);
365        // a group past the table is "not recorded", not a wild read.
366        match self.fields.as_ref()[fi].get(oi).copied() {
367            Some(c) if c != NULL => Some(c as usize),
368            _ => None,
369        }
370    }
371
372    /// The accumulated change for a pair. `entries.hh:156`, honestly `&self`.
373    pub fn delta_of(&self, s: Group, t: Group) -> W {
374        match self.lookup(s, t) {
375            Some(i) => self.entries[i].delta,
376            // `_zero` (`entries.hh:224`), which `get_delta` returns by
377            // reference for both the null-cell and the out-of-plane case.
378            None => W::ZERO,
379        }
380    }

382    /// The interned block-graph edge for a pair, if the pair is recorded.
383    ///
384    /// A genuine shared-borrow read. `get_mes` (`entries.hh:181-191`) lazily
385    /// extends `_mes` and is reached from the `get_move_prob` *read* path
386    /// (`state.hh:1661`) through `cget_field`'s
387    /// `const_cast<EntrySet*>(this)` (`:121-123`). Interning `me` eagerly is
388    /// what makes this an ordinary `&self` method.
389    ///
390    /// `None` means "not recorded", where `get_me` (`:194-199`) falls back to
391    /// `emat.get_me(r, s)`; the caller owns that fallback, because this type
392    /// has no access to the state.
393    pub fn me_of(&self, s: Group, t: Group) -> Option<Option<BEdge>> {
394        self.lookup(s, t).map(|i| self.entries[i].me)
395    }

397    /// The pair's weight before the move, if the pair is recorded.
398    ///
399    /// The other half of what `get_mes`'s caller reads at `state.hh:1225`.
400    pub fn mrs_before_of(&self, s: Group, t: Group) -> Option<W> {
401        self.lookup(s, t).map(|i| self.entries[i].mrs_before)
402    }

442    /// Every `(half-field, other)` the recorder is allowed to use routes to
