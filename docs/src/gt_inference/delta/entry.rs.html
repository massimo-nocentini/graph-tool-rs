1//! What one recorded transition consists of.
2
3use crate::ids::{BEdge, Group, Weight};
4
5/// One changed block pair.
6///
7/// `me` and `mrs_before` are the **before-image**, interned at record time.
8/// Carrying them is free: the sparse pricing loop already performs exactly
9/// this `_emat.get_me(t, w)` lookup per entry (`blockmodel/state.hh:1225`),
10/// and `update_rs` performs it again (`entries.hh:359`). Paying it once and
11/// letting both consumers read it removes two cache-cold dependent loads per
12/// entry from the hottest kernel in the library.
13///
14/// It is *not* free everywhere, and the cost is stated rather than hidden:
15/// `virtual_move_groups` returns early when `s == t || n == 0`
16/// (`state.hh:1340-1342`) and a clabel-rejected move returns `+inf` at
17/// `:1346`, so those paths now pay one block-edge lookup per distinct entry
18/// that the C++ skips.
19#[derive(Clone, Copy, PartialEq, Debug)]
20pub struct Entry<W: Weight> {
21    /// Source group of the changed pair.
22    pub r: Group,
23    /// Target group of the changed pair.
24    pub s: Group,
25    /// How much the `(r, s)` edge weight changes.
26    pub delta: W,
27    /// The block-graph edge, if it already existed.
28    pub me: Option<BEdge>,
29    /// Its weight before the move.
30    pub mrs_before: W,
31}
32
33/// The per-endpoint scalars the pricing function needs.
34///
35/// Exactly the six values `entries_dS` reads at `state.hh:1239-1253`.
36#[derive(Clone, Copy, PartialEq, Debug, Default)]
37pub struct EndImage<W: Weight> {
38    /// Out-weight of the group.
39    pub mrp: W,
40    /// In-weight of the group.
41    pub mrm: W,
42    /// Vertex weight of the group.
43    pub wr: W,
44}
45
46/// The non-entry half of the before-image: six scalars plus the move's own
47/// degree deltas.
48#[derive(Clone, Copy, PartialEq, Debug)]
49pub struct MoveHeader<W: Weight> {
50    /// Source group, or `None` for a single-endpoint (insertion) move.
51    pub r: Option<Group>,
52    /// Target group, or `None` for a removal move.
53    pub nr: Option<Group>,
54    /// `r`'s scalars before the move.
55    pub r_img: EndImage<W>,
56    /// `nr`'s scalars before the move.
57    pub nr_img: EndImage<W>,
58    /// In-degree contribution of the moved vertex.
59    pub dkin: W,
60    /// Out-degree contribution of the moved vertex.
61    pub dkout: W,
62    /// Vertex weight leaving `r`.
63    pub dr: W,
64    /// Vertex weight entering `nr`.
65    pub dnr: W,
66}
67
68impl<W: Weight> Default for MoveHeader<W> {
69    fn default() -> Self {
70        MoveHeader {
71            r: None,
72            nr: None,
73            r_img: EndImage::default(),
74            nr_img: EndImage::default(),
75            dkin: W::ZERO,
76            dkout: W::ZERO,
77            dr: W::ZERO,
78            dnr: W::ZERO,
79        }
80    }
81}
