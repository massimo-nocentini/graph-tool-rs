1//! Connected, weakly connected and strongly connected components.
2//!
3//! The port of `label_components` (`topology/graph_components.hh:102-129`),
4//! which is a two-arm tag dispatch: `boost::strong_components` when
5//! `directed_category` converts to `directed_tag`, `boost::connected_components`
6//! otherwise. That single entry point is split in two here, and the split is
7//! the point.
8//!
9//! `boost::connected_components` carries
10//! `BOOST_STATIC_ASSERT((is_same<directed, undirected_tag>::value))`, so the
11//! C++ can only reach it through the `std::false_type` arm --- and the tag it
12//! branches on comes from `graph_traits<Graph>::directed_category`, i.e. from
13//! the *view type*, not from what the caller meant. `GraphView(g,
14//! directed=False)` (`__init__.py:1370`) is how a caller says "weak
15//! components", and forgetting it silently buys the strong ones instead.
16//!
17//! Here [`components`] and [`strong_components`] are separate functions and
18//! the second is bounded on [`Bidirectional`], which no undirected view
19//! implements, so the two are not confusable in either direction: asking for
20//! strong components of `g.undirect()` is `error[E0277]`, and asking for
21//! connected components is `components(g.undirect())`, written out.
22//!
23//! ## Labels
24//!
25//! Both functions reproduce boost's numbering exactly, because
26//! `graph_components.hh:110` wraps the caller's map in a
27//! `HistogramPropertyMap` whose histogram is indexed *by the label*: the
28//! numbering is observable from Python as `hist`, and
29//! `label_largest_component` (`__init__.py:1431`) then does
30//! `c.fa == h.argmax()`.
31//!
32//! * [`components`] numbers by root order: the component of the first
33//!   unvisited vertex in [`VertexList::vertices`] order is 0, the next is 1.
34//!   That is `components_recorder::start_vertex`, which increments the counter
35//!   once per DFS root, with `discover_vertex` writing the current value.
36//! * [`strong_components`] numbers in the order the SCC *roots finish*, which
37//!   is `tarjan_scc_visitor::finish_vertex` incrementing `c_count` after
38//!   popping a component off its stack. The observable consequence is that the
39//!   labels are a reverse topological order of the condensation: for every
40//!   edge `u -> v`, `label[u] >= label[v]`.
41//!
42//! ## The sentinel
43//!
44//! Both functions first fill the whole *bound* with [`UNLABELED`] and only
45//! then label what the view exposes. On a filtered view that matters: the map
46//! is sized for the unfiltered index space (it must be --- the labels are
47//! indexed by unfiltered `VertexId`), while only the surviving vertices are
48//! reachable from `vertices()`. C++ leaves the rest holding whatever the
49//! property map's default was, which for a fresh `int32_t` map is `0` ---
50//! indistinguishable from a genuine member of component 0. `label_components`
51//! is documented to return labels "from 0 to N-1" (`graph_components.hh:99`)
52//! and its histogram counts only the vertices it wrote, so the two disagree on
53//! a filtered graph. Here a vertex the view does not expose reads `-1`.
54
55use gt_core::graph::{Bidirectional, GraphRef, VertexList};
56use gt_core::ids::{VertexId, VertexTag};
57use gt_core::prop::DenseProp;
58
59/// The value a vertex outside the view carries after a labelling run.
60///
61/// Distinct from every real label, which are `0..n`.
62pub const UNLABELED: i64 = -1;
63
64/// Panic message for the one failure `sized_for` can report here. The bound
65/// comes from `g` itself, so the only way to reach it is to pass a map minted
66/// for a *different* graph --- a caller error, not a runtime condition, and
67/// the one `GraphId` comparison of [DESIGN](gt_core::design) D-`Bound` exists to catch it.
68const WRONG_GRAPH: &str = "the label map belongs to a different graph";
69
70/// Label each vertex with its component.
71///
72/// On an [`Und`](gt_core::view::Und) view this is the connected components; on
73/// a directed view it is the *out*-reachability partition, which is usually
74/// not what the caller means. Callers wanting weak components should pass
75/// `g.undirect()` explicitly rather than relying on the algorithm to guess.
76///
77/// Returns the number of components. Vertices the view does not expose are set
78/// to [`UNLABELED`].
79///
80/// # Panics
81///
82/// If `label` was minted for a different graph than `g`.
83pub fn components<G>(g: G, label: &mut DenseProp<i64, VertexTag>) -> usize
84where
85    G: GraphRef + VertexList,
86{
87    let bound = g.vertex_bound();
88    let mut view = label.sized_for(bound).expect(WRONG_GRAPH);
89    let lab = view.as_mut_slice();
90    lab.fill(UNLABELED);
91
92    // One frontier for the whole run, so the per-root cost is amortised to
93    // zero allocations after the first component: `clear` keeps the capacity.
94    // The frontier holds `VertexId`s and not `G::Out` iterators because the
95    // label of a vertex depends only on *which* root reached it, never on the
96    // order the root's tree was walked in -- `components_recorder` changes its
97    // counter in `start_vertex` alone. So a stack of 4-byte ids is exactly as
98    // faithful as boost's recursive DFS and does not risk its stack depth.
99    let mut frontier: Vec<VertexId> = Vec::new();
100    let mut n_comp: i64 = 0;
101
102    for root in g.vertices() {
103        if lab[root.index()] != UNLABELED {
104            continue;
105        }
106        lab[root.index()] = n_comp;
107        frontier.clear();
108        frontier.push(root);
109        while let Some(v) = frontier.pop() {
110            for i in g.out_edges(v) {
111                let w = i.other.index();
112                if lab[w] == UNLABELED {
113                    lab[w] = n_comp;
114                    frontier.push(i.other);
115                }
116            }
117        }
118        n_comp += 1;
119    }
120
121    n_comp as usize
122}

124/// Strongly connected components (Tarjan).
125///
126/// Bounded on [`Bidirectional`], so it cannot be called on an undirected view
127/// at all -- where the answer would be the connected components under a name
128/// that promises something else.
129///
130/// Returns the number of components. Vertices the view does not expose are set
131/// to [`UNLABELED`].
132///
133/// Labels are assigned in the order SCC roots finish, exactly as
134/// `boost::strong_components` assigns them, so they are a reverse topological
135/// order of the condensation: `label[source] >= label[target]` for every edge,
136/// with equality exactly when the two endpoints are mutually reachable.
137///
138/// # Panics
139///
140/// If `label` was minted for a different graph than `g`.
141pub fn strong_components<G>(g: G, label: &mut DenseProp<i64, VertexTag>) -> usize
142where
143    G: Bidirectional + VertexList,
144{
145    let bound = g.vertex_bound();
146    let n = bound.len();
147    let mut view = label.sized_for(bound).expect(WRONG_GRAPH);
148    let lab = view.as_mut_slice();
149    lab.fill(UNLABELED);
150
151    /// Discovery time of a vertex the DFS has not reached yet.
152    const UNVISITED: usize = usize::MAX;
153
154    let mut disc: Vec<usize> = vec![UNVISITED; n];
155    let mut low: Vec<usize> = vec![UNVISITED; n];
156    // The Tarjan stack: vertices discovered and not yet assigned a component.
157    let mut open: Vec<VertexId> = Vec::new();
158    // The explicit DFS stack. `boost::depth_first_search` recurses; a
159    // 10^7-vertex path would overflow the thread stack there, and does not
160    // here. Each frame owns its own incidence iterator, which is what makes
161    // "resume where this vertex left off" expressible without re-scanning.
162    let mut frames: Vec<(VertexId, G::Out)> = Vec::new();
163    let mut time: usize = 0;
164    let mut n_comp: i64 = 0;
165
166    for root in g.vertices() {
167        if disc[root.index()] != UNVISITED {
168            continue;
169        }
170        disc[root.index()] = time;
171        low[root.index()] = time;
172        time += 1;
173        open.push(root);
174        frames.push((root, g.out_edges(root)));
175
176        // `v` is copied out of the frame and the iterator step taken before
177        // anything touches `frames` again, so the frame borrow ends on this
178        // line and the recursive `push` below is an ordinary statement rather
179        // than an aliasing problem.
180        while let Some(&mut (v, ref mut it)) = frames.last_mut() {
181            let step = it.next();
182
183            if let Some(i) = step {
184                let w = i.other;
185                if disc[w.index()] == UNVISITED {
186                    disc[w.index()] = time;
187                    low[w.index()] = time;
188                    time += 1;
189                    open.push(w);
190                    frames.push((w, g.out_edges(w)));
191                } else if lab[w.index()] == UNLABELED {
192                    // Discovered and not yet in a component, i.e. still on the
193                    // Tarjan stack. This is `tarjan_scc_visitor`'s own test:
194                    // it compares `get(m_comp, w)` against the map's sentinel
195                    // rather than carrying a separate `on_stack` array.
196                    let dw = disc[w.index()];
197                    let lv = &mut low[v.index()];
198                    if dw < *lv {
199                        *lv = dw;
200                    }
201                } // else: a cross edge into a finished component, ignored.
202                continue;
203            }
204
205            frames.pop();
206            if low[v.index()] == disc[v.index()] {
207                // `v` roots an SCC: unwind the open stack down to it.
208                while let Some(w) = open.pop() {
209                    lab[w.index()] = n_comp;
210                    if w == v {
211                        break;
212                    }
213                }
214                n_comp += 1;
215            }
216            if let Some((p, _)) = frames.last() {
217                let lv = low[v.index()];
218                let lp = &mut low[p.index()];
219                if lv < *lp {
220                    *lp = lv;
221                }
222            }
223        }
224    }
225
226    debug_assert!(open.is_empty(), "every discovered vertex joins a component");
227    n_comp as usize
228}

230/// The largest component, as a vertex mask suitable for
231/// [`MaskFilter`](gt_core::view::MaskFilter).
232///
233/// The mask is indexed by the *unfiltered* vertex index space and has exactly
234/// [`vertex_bound`](gt_core::graph::GraphBase::vertex_bound) entries, which is
235/// what [`Filtered::masked`](gt_core::view::Filtered::masked) demands: sizing
236/// it from `num_vertices` instead is `graph_copy.cc:66-73`'s error, and on a
237/// filtered input it is not even a big-enough allocation.
238///
239/// This is [`components`], not [`strong_components`] --- the bound is
240/// `GraphRef`, so a directed view's answer is its out-reachability partition.
241/// `label_largest_component` (`__init__.py:1385-1432`) instead inherits
242/// `label_components`' tag dispatch and therefore silently means the largest
243/// *strong* component on a directed graph. Callers who want that should run
244/// [`strong_components`] and take the argmax themselves; callers who want the
245/// largest connected component pass `g.undirect()`, which is the explicit
246/// spelling of `label_largest_component(g, directed=False)`.
247///
248/// Ties go to the lowest label, matching `h.argmax()` (`__init__.py:1431`),
249/// which numpy resolves to the first maximum. An empty view yields an
250/// all-zero mask rather than the `ValueError` numpy's `argmax` raises on an
251/// empty histogram.
252///
253/// # Panics
254///
255/// Never, for any `g`: the scratch map is minted from `g`'s own id.
256pub fn largest_component_mask<G>(g: G) -> Vec<u8>
257where
258    G: GraphRef + VertexList,
259{
260    let n = g.vertex_bound().len();
261    let mut label = DenseProp::<i64, VertexTag>::new(g.graph_id());
262    let n_comp = components(g, &mut label);
263
264    let lab = label.as_slice();
265    let mut mask = vec![0u8; n];
266    if n_comp == 0 {
267        return mask;
268    }
269
270    // The histogram `HistogramPropertyMap` builds incrementally
271    // (`graph_components.hh:63-74`), built in one pass instead: its `put`
272    // hook exists only because boost writes the labels one at a time.
273    let mut hist = vec![0usize; n_comp];
274    for &c in lab.iter().take(n) {
275        if c != UNLABELED {
276            hist[c as usize] += 1;
277        }
278    }
279
280    // `argmax`: first maximum wins.
281    let mut best = 0usize;
282    for (c, &h) in hist.iter().enumerate() {
283        if h > hist[best] {
284            best = c;
285        }
286    }
287
288    let best = best as i64;
289    for (slot, &c) in mask.iter_mut().zip(lab.iter()) {
290        *slot = u8::from(c == best);
291    }
292    mask
293}
