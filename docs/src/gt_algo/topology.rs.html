1//! Topological predicates and orderings.
2//!
3//! Six kernels from five different C++ translation units, and the thing they
4//! have in common is that each one is a place where "which view am I on"
5//! decides the answer.
6//!
7//! ## The direction each kernel actually reads
8//!
9//! | kernel | incidence used | why |
10//! |---|---|---|
11//! | [`is_dag`], [`topological_sort`] | `out_edges` | a topological order is a statement about arcs; `boost::topological_sort` walks out-edges and throws `not_a_dag` on a back edge (`topology/graph_topological_sort.cc:34`) |
12//! | [`is_bipartite`] | `all_edges` | `graph_bipartite.cc:84` dispatches under `{.tr=never_directed}` and the Python wrapper builds `GraphView(g, directed=False)` first, so bipartiteness never sees an orientation |
13//! | [`global_clustering`] | `all_edges` | `clustering/__init__.py:226-227` undirects a directed graph before calling the C++ at all |
14//! | [`count_triangles`], [`reciprocity`] | `edges()` | the count is per *edge*; see below |
15//! | [`kcore_decomposition`] | `degree` + `all_edges` | `graph_kcore.hh:41`/`:62` read the whole block, so the core number is the total-degree one on a directed graph and a self-loop counts twice (`topology/__init__.py:1832-1836`) |
16//!
17//! Because [`Und<G>::All`](gt_core::view::Und) **is** `G::All`
18//! (`view/undirected.rs:107-116`), the three direction-insensitive kernels give
19//! the same answer on `&g` and on `g.undirect()` by construction rather than by
20//! a runtime `is_directed(g)` test. That equality is asserted in
21//! `tests/u17_topology.rs`.
22//!
23//! ## Why the counting kernels are bounded on `EdgeList`
24//!
25//! `get_triangles` (`clustering/graph_clustering.hh:50-93`) counts ordered
26//! neighbour pairs per vertex and then repairs the multiplicity twice: once
27//! with `/ 2` for the undirected case at `:92`, and once with `/ 3` at `:132`
28//! when the caller wants a triangle *count*. Both repairs are divisions
29//! applied to a degree summation, and a degree summation over an undirected
30//! view is exactly the arithmetic that makes a filtered `num_edges` wrong
31//! ([DESIGN](gt_core::design) D3). Here the kernels enumerate the edge set once, build a
32//! sorted simple adjacency, and count each triangle at its unique `u < v < w`
33//! witness -- so there is no multiplicity to divide out and no division to get
34//! wrong.
35//!
36//! ## Where this deliberately does not reproduce the C++
37//!
38//! `get_triangles` marks a neighbour with `mark[u] = w` (`:64`, an
39//! *assignment*) while accumulating the degree with `k += w` (`:65`, a
40//! *sum*). On a simple graph the two agree. On a multigraph they do not: the
41//! numerator then uses the 0/1 adjacency of the underlying simple graph and
42//! the denominator uses the multiplicity-weighted one, so the ratio is not
43//! `Tr(A³) / Σ_{i≠j}[A²]_{ij}` for any single `A` -- which is the definition
44//! `clustering/__init__.py:194-201` states. This port uses the *simple*
45//! underlying graph throughout (parallel edges collapsed, self-loops dropped,
46//! matching `:60` and `:72`'s `u == v` skips), so numerator and denominator
47//! are consistent and `global_clustering == 3 · count_triangles / triples`
48//! holds exactly. For a simple graph -- graph-tool's own karate doctest
49//! included -- the two implementations agree.
50

57// ---------------------------------------------------------------------------
58// Sorted adjacency in CSR form: the shared substrate of the counting kernels.
59// ---------------------------------------------------------------------------
60
61/// Vertices per parallel chunk.
62///
63/// A fixed [`Plan`] gives up load balance (`par/plan.rs:12-19`) and the
64/// counting work is ragged -- an intersection costs `O(d_u + d_v)` -- so the
65/// grain is small enough that a high-degree row cannot monopolise a chunk,
66/// and large enough that a chunk is not one cache line of work.
67const GRAIN: usize = 256;
68
69/// The seed is structurally required by [`det_reduce`] and structurally
70/// unused: these folds are over `u64`, which is associative, and no chunk
71/// draws a random number. It is named rather than inlined so that the fact is
72/// stated once.
73const NO_RANDOMNESS: Seed = Seed([0; 32]);
74
75/// A sorted adjacency list, one row per vertex index in the view's
76/// [`vertex_bound`](gt_core::graph::GraphBase::vertex_bound).
77///
78/// Two allocations, not one per vertex: rows live end to end in `data` and
79/// `start` holds the `n + 1` offsets, so a row is a `&[VertexId]` and the
80/// intersection kernel below never touches a pointer it did not just read.
81struct Csr {
82    /// `n + 1` offsets; row `v` is `data[start[v]..start[v + 1]]`.
83    start: Vec<usize>,
84    /// Row contents, each row sorted ascending.
85    data: Vec<VertexId>,
86}
87
88impl Csr {
89    /// Row `v`, sorted ascending.
90    #[inline]
91    fn row(&self, v: usize) -> &[VertexId] {
92        &self.data[self.start[v]..self.start[v + 1]]
93    }
94
95    /// Number of rows.
96    #[inline]
97    fn len(&self) -> usize {
98        self.start.len() - 1
99    }
100
101    /// Turn a degree histogram in `start[1..]` into offsets, and allocate.
102    ///
103    /// Returns the fill cursor, one per row. Filling from a cursor rather than
104    /// from `start` itself is what lets the rows be written in whatever order
105    /// the edge enumeration produces.
106    fn allocate(mut start: Vec<usize>) -> (Self, Vec<usize>) {
107        let mut acc = 0usize;
108        for s in &mut start {
109            acc += *s;
110            *s = acc;
111        }
112        // `start` now holds the *end* of each row shifted by one slot, which
113        // is exactly the exclusive prefix sum: `start[0] == 0` because the
114        // histogram was written into `start[1..]`.
115        let cursor = start[..start.len() - 1].to_vec();
116        let csr = Csr {
117            data: vec![VertexId::from_index(0); acc],
118            start,
119        };
120        (csr, cursor)
121    }
122
123    /// Sort every row, drop duplicates, and close the gaps.
124    ///
125    /// The compaction is in place and always writes behind the read head
126    /// (`w <= s` at every step, because nothing ever grows), so one buffer
127    /// serves for both.
128    fn simplify(&mut self) {
129        let n = self.len();
130        let mut w = 0usize;
131        for v in 0..n {
132            let s = self.start[v];
133            let e = self.start[v + 1];
134            let row = &mut self.data[s..e];
135            row.sort_unstable();
136            // `Vec::dedup` would need a row-shaped `Vec`; this is the same
137            // stable partition, written in place over the shared buffer.
138            let mut kept = 0usize;
139            for k in 0..row.len() {
140                if kept == 0 || row[kept - 1] != row[k] {
141                    row[kept] = row[k];
142                    kept += 1;
143                }
144            }
145            self.data.copy_within(s..s + kept, w);
146            self.start[v] = w;
147            w += kept;
148        }
149        self.start[n] = w;
150        self.data.truncate(w);
151    }
152
153    /// Sort every row, keeping duplicates.
154    fn sort_rows(&mut self) {
155        for v in 0..self.len() {
156            let s = self.start[v];
157            let e = self.start[v + 1];
158            self.data[s..e].sort_unstable();
159        }
160    }
161}

163/// The simple undirected adjacency underlying `g`, built from its **edge
164/// set**.
165///
166/// Self-loops are dropped (`graph_clustering.hh:60`) and parallel edges are
167/// collapsed. Two passes over `edges()`, which is `Clone` on every view.
168fn simple_adjacency_from_edges<G: EdgeList>(g: G) -> Csr {
169    let n = g.vertex_bound().len();
170    let mut start = vec![0usize; n + 1];
171    for e in g.edges() {
172        let (s, t) = (e.source().index(), e.target().index());
173        if s != t {
174            start[s + 1] += 1;
175            start[t + 1] += 1;
176        }
177    }
178    let (mut csr, mut cursor) = Csr::allocate(start);
179    for e in g.edges() {
180        let (s, t) = (e.source().index(), e.target().index());
181        if s != t {
182            csr.data[cursor[s]] = e.target();
183            cursor[s] += 1;
184            csr.data[cursor[t]] = e.source();
185            cursor[t] += 1;
186        }
187    }
188    csr.simplify();
189    csr
190}

192/// The simple undirected adjacency underlying `g`, built from its
193/// **incidence**.
194///
195/// The same structure as [`simple_adjacency_from_edges`] for a kernel that has
196/// [`VertexList`] but not [`EdgeList`]. Each row is filled from its own
197/// vertex's `all_edges`, so symmetry follows from the storage rather than from
198/// this function pushing both directions -- and a filtered view whose excluded
199/// vertices are absent from `vertices()` simply leaves those rows empty.
200///
201/// A self-loop occupies a slot in each half of the adjacency block and is
202/// therefore yielded *twice* by `all_edges` (`graph_adjacency.hh:1069-1073`);
203/// the `other != v` test drops both copies.
204fn simple_adjacency_from_incidence<G: GraphRef + VertexList>(g: G) -> Csr {
205    let n = g.vertex_bound().len();
206    let mut start = vec![0usize; n + 1];
207    for v in g.vertices() {
208        let mut d = 0usize;
209        for i in g.all_edges(v) {
210            d += usize::from(i.other != v);
211        }
212        start[v.index() + 1] = d;
213    }
214    let (mut csr, mut cursor) = Csr::allocate(start);
215    for v in g.vertices() {
216        let vi = v.index();
217        for i in g.all_edges(v) {
218            if i.other != v {
219                csr.data[cursor[vi]] = i.other;
220                cursor[vi] += 1;
221            }
222        }
223    }
224    csr.simplify();
225    csr
226}

228/// Number of elements common to two ascending slices.
229///
230/// A merge, not a `contains` loop: both rows are already sorted, so the cost
231/// is `O(|a| + |b|)` with no allocation and no hashing anywhere in the kernel.
232#[inline]
233fn intersection_len(a: &[VertexId], b: &[VertexId]) -> u64 {
234    let (mut i, mut j, mut c) = (0usize, 0usize, 0u64);
235    while i < a.len() && j < b.len() {
236        match a[i].cmp(&b[j]) {
237            Ordering::Less => i += 1,
238            Ordering::Greater => j += 1,
239            Ordering::Equal => {
240                c += 1;
241                i += 1;
242                j += 1;
243            }
244        }
245    }
246    c
247}

249/// Triangles witnessed at `u`, plus the connected triples centred on `u`.
250///
251/// The witness is the unique ordering `u < v < w`, so a triangle is counted
252/// **once** over the whole vertex range -- where `get_triangles` counts it
253/// six times (twice per corner) and divides by two and then by three
254/// (`graph_clustering.hh:92`, `:132`).
255#[inline]
256fn triangles_and_triples(csr: &Csr, rows: std::ops::Range<usize>) -> (u64, u64) {
257    let (mut tri, mut triples) = (0u64, 0u64);
258    for u in rows {
259        let au = csr.row(u);
260        // `(k * k - k2) / 2` at `graph_clustering.hh:92` with unit weights is
261        // `C(d, 2)`, written here without the intermediate that overflows.
262        let d = au.len() as u64;
263        triples += d * d.saturating_sub(1) / 2;
264
265        let uu = VertexId::from_index(u);
266        let first = au.partition_point(|&x| x <= uu);
267        for (k, &v) in au.iter().enumerate().skip(first) {
268            let av = csr.row(v.index());
269            // Both suffixes hold only vertices greater than `v`: `au[k + 1..]`
270            // because `au` is sorted and `au[k] == v`, `av[hi..]` by search.
271            let hi = av.partition_point(|&x| x <= v);
272            tri += intersection_len(&au[k + 1..], &av[hi..]);
273        }
274    }
275    (tri, triples)
276}

278/// [`triangles_and_triples`] over every row, folded in [`Plan`] order.
279fn count_over_rows(csr: &Csr) -> (u64, u64) {
280    let plan = Plan::new(csr.len(), GRAIN);
281    det_reduce(
282        plan,
283        NO_RANDOMNESS,
284        |rows, _rng| triangles_and_triples(csr, rows),
285        (0, 0),
286        |a, b| (a.0 + b.0, a.1 + b.1),
287    )
288}

290// ---------------------------------------------------------------------------
291// Depth-first machinery for the ordering kernels
292// ---------------------------------------------------------------------------
293
294/// Not yet reached.
295const WHITE: u8 = 0;
296/// On the DFS stack.
297const GREY: u8 = 1;
298/// Finished.
299const BLACK: u8 = 2;
300
301/// One iterative DFS over out-edges, recording finish order when asked.
302///
303/// `RECORD` is a const parameter rather than an `Option<&mut Vec<_>>` so that
304/// [`is_dag`] and [`topological_sort`] are the *same* traversal -- the
305/// acceptance condition for this unit is that they agree on every graph, and
306/// two hand-written walks could drift. The predicate arm monomorphises with
307/// the pushes removed.
308///
309/// Iterative, not recursive: `boost::depth_first_search` recurses, so
310/// `topological_sort` on a path of a few hundred thousand vertices is a stack
311/// overflow in graph-tool and a heap `Vec` here.
312fn dfs_finish_order<G, const RECORD: bool>(g: G, order: &mut Vec<VertexId>) -> bool
313where
314    G: GraphRef + VertexList,
315{
316    let mut colour = vec![WHITE; g.vertex_bound().len()];
317    // `(vertex, its unconsumed out-edges)`: the explicit form of the recursion
318    // variable. `G::Out` is `Clone` but never cloned here; it is moved in.
319    let mut stack: Vec<(VertexId, G::Out)> = Vec::new();
320
321    for s in g.vertices() {
322        if colour[s.index()] != WHITE {
323            continue;
324        }
325        colour[s.index()] = GREY;
326        stack.push((s, g.out_edges(s)));
327
328        while !stack.is_empty() {
329            // The top frame is read, advanced and released before the body
330            // touches `stack` again: the shorter
331            // `while let Some((v, edges)) = stack.last_mut()` holds the borrow
332            // across the whole body, so pushing a successor inside it is
333            // `error[E0499]` -- the borrow checker saying what the recursive
334            // C++ formulation leaves to convention.
335            let top = stack.len() - 1;
336            let (v, edges) = &mut stack[top];
337            let v = *v;
338            let step = edges.next();
339
340            match step {
341                Some(i) => match colour[i.other.index()] {
342                    WHITE => {
343                        colour[i.other.index()] = GREY;
344                        stack.push((i.other, g.out_edges(i.other)));
345                    }
346                    // A back edge, which `boost::topo_sort_visitor` reports by
347                    // throwing `not_a_dag` (caught at
348                    // `graph_topological_sort.cc:51`). A self-loop is a back
349                    // edge to a grey vertex too, so it is a cycle here exactly
350                    // as it is there.
351                    GREY => return false,
352                    _ => {}
353                },
354                None => {
355                    colour[v.index()] = BLACK;
356                    if RECORD {
357                        order.push(v);
358                    }
359                    stack.pop();
360                }
361            }
362        }
363    }
364    true
365}

371/// Whether the directed view is acyclic.
372///
373/// Exactly `topological_sort(g).is_some()`, and shares its traversal: both are
374/// [`dfs_finish_order`], which is the only place a back edge is recognised.
375/// graph-tool reaches the same answer through an exception
376/// (`graph_topological_sort.cc:38-55`), so `is_DAG` and `topological_sort`
377/// there are one C++ entry point returning a `bool` and filling an out
378/// parameter.
379///
380/// A self-loop is a cycle. An undirected view is a DAG only when it has no
381/// edges at all, because every incidence is then its own back edge -- which is
382/// the honest answer to a question that was asked of the wrong view, not a
383/// silent zero.
384pub fn is_dag<G>(g: G) -> bool
385where
386    G: GraphRef + VertexList,
387{
388    let mut sink = Vec::new();
389    dfs_finish_order::<G, false>(g, &mut sink)
390}
391
392/// A topological order, or `None` if the view has a cycle.
393///
394/// Returned **in sort order**: `u` precedes `v` for every arc `u -> v`. The
395/// C++ writes DFS finish order through a `back_inserter`
396/// (`graph_topological_sort.cc:34`), which is the *reverse* topological order,
397/// and the Python wrapper repairs it with `topological_order.a[::-1]`
398/// (`topology/__init__.py`). One reversal, here, at the only place that knows
399/// the convention.
400///
401/// `None` rather than graph-tool's `ValueError`: the caller cannot forget to
402/// check it, and there is no second entry point that returns a half-filled
403/// vector alongside a `false`.
404pub fn topological_sort<G>(g: G) -> Option<Vec<VertexId>>
405where
406    G: GraphRef + VertexList,
407{
408    let mut order = Vec::with_capacity(g.num_vertices());
409    if !dfs_finish_order::<G, true>(g, &mut order) {
410        return None;
411    }
412    order.reverse();
413    Some(order)
414}

416/// Whether the view is bipartite, and the two-colouring if so.
417///
418/// Direction-insensitive, walking `all_edges`: `graph_bipartite.cc:84`
419/// dispatches under `{.tr=never_directed}` and the Python wrapper undirects
420/// the graph first, so an orientation never reaches the algorithm.
421///
422/// The colouring uses graph-tool's convention: each search root is `1` and its
423/// neighbours `0`, because `boost::is_bipartite` paints the start vertex
424/// `white` (`bipartite.hpp`, `put_property(..., white, on_start_vertex())`)
425/// and `graph_bipartite.cc:51` writes `part[v] == white` into the property
426/// map. A DFS and a BFS agree on this map -- inside a component the colour is
427/// the parity of the distance to the root, whatever order the vertices were
428/// reached in -- so the queue below reproduces the C++ map exactly.
429///
430/// Indices outside the view (a filtered-out vertex) are left `0`. A self-loop
431/// makes a graph non-bipartite, as it does in boost, where it is a back edge
432/// whose endpoints are one vertex.
433///
434/// `None` rather than a filled map: boost documents that "if the graph is not
435/// bipartite, the contents of the color map are undefined", which the Python
436/// wrapper patches up after the fact with `part.a = 0`. An undefined map is
437/// not a value this signature can return.
438pub fn is_bipartite<G>(g: G) -> Option<Vec<u8>>
439where
440    G: GraphRef + VertexList,
441{
442    /// Not yet coloured. Erased to `0` before the map is returned.
443    const UNSEEN: u8 = 2;
444
445    let mut colour = vec![UNSEEN; g.vertex_bound().len()];
446    let mut queue: Vec<VertexId> = Vec::new();
447
448    for s in g.vertices() {
449        if colour[s.index()] != UNSEEN {
450            continue;
451        }
452        // White, i.e. `1` in graph-tool's partition map.
453        colour[s.index()] = 1;
454        queue.push(s);
455
456        while let Some(v) = queue.pop() {
457            let c = colour[v.index()];
458            for i in g.all_edges(v) {
459                let u = i.other.index();
460                if colour[u] == UNSEEN {
461                    colour[u] = 1 - c;
462                    queue.push(i.other);
463                } else if colour[u] == c {
464                    return None;
465                }
466            }
467        }
468    }
469
470    for c in &mut colour {
471        if *c == UNSEEN {
472            *c = 0;
473        }
474    }
475    Some(colour)
476}

478/// Global clustering coefficient (transitivity).
479///
480/// `c = 3 · triangles / triples`, the definition at
481/// `clustering/__init__.py:190-193`, over the simple undirected graph
482/// underlying this view -- which is what `global_clustering` computes in
483/// graph-tool too, because the Python wrapper replaces a directed graph with
484/// `GraphView(g, directed=False)` before dispatching
485/// (`clustering/__init__.py:226-227`). So this returns the same number on
486/// `&g` and on `g.undirect()`, and both match the C++.
487///
488/// `triples` is `Σ_v C(d_v, 2)`, which is `(k² - k2) / 2` at
489/// `graph_clustering.hh:92` with unit weights. Vertices of degree below two
490/// contribute nothing, matching the `out_degree(v, g) > 1` guard at `:56`.
491///
492/// Returns `f64::NAN` for a graph with no connected triple, which is what
493/// `double(triangles) / n` yields at `graph_clustering.hh:116` when `n == 0`.
494/// Zero would be a claim -- "this graph has no transitivity" -- that an empty
495/// graph does not support.
496///
497/// The jackknife error term (`:118-131`) is not ported: it is a second
498/// statistic, not part of the coefficient, and it belongs with the counts a
499/// caller asks for explicitly.
500pub fn global_clustering<G>(g: G) -> f64
501where
502    G: GraphRef + VertexList + Sync,
503{
504    let csr = simple_adjacency_from_incidence(g);
505    let (triangles, triples) = count_over_rows(&csr);
506    3.0 * triangles as f64 / triples as f64
507}

509/// Number of triangles, counted once each.
510///
511/// Bounded on [`EdgeList`] because the count is per *edge*, not per incidence:
512/// summing over out-edges of an undirected view double-counts, which is the
513/// same arithmetic error that makes a filtered undirected `num_edges` wrong.
514///
515/// A triangle is a set of three distinct, pairwise adjacent vertices of the
516/// simple graph underlying this view: parallel edges do not multiply it and
517/// self-loops do not create one. Each is counted at its unique `u < v < w`
518/// witness, so the answer is the same on a directed view and on its
519/// [`undirect`](gt_core::view::Undirect)ed counterpart, and no division
520/// repairs a multiplicity afterwards.
521///
522/// graph-tool exposes no directed triangle count at all -- `global_clustering`
523/// undirects first -- so the directed arm of `get_triangles`
524/// (`graph_clustering.hh:89-92`) counts *transitive triples* `v→u, v→w, u→w`,
525/// of which a three-cycle `v→u→w→v` contributes none. That quantity is not a
526/// triangle count and `triangles / 3` at `:132` is not even an integer for it.
527pub fn count_triangles<G>(g: G) -> u64
528where
529    G: GraphRef + EdgeList + Sync,
530{
531    let csr = simple_adjacency_from_edges(g);
532    count_over_rows(&csr).0
533}

535/// Reciprocity: the fraction of edges whose reverse also exists.
536///
537/// `E↔ / E`, ported from `get_reciprocity`
538/// (`topology/graph_reciprocity.cc:29-70`) with unit weights and the Python
539/// default `self_loops=False` (`topology/__init__.py`, `edge_reciprocity`).
540/// The C++ accumulates, per vertex `v` and per out-neighbour `u`,
541/// `Lbd += min(m_uv, m_vu)` and `L += m_vu` (`:59-60`), so a parallel edge
542/// counts towards `L` with its multiplicity and towards `Lbd` only as far as
543/// the reverse multiplicity supports it. That `min` is reproduced exactly
544/// here: run-lengths in one sorted adjacency row against an equal-range probe
545/// in the other, rather than the `O(Σ_v d_v²)` rescan at `:54-58`.
546///
547/// Self-loops are excluded from both numerator and denominator (`:46-47`).
548/// A graph with no non-self-loop edge returns `f64::NAN`, as `Lbd / double(L)`
549/// does at `:69`.
550///
551/// Bounded on [`Bidirectional`], so it cannot be asked of an undirected view,
552/// where the answer is `1` by construction and the question is a mistake.
553pub fn reciprocity<G>(g: G) -> f64
554where
555    G: Bidirectional + EdgeList,
556{
557    let n = g.vertex_bound().len();
558    let mut start = vec![0usize; n + 1];
559    for e in g.edges() {
560        if e.source() != e.target() {
561            start[e.source().index() + 1] += 1;
562        }
563    }
564    let (mut out, mut cursor) = Csr::allocate(start);
565    for e in g.edges() {
566        if e.source() != e.target() {
567            let s = e.source().index();
568            out.data[cursor[s]] = e.target();
569            cursor[s] += 1;
570        }
571    }
572    // Sorted but *not* deduplicated: the multiplicities are the whole point of
573    // the `min`.
574    out.sort_rows();
575
576    let (mut reciprocated, mut total) = (0u64, 0u64);
577    for s in 0..n {
578        let row = out.row(s);
579        total += row.len() as u64;
580        let sv = VertexId::from_index(s);
581        let mut i = 0usize;
582        while i < row.len() {
583            let t = row[i];
584            let j = i + row[i..].partition_point(|&x| x == t);
585            let forward = (j - i) as u64;
586            let back = out.row(t.index());
587            let lo = back.partition_point(|&x| x < sv);
588            let hi = back.partition_point(|&x| x <= sv);
589            reciprocated += forward.min((hi - lo) as u64);
590            i = j;
591        }
592    }
593
594    reciprocated as f64 / total as f64
595}

599/// The core number of every vertex: Batagelj--Zaversnik bin sort.
600///
601/// Ports `kcore_decomposition` (`topology/graph_kcore.hh:25-80`), which
602/// `do_kcore_decomposition` (`graph_kcore.cc:31-40`) dispatches over
603/// `all_graph_views`, so the same body runs on a directed graph, an undirected
604/// one and any filtered view of either.
605///
606/// ## The two incidence choices, which are the whole of the semantics
607///
608/// The C++ reads `degree(v, g)` (`:41`) and `all_neighbors_range(v, g)`
609/// (`:62`). For `adj_list` both are the *whole* block
610/// (`graph_adjacency.hh:1068-1072`, `:1163-1170`), so:
611///
612/// * on a directed graph the degree is the total in + out degree --- which the
613///   Python docstring states outright (`topology/__init__.py:1832-1833`);
614/// * a **self-loop contributes two**, because it occupies both halves of its
615///   vertex's block, and "these edges contribute to the degree in the usual
616///   fashion" (`:1835-1836`);
617/// * a parallel edge contributes its multiplicity.
618///
619/// This port keeps those reads --- [`GraphRef::degree`] and
620/// [`GraphRef::all_edges`] --- so `kcore_decomposition(&g)` and
621/// `kcore_decomposition(g.undirect())` are the same computation, exactly as
622/// the C++'s single body is.
623///
624/// ## The bookkeeping
625///
626/// `bins[k]` holds the vertices whose *remaining* degree is `k`, and `pos[v]`
627/// is `v`'s index inside its bin, so removing a vertex from the middle of a
628/// bin is the swap-with-back at `:66-71`. Vertices are drained from the
629/// smallest bin upwards; peeling `v` out of bin `k` lowers each still-present
630/// neighbour whose remaining degree exceeds `deg[v]` by one and moves it down
631/// a bin. `deg[v]` itself is never lowered after `v` is popped, which is what
632/// makes the `ku > deg[v]` test at `:64` mean "u has not been peeled yet".
633///
634/// Returns the largest core number, i.e. the degeneracy of the graph.
635///
636/// # Panics
637///
638/// If `core` was minted for a different graph than `g`.
639pub fn kcore_decomposition<G>(
640    g: G,
641    core: &mut gt_core::prop::DenseProp<i64, gt_core::ids::VertexTag>,
642) -> usize
643where
644    G: GraphRef + VertexList,
645{
646    let bound = g.vertex_bound();
647    let mut view = core
648        .sized_for(bound)
649        .expect("the core map belongs to a different graph");
650    let out = view.as_mut_slice();
651    // A vertex the view does not expose is never binned and never popped, so
652    // it would otherwise keep whatever the map held. `components::UNLABELED`
653    // is the same convention; the C++ leaves the map's previous contents.
654    out.fill(crate::components::UNLABELED);
655
656    let n = bound.len();
657    let mut deg = vec![0usize; n];
658    let mut pos = vec![0usize; n];
659    // `bins` is sized by the maximum degree, as `:46-47` sizes it.
660    let mut bins: Vec<Vec<VertexId>> = Vec::new();
661    for v in g.vertices() {
662        let k = g.degree(v);
663        deg[v.index()] = k;
664        if k >= bins.len() {
665            bins.resize(k + 1, Vec::new());
666        }
667        bins[k].push(v);
668        pos[v.index()] = bins[k].len() - 1;
669    }
670
671    let mut max_core = 0usize;
672    for k in 0..bins.len() {
673        // `while (!bins_k.empty()) { v = bins_k.back(); bins_k.pop_back(); }`
674        // (`:56-59`). The bin can *grow* while it is being drained --- a
675        // neighbour demoted into it --- which is why this is a `while` and not
676        // an iteration over a snapshot.
677        while let Some(v) = bins[k].pop() {
678            out[v.index()] = k as i64;
679            max_core = max_core.max(k);
680            let dv = deg[v.index()];
681            for inc in g.all_edges(v) {
682                let u = inc.other;
683                let ku = deg[u.index()];
684                // `if (ku > deg[v])` (`:64`): a peeled neighbour has a
685                // remaining degree at most `deg[v]`, and so does a self-loop's
686                // `u == v`, so neither is demoted.
687                if ku <= dv {
688                    continue;
689                }
690                // Swap `u` out of the middle of `bins[ku]` (`:66-71`).
691                let w = *bins[ku].last().expect("u is in this bin");
692                let pw = pos[u.index()];
693                pos[w.index()] = pw;
694                bins[ku][pw] = w;
695                bins[ku].pop();
696                deg[u.index()] = ku - 1;
697                bins[ku - 1].push(u);
698                pos[u.index()] = bins[ku - 1].len() - 1;
699            }
700        }
701    }
702
703    max_core
704}

715    fn csr(rows: &[&[usize]]) -> Csr {
716        let n = rows.len();
717        let mut start = vec![0usize; n + 1];
718        for (v, r) in rows.iter().enumerate() {
719            start[v + 1] = r.len();
720        }
721        let (mut c, mut cursor) = Csr::allocate(start);
722        for (v, r) in rows.iter().enumerate() {
723            for &u in *r {
724                c.data[cursor[v]] = VertexId::from_index(u);
725                cursor[v] += 1;
726            }
727        }
728        c
729    }
