1//! GraphML.
2//!
3//! Ports the writer of `src/boost-workaround/boost/graph/graphml.hpp`
4//! (`write_graphml`, `:353-528`) and the reader of `src/graph/graphml.cpp`
5//! (`graphml_reader`, `:44-479`), together with the `mutate_graph_impl`
6//! glue that sits between the reader and the property maps
7//! (`graphml.hpp:93-286`).
8//!
9//! ## The type-name table
10//!
11//! GraphML's `attr.type` is *not* [`ValueKind::name`]. It is
12//! `prop_type_names[]` (`graphml.cpp:27-31`) indexed by position in
13//! `prop_value_types` (`graphml.hpp:38-43`), and neither the names nor the
14//! order match `type_names[]`/`value_types` in `graph_properties.hh`:
15//!
16//! | this port's [`ValueKind`] | `attr.type` | note |
17//! |---|---|---|
18//! | `F64` (`double`) | `float` | |
19//! | `LongDouble` (`long double`) | `double` | *not* IEEE double |
20//! | `VecF64` | `vector_float` | |
21//! | `VecLongDouble` | `vector_double` | |
22//! | `PyObject` | `python_object` | base64 of the pickle payload |
23//!
24//! So a GraphML file written by another tool that declares `attr.type="double"`
25//! -- which in the GraphML specification means IEEE double -- is read by
26//! graph-tool as `long double`. That is a compatibility hazard, not a defect
27//! in the code, and this port reproduces it: changing the table would make
28//! graph-tool unable to read files this crate writes.
29//!
30//! ## `long double` in a text format
31//!
32//! [`LongDouble`] is an opaque 16-byte payload with no arithmetic
33//! ([DESIGN](gt_core::design) row 49), and a text format needs a *representation*. This module
34//! writes it as a C99 hexadecimal float -- `0x8p-3` for `1.0` -- because that
35//! is exact, and because graph-tool's own reader takes long doubles through
36//! `sscanf("%La")` (`str_repr.hh:81-84`), which accepts precisely that form.
37//! Decimal input is accepted too, by parsing it as `f64` and widening, which
38//! is exact for every value a `double` can hold and is what a file *written*
39//! by graph-tool contains (`print_float`, `:62-69`). The interpretation is the
40//! x86-64 80-bit extended layout; see [`long_double_to_text`].
41//!
42//! ## Ordering
43//!
44//! `dynamic_properties` is a `std::multimap<std::string, ...>`, so
45//! `write_graphml`'s three walks over `dp` (`graphml.hpp:383`, `:430`, `:458`)
46//! visit the properties **sorted by name**, whatever order they were added in.
47//! [`write`] sorts a stable copy of the index range the same way, so the key
48//! ids and the `<data>` order are a function of the document alone. Edges are
49//! emitted in [`EdgeId`] order, per the crate-level ordering contract.

144/// `print_float` (`str_repr.hh:61-69`): `max_digits10` significant digits, the
145/// `C` locale, and `ostream`'s default float format -- that is, `%.17g`.
146///
147/// Rust has no `%g`, so this derives it from `{:.16e}`: take the decimal
148/// exponent `x`, and use a fixed form with `16 - x` fraction digits when
149/// `-4 <= x < 17` and a scientific form otherwise, in both cases with trailing
150/// zeros removed. The exponent is written with a sign and at least two digits,
151/// which is what `ostream` does.
152pub(crate) fn print_float(v: f64) -> String {
153    if v.is_nan() {
154        // glibc's `ostream` prints the sign of a NaN; `sscanf("%la")` reads
155        // both spellings back.
156        return if v.is_sign_negative() { "-nan" } else { "nan" }.to_owned();
157    }
158    if v.is_infinite() {
159        return if v < 0.0 { "-inf" } else { "inf" }.to_owned();
160    }
161
162    /// `std::numeric_limits<double>::max_digits10`.
163    const P: usize = 17;
164    let sci = format!("{:.*e}", P - 1, v);
165    let (mantissa, exp) = sci.split_once('e').expect("Rust always emits the e");
166    let x: i32 = exp.parse().expect("Rust always emits a decimal exponent");
167
168    if x < -4 || x >= P as i32 {
169        let mut s = trim_zeros(mantissa).to_owned();
170        let _ = write!(s, "e{}{:02}", if x < 0 { '-' } else { '+' }, x.abs());
171        s
172    } else {
173        let places = (P as i32 - 1 - x) as usize;
174        trim_zeros(&format!("{v:.places$}")).to_owned()
175    }
176}

184    s.strip_suffix('.').unwrap_or(s)

187/// An exact C99 hexadecimal float for the opaque 16-byte `long double`.
188///
189/// The payload is read as the x86-64 System V layout: bytes 0..8 are the
190/// 64-bit significand *including* its explicit integer bit, bytes 8..10 are
191/// the sign and the 15-bit exponent, bytes 10..16 are padding. That is the
192/// layout every `.gt` file in circulation carries, and it is the layout
193/// `%La` prints. On a target whose `long double` is IEEE binary128 the bytes
194/// mean something else; this module says so rather than pretending, and the
195/// binary `.gt` path -- which never interprets the payload -- is unaffected.
196///
197/// The padding bytes are *not* representable in text: a value that goes
198/// through a text format comes back with bytes 10..16 zeroed.
199pub(crate) fn long_double_to_text(v: LongDouble) -> String {
200    let frac = u64::from_le_bytes([
201        v.0[0], v.0[1], v.0[2], v.0[3], v.0[4], v.0[5], v.0[6], v.0[7],
202    ]);
203    let se = u16::from_le_bytes([v.0[8], v.0[9]]);
204    let sign = if se & 0x8000 != 0 { "-" } else { "" };
205    let exp = i32::from(se & 0x7FFF);
206
207    if exp == 0x7FFF {
208        // Infinity is the integer bit alone; everything else is a NaN.
209        return if frac == 0x8000_0000_0000_0000 {
210            format!("{sign}inf")
211        } else {
212            format!("{sign}nan")
213        };
214    }
215    if frac == 0 && exp == 0 {
216        return format!("{sign}0x0p+0");
217    }
218
219    // value = frac * 2^(e - 63), and `d0.d1..d15` denotes frac / 2^60, so the
220    // printed exponent is e - 3. `e` is -16382 for the subnormal range, where
221    // the integer bit is clear.
222    let e = if exp == 0 { -16382 } else { exp - 16383 };
223    let digits = format!("{frac:016x}");
224    let (head, tail) = digits.split_at(1);
225    let rest = trim_zeros_hex(tail);
226    if rest.is_empty() {
227        format!("{sign}0x{head}p{:+}", e - 3)
228    } else {
229        format!("{sign}0x{head}.{rest}p{:+}", e - 3)
230    }
231}

237/// `lexical_cast<long double>` (`str_repr.hh:140-149`), as far as an opaque
238/// payload allows: an exact inverse of [`long_double_to_text`] for the
239/// hexadecimal form, and a widening of the `f64` parse for the decimal form.
240pub(crate) fn long_double_from_text(s: &str) -> Result<LongDouble, String> {
241    let s = s.trim();
242    let (sign, body) = match s.strip_prefix('-') {
243        Some(rest) => (0x8000u16, rest),
244        None => (0u16, s.strip_prefix('+').unwrap_or(s)),
245    };
246    let lower = body.to_ascii_lowercase();
247    if lower == "nan" {
248        return Ok(compose_long_double(sign, 0x7FFF, 0xC000_0000_0000_0000));
249    }
250    if lower == "inf" || lower == "infinity" {
251        return Ok(compose_long_double(sign, 0x7FFF, 0x8000_0000_0000_0000));
252    }
253
254    if let Some(hex) = lower.strip_prefix("0x") {
255        let (frac, e2) = parse_hex_significand(hex)?;
256        if frac == 0 {
257            return Ok(compose_long_double(sign, 0, 0));
258        }
259        // Normalise so the integer bit is bit 63.
260        let shift = frac.leading_zeros();
261        let frac = frac << shift;
262        let e = e2 + 63 - i64::from(shift);
263        return Ok(scale_long_double(sign, frac, e));
264    }
265
266    let v: f64 = body
267        .parse()
268        .map_err(|_| format!("not a floating point value: {s:?}"))?;
269    Ok(widen_f64(if sign != 0 { -v } else { v }))
270}

272/// The significand of a hexadecimal float as an integer, and the power of two
273/// it must be multiplied by: the pair `(v, e)` with `value == v * 2^e`.
274///
275/// A significand with more than sixteen significant hexadecimal digits does
276/// not fit `u64` and is truncated towards zero. `%La` never emits more than
277/// sixteen, so that only costs precision on hand-written input.
278fn parse_hex_significand(hex: &str) -> Result<(u64, i64), String> {
279    let (digits, pexp) = match hex.split_once('p') {
280        Some((d, e)) => (
281            d,
282            e.trim_start_matches('+')
283                .parse::<i64>()
284                .map_err(|_| format!("bad binary exponent in {hex:?}"))?,
285        ),
286        None => (hex, 0),
287    };
288    let (int_part, frac_part) = match digits.split_once('.') {
289        Some((a, b)) => (a, b),
290        None => (digits, ""),
291    };
292    if int_part.is_empty() && frac_part.is_empty() {
293        return Err(format!("no hexadecimal digits in {hex:?}"));
294    }
295
296    // Room for one more digit, or the value is already as precise as `u64`
297    // can carry and further digits only move the exponent.
298    const ROOM: u64 = u64::MAX >> 4;
299    let mut value: u64 = 0;
300    let mut e: i64 = 0;
301    for c in int_part.chars() {
302        let d = digit16(c)?;
303        if value <= ROOM {
304            value = (value << 4) | u64::from(d);
305        } else {
306            e += 4;
307        }
308    }
309    for c in frac_part.chars() {
310        let d = digit16(c)?;
311        if value <= ROOM {
312            value = (value << 4) | u64::from(d);
313            e -= 4;
314        }
315    }
316    Ok((value, e + pexp))
317}

324/// Build the payload for `(frac / 2^63) * 2^e`, where `frac` carries its
325/// integer bit at bit 63 -- that is, `e` is the exponent of the unit bit, and
326/// the stored exponent is just `e` biased by 16383.
327fn scale_long_double(sign: u16, frac: u64, e: i64) -> LongDouble {
328    let biased = e + 16383;
329    if biased >= 0x7FFF {
330        return compose_long_double(sign, 0x7FFF, 0x8000_0000_0000_0000);
331    }
332    if biased <= 0 {
333        // Subnormal: shift the significand right into the fixed exponent.
334        let shift = 1 - biased;
335        if shift >= 64 {
336            return compose_long_double(sign, 0, 0);
337        }
338        return compose_long_double(sign, 0, frac >> shift);
339    }
340    compose_long_double(sign, biased as u16, frac)
341}

343fn compose_long_double(sign: u16, exp: u16, frac: u64) -> LongDouble {
344    let mut b = [0u8; 16];
345    b[..8].copy_from_slice(&frac.to_le_bytes());
346    b[8..10].copy_from_slice(&(sign | exp).to_le_bytes());
347    LongDouble(b)
348}

350/// Widen an IEEE double into the 80-bit extended payload. Exact.
351fn widen_f64(v: f64) -> LongDouble {
352    let bits = v.to_bits();
353    let sign = if bits >> 63 != 0 { 0x8000u16 } else { 0 };
354    let exp = ((bits >> 52) & 0x7FF) as i64;
355    let frac = bits & 0x000F_FFFF_FFFF_FFFF;
356    if exp == 0x7FF {
357        return if frac == 0 {
358            compose_long_double(sign, 0x7FFF, 0x8000_0000_0000_0000)
359        } else {
360            compose_long_double(sign, 0x7FFF, 0xC000_0000_0000_0000)
361        };
362    }
363    if exp == 0 {
364        if frac == 0 {
365            return compose_long_double(sign, 0, 0);
366        }
367        // A double's subnormal is an extended normal: shift the leading one
368        // up to bit 63 and pay for it in the exponent.
369        let shift = frac.leading_zeros();
370        let significand = frac << shift;
371        let e = -1022 - (i64::from(shift) - 11);
372        return scale_long_double(sign, significand, e);
373    }
374    compose_long_double(sign, (exp - 1023 + 16383) as u16, 0x8000_0000_0000_0000 | (frac << 11))
375}

377/// `lexical_cast<double>` (`str_repr.hh:122-131`).
378///
379/// `sscanf("%la")` accepts a hexadecimal float, so this does too; it does not
380/// accept the trailing garbage `sscanf` would silently stop at, because a
381/// reader that takes `12abc` for `12` turns a corrupt file into a graph.
382pub(crate) fn parse_f64(s: &str) -> Result<f64, String> {
383    let t = s.trim();
384    let (neg, body) = match t.strip_prefix('-') {
385        Some(rest) => (true, rest),
386        None => (false, t.strip_prefix('+').unwrap_or(t)),
387    };
388    if body.len() > 2 && (body.starts_with("0x") || body.starts_with("0X")) {
389        let (frac, e) = parse_hex_significand(&body[2..].to_ascii_lowercase())?;
390        let v = (frac as f64) * exp2(e);
391        return Ok(if neg { -v } else { v });
392    }
393    t.parse()
394        .map_err(|_| format!("not a floating point value: {s:?}"))
395}

397/// `2^e` without `powi`'s repeated multiplication error for large `|e|`.
398fn exp2(e: i64) -> f64 {
399    let mut acc = 1.0f64;
400    let mut n = e;
401    while n > 1023 {
402        acc *= f64::from_bits(0x7FE0_0000_0000_0000); // 2^1023
403        n -= 1023;
404    }
405    while n < -1022 {
406        acc *= f64::from_bits(0x0010_0000_0000_0000); // 2^-1022
407        n += 1022;
408    }
409    acc * f64::from_bits((((n + 1023) as u64) & 0x7FF) << 52)
410}

463/// `base64_encode` (`base64.cc:27-38`).
464pub(crate) fn base64_encode(bytes: &[u8]) -> String {
465    let mut out = String::with_capacity(bytes.len().div_ceil(3) * 4);
466    for chunk in bytes.chunks(3) {
467        let b = [chunk[0], *chunk.get(1).unwrap_or(&0), *chunk.get(2).unwrap_or(&0)];
468        let n = (u32::from(b[0]) << 16) | (u32::from(b[1]) << 8) | u32::from(b[2]);
469        let take = chunk.len() + 1;
470        for i in 0..take {
471            out.push(B64[((n >> (18 - 6 * i)) & 0x3F) as usize] as char);
472        }
473        for _ in take..4 {
474            out.push('=');
475        }
476    }
477    out
478}

480/// `base64_decode` (`base64.cc:40-63`).
481pub(crate) fn base64_decode(s: &str) -> Result<Vec<u8>, String> {
482    let mut acc: u32 = 0;
483    let mut bits = 0u32;
484    let mut out = Vec::with_capacity(s.len() / 4 * 3);
485    for c in s.bytes() {
486        if c == b'=' {
487            break;
488        }
489        // Whitespace is what an XML pretty-printer inserts; expat hands it to
490        // the reader and `binary_from_base64` would take it for a digit.
491        if c.is_ascii_whitespace() {
492            continue;
493        }
494        let v = B64
495            .iter()
496            .position(|&x| x == c)
497            .ok_or_else(|| format!("not base64: {:?}", c as char))?;
498        acc = (acc << 6) | v as u32;
499        bits += 6;
500        if bits >= 8 {
501            bits -= 8;
502            out.push(((acc >> bits) & 0xFF) as u8);
503        }
504    }
505    Ok(out)
506}

577/// Extend a column with default-constructed values up to `n`.
578///
579/// This is `vector_property_map::operator[]`'s implicit resize
580/// (`fast_vector_property_map.hh`): a key never written keeps the member's

587            }
588        };

590    match column {

592        PropertyColumn::I16(v) => grow!(v),
593        PropertyColumn::I32(v) => grow!(v),
594        PropertyColumn::I64(v) => grow!(v),

596        PropertyColumn::LongDouble(v) => grow!(v),
597        PropertyColumn::Str(v) => grow!(v),
598        PropertyColumn::VecBool(v) => grow!(v),
599        PropertyColumn::VecI16(v) => grow!(v),
600        PropertyColumn::VecI32(v) => grow!(v),
601        PropertyColumn::VecI64(v) => grow!(v),
602        PropertyColumn::VecF64(v) => grow!(v),
603        PropertyColumn::VecLongDouble(v) => grow!(v),
604        PropertyColumn::VecStr(v) => grow!(v),
605        PropertyColumn::PyObject(v) => grow!(v),
606    }

609/// The `dynamic_properties` a text reader fills, as columns.
610///
611/// `dp` creates a map the first time a name is written and infers the member
612/// from the value (`graph_io.cc:209-265`); the same name written later with a
613/// different member throws `bad_any_cast` there and is an error here.
614#[derive(Default)]
615pub(crate) struct Columns {
616    props: Vec<NamedProperty>,
617    /// One name index per domain, so a lookup borrows the name it was handed
618    /// instead of allocating a key for every value in the file.
619    index: [HashMap<String, usize>; 3],
620}
621
622impl Columns {
623    /// `put(name, dp, key, value)` for a key at position `at` of its domain.
624    pub(crate) fn put(
625        &mut self,
626        name: &str,
627        domain: PropertyDomain,
628        kind: ValueKind,
629        at: usize,
630        text: &str,
631    ) -> Result<(), String> {
632        let by_name = &mut self.index[domain as usize];
633        let slot = match by_name.get(name) {
634            Some(&i) => i,
635            None => {
636                self.props.push(NamedProperty {
637                    name: name.to_owned(),
638                    domain,
639                    values: PropertyColumn::empty(kind),
640                });
641                by_name.insert(name.to_owned(), self.props.len() - 1);
642                self.props.len() - 1
643            }
644        };
645        let column = &mut self.props[slot].values;
646        if column.kind() != kind {
647            return Err(format!(
648                "property {name:?} was read as {} and is now declared {}",
649                graphml_type_name(column.kind()),
650                graphml_type_name(kind)
651            ));
652        }
653        grow_column(column, at + 1);
654        store_at(column, at, text).map_err(|e| format!("invalid value for key {name:?}: {e}"))
655    }
656
657    /// Size every column to its domain and hand the columns over.
658    ///
659    /// A property map must cover its whole index space before it can be
660    /// written again ([`IoError::ShortProperty`]), and a key mentioned only
661    /// for vertex 3 covers nothing else until this runs.
662    pub(crate) fn finish(mut self, n_vertices: usize, edge_bound: usize) -> Vec<NamedProperty> {
663        for p in &mut self.props {
664            let n = match p.domain {
665                PropertyDomain::Graph => 1,
666                PropertyDomain::Vertex => n_vertices,
667                PropertyDomain::Edge => edge_bound,
668            };
669            grow_column(&mut p.values, n);
670        }
671        self.props
672    }
673}

679/// Read a GraphML document.
680///
681/// Ports `graphml_reader` (`graphml.cpp:44-479`) as graph-tool drives it:
682/// `read_graphml(stream, g, dp, true, true, true, ...)` (`graph_io.cc:337`),
683/// that is `store_ids = true`, `integer_vertices = true` and
684/// `ignore_directedness = true`.
685///
686/// * With `parse.nodeids="canonical"` a node id is `n` followed by the vertex
687///   index (`graphml.cpp:288-320`); the first character is dropped whatever it
688///   is, and the rest must parse.
689/// * Otherwise node ids are interned in first-seen order and kept in the
690///   vertex property [`VERTEX_ID_KEY`], which is what lets [`write`] emit the
691///   file's own ids again.
692/// * Edges are added in document order, so the `i`-th `<edge>` is
693///   [`EdgeId`] `i`, and edge property columns are keyed by that.
694/// * The directedness is the disjunction of every `edgedefault="directed"`
695///   and every directed `<edge>`: `flip_directed` never clears
696///   (`graphml.hpp:120-124`).
697///
698/// # Errors
699///
700/// [`IoError::Parse`], with the one-based line the offending token starts on,
701/// for malformed XML, an unknown `attr.type`, a value that does not parse as
702/// its declared type, or a node id that is not canonical where the file says
703/// it is.
704pub fn read<R: Read, H: Lookup>(mut r: R, lookup: H) -> Result<Document<H>, IoError> {
705    let mut buf = Vec::new();
706    r.read_to_end(&mut buf)?;
707    let text = std::str::from_utf8(&buf).map_err(|e| IoError::Parse {
708        line: line_of(&buf, e.valid_up_to()),
709        msg: "input is not valid UTF-8".to_owned(),
710    })?;
711
712    let mut rd = Reader::new(lookup);
713    rd.run(text)?;
714    rd.finish()
715}
716
717fn line_of(buf: &[u8], at: usize) -> usize {
718    1 + buf[..at.min(buf.len())]
719        .iter()
720        .filter(|&&c| c == b'\n')
721        .count()
722}

724/// A `<key>` declaration (`graphml.cpp:166-207`).
725struct Key {
726    /// `for=` -- only `graph`, `node` and `edge` carry data here.
727    domain: Option<PropertyDomain>,
728    name: String,
729    kind: Option<ValueKind>,
730    /// The raw `attr.type`, kept for the diagnostic when `kind` is `None`.
731    type_name: String,
732}

734/// What a `<data>` element currently applies to (`graphml.cpp:94-99`).
735#[derive(Clone, Copy)]
736enum Active {
737    /// Before any `<graph>`: the C++ reads an uninitialised `m_descriptor_kind`
738    /// here. Ignoring the datum is the only defined behaviour available.
739    None,
740    Graph,
741    Vertex(VertexId),
742    Edge(EdgeId),
743}

745struct Reader<H: Lookup> {
746    g: AdjList<H>,
747    directed: bool,
748    canonical_vertices: bool,
749    canonical_edges: bool,
750    keys: HashMap<String, Key>,
751    /// Key id -> default text, ordered: `m_key_default` is a `std::map`
752    /// (`graphml.cpp:455`) and the order decides which default lands first.
753    defaults: BTreeMap<String, String>,
754    vertex: HashMap<String, VertexId>,
755    cols: Columns,
756    active_key: String,
757    active: Active,
758    chardata: String,

761impl<H: Lookup> Reader<H> {
762    fn new(lookup: H) -> Self {
763        Reader {
764            g: AdjList::with_lookup(lookup),
765            directed: false,
766            canonical_vertices: false,
767            canonical_edges: false,
768            keys: HashMap::new(),
769            defaults: BTreeMap::new(),
770            vertex: HashMap::new(),
771            cols: Columns::default(),
772            active_key: String::new(),
773            active: Active::None,
774            chardata: String::new(),
775        }
776    }

778    fn run(&mut self, text: &str) -> Result<(), IoError> {
779        let mut xml = Xml::new(text);
780        let mut stack: Vec<String> = Vec::new();
781        loop {
782            let line = xml.line;
783            match xml.next()? {
784                Event::Eof => {
785                    if let Some(open) = stack.pop() {
786                        return Err(parse_at(line, format_args!("unclosed element <{open}>")));
787                    }
788                    return Ok(());
789                }
790                Event::Text(t) => self.chardata.push_str(&t),
791                Event::Start { name, attrs, empty } => {
792                    self.start(local(name), &attrs, line)?;
793                    if empty {
794                        self.end(local(name), line)?;
795                    } else {
796                        stack.push(local(name).to_owned());
797                    }
798                }
799                Event::End { name } => {
800                    match stack.pop() {
801                        Some(open) if open == local(name) => {}
802                        Some(open) => {
803                            return Err(parse_at(
804                                line,
805                                format_args!("</{}> closes <{open}>", local(name)),
806                            ));
807                        }
808                        None => {
809                            return Err(parse_at(
810                                line,
811                                format_args!("</{}> with nothing open", local(name)),
812                            ));
813                        }
814                    }
815                    self.end(local(name), line)?;
816                }
817            }
818        }
819    }

821    /// `on_start_element` (`graphml.cpp:102-242`).
822    fn start(&mut self, name: &str, attrs: &[(String, String)], line: usize) -> Result<(), IoError> {
823        match name {
824            "edge" => {
825                let mut id = "";
826                let mut source = "";
827                let mut target = "";
828                for (a, v) in attrs {
829                    match a.as_str() {
830                        "id" => id = v,
831                        "source" => source = v,
832                        "target" => target = v,
833                        // `graphml.cpp:125` compares this attribute with
834                        // "directed", but GraphML spells it "true"/"false"
835                        // (and `edgedefault`, which *is* spelled "directed",
836                        // is a different attribute). Both spellings are taken
837                        // as directed here: the C++ reading is a defect that
838                        // silently loses the directedness of a per-edge
839                        // declaration, and `flip_directed` only ever sets.
840                        "directed" if v == "directed" || v == "true" => self.directed = true,
841                        _ => {}
842                    }
843                }
844                self.active = Active::Edge(self.handle_edge(id, source, target, line)?);
845            }
846            "node" => {
847                let mut id = "";
848                for (a, v) in attrs {
849                    if a == "id" {
850                        id = v;
851                    }
852                }
853                self.active = Active::Vertex(self.handle_vertex(id, line)?);
854            }
855            "data" => {
856                for (a, v) in attrs {
857                    if a == "key" {
858                        self.active_key = v.clone();
859                    }
860                }
861            }
862            "key" => {
863                let mut id = String::new();
864                let mut key = Key {
865                    domain: None,
866                    name: String::new(),
867                    kind: None,
868                    type_name: String::new(),
869                };
870                for (a, v) in attrs {
871                    match a.as_str() {
872                        "id" => id = v.clone(),
873                        "attr.name" => key.name = v.clone(),
874                        "attr.type" => {
875                            key.kind = graphml_type_kind(v);
876                            key.type_name = v.clone();
877                        }
878                        "for" => {
879                            key.domain = match v.as_str() {
880                                "graph" => Some(PropertyDomain::Graph),
881                                "node" => Some(PropertyDomain::Vertex),
882                                "edge" => Some(PropertyDomain::Edge),
883                                // `hyperedge`, `port`, `endpoint` and `all`
884                                // are recognised and carry no data
885                                // (`graphml.cpp:186-189`).
886                                "hyperedge" | "port" | "endpoint" | "all" => None,
887                                other => {
888                                    return Err(parse_at(
889                                        line,
890                                        format_args!("unrecognized key kind '{other}'"),
891                                    ));
892                                }
893                            };
894                        }
895                        _ => {}
896                    }
897                }
898                self.active_key = id.clone();
899                self.keys.insert(id, key);
900            }
901            "graph" => {
902                for (a, v) in attrs {
903                    match a.as_str() {
904                        // `flip_directed` never clears (`graphml.hpp:120-124`),
905                        // so the document is directed if anything says so.
906                        "edgedefault" if v == "directed" => self.directed = true,
907                        "parse.nodeids" => self.canonical_vertices = v == "canonical",
908                        "parse.edgeids" => self.canonical_edges = v == "canonical",
909                        _ => {}
910                    }
911                }
912                self.active = Active::Graph;
913            }
914            _ => {}
915        }
916        // `self->m_character_data.clear()` (`graphml.cpp:241`), for every
917        // element, not only the ones above.
918        self.chardata.clear();
919        Ok(())
920    }

922    /// `on_end_element` (`graphml.cpp:244-274`).
923    fn end(&mut self, name: &str, line: usize) -> Result<(), IoError> {
924        match name {
925            "data" => {
926                let key = std::mem::take(&mut self.active_key);
927                let value = std::mem::take(&mut self.chardata);
928                let r = self.apply(&key, self.active, &value);
929                self.active_key = key;
930                self.chardata = value;
931                r.map_err(|e| parse_at(line, e))?;
932            }
933            "default" => {
934                self.defaults
935                    .insert(self.active_key.clone(), self.chardata.clone());
936            }
937            _ => {}
938        }
939        Ok(())
940    }

942    /// `set_{graph,vertex,edge}_property` (`graphml.hpp:150-225`).
943    fn apply(&mut self, key_id: &str, to: Active, value: &str) -> Result<(), String> {
944        let Some(key) = self.keys.get(key_id) else {
945            // An unknown key id: `m_key_name[key_id]` default-constructs in
946            // the C++ and writes a property named "" of type "", which then
947            // fails `type_found`. Saying so is more useful.
948            return Err(format!("no <key> declares id {key_id:?}"));
949        };
950        let Some(domain) = key.domain else {
951            return Ok(());
952        };
953        let kind = key.kind.ok_or_else(|| {
954            format!(
955                "unrecognized type {:?} for key {}",
956                key.type_name, key.name
957            )
958        })?;
959        let name = key.name.clone();
960        let at = match (domain, to) {
961            (PropertyDomain::Graph, Active::Graph) => 0,
962            (PropertyDomain::Vertex, Active::Vertex(v)) => v.index(),
963            (PropertyDomain::Edge, Active::Edge(e)) => e.index(),
964            // The descriptor and the key disagree: the C++ `any_cast` in
965            // `set_*_property` throws `bad_any_cast`, which is not caught.
966            (_, Active::None) => return Ok(()),
967            _ => {
968                return Err(format!(
969                    "key {name:?} is declared for={} but appears on another element",
970                    match domain {
971                        PropertyDomain::Graph => "graph",
972                        PropertyDomain::Vertex => "node",
973                        PropertyDomain::Edge => "edge",
974                    }
975                ));
976            }
977        };
978        self.cols.put(&name, domain, kind, at, value)
979    }

981    /// `handle_vertex` (`graphml.cpp:283-345`), on the
982    /// `integer_vertices == true` path.
983    fn handle_vertex(&mut self, id: &str, line: usize) -> Result<VertexId, IoError> {
984        let (v, is_new) = if self.canonical_vertices {
985            let n: usize = strip_head(id).parse().map_err(|_| {
986                parse_at(line, format_args!("invalid vertex: {id}"))
987            })?;
988            if n > MAX_INDEX {
989                return Err(IoError::IndexWidthExceeded);
990            }
991            let is_new = self.g.num_vertices() <= n;
992            while self.g.num_vertices() <= n {
993                self.g.add_vertex().map_err(graph_error)?;
994            }
995            (VertexId::from_index(n), is_new)
996        } else if let Some(&v) = self.vertex.get(id) {
997            (v, false)
998        } else {
999            let v = self.g.add_vertex().map_err(graph_error)?;
1000            self.vertex.insert(id.to_owned(), v);
1001            (v, true)
1002        };
1003
1004        if is_new {
1005            self.apply_defaults(PropertyDomain::Vertex, Active::Vertex(v), line)?;
1006            if !self.canonical_vertices {
1007                self.cols
1008                    .put(VERTEX_ID_KEY, PropertyDomain::Vertex, ValueKind::Str, v.index(), id)
1009                    .map_err(|e| parse_at(line, e))?;
1010            }
1011        }
1012        Ok(v)
1013    }

1015    /// `handle_edge` (`graphml.cpp:364-392`).
1016    fn handle_edge(
1017        &mut self,
1018        id: &str,
1019        source: &str,
1020        target: &str,
1021        line: usize,
1022    ) -> Result<EdgeId, IoError> {
1023        let s = self.handle_vertex(source, line)?;
1024        let t = self.handle_vertex(target, line)?;
1025        let e = self.g.add_edge(s, t).map_err(graph_error)?.id();
1026        self.apply_defaults(PropertyDomain::Edge, Active::Edge(e), line)?;
1027        if !self.canonical_edges {
1028            self.cols
1029                .put(EDGE_ID_KEY, PropertyDomain::Edge, ValueKind::Str, e.index(), id)
1030                .map_err(|e| parse_at(line, e))?;
1031        }
1032        Ok(e)
1033    }

1035    /// The `<default>` of every key of this domain, applied to a new element
1036    /// (`graphml.cpp:333-339`, `:381-386`).
1037    fn apply_defaults(
1038        &mut self,
1039        domain: PropertyDomain,
1040        to: Active,
1041        line: usize,
1042    ) -> Result<(), IoError> {
1043        if self.defaults.is_empty() {
1044            return Ok(());
1045        }
1046        let pending: Vec<(String, String)> = self
1047            .defaults
1048            .iter()
1049            .filter(|(id, _)| self.keys.get(id.as_str()).and_then(|k| k.domain) == Some(domain))
1050            .map(|(id, v)| (id.clone(), v.clone()))
1051            .collect();
1052        for (id, v) in pending {
1053            self.apply(&id, to, &v).map_err(|e| parse_at(line, e))?;
1054        }
1055        Ok(())
1056    }

1058    fn finish(self) -> Result<Document<H>, IoError> {
1059        let n = self.g.num_vertices();
1060        let m = self.g.edge_bound().len();
1061        Ok(Document {
1062            graph: self.g,
1063            directed: self.directed,
1064            properties: self.cols.finish(n, m),
1065            comment: None,
1066        })
1067    }

1130/// A byte scanner over XML, counting lines.
1131///
1132/// This replaces expat. It implements the subset GraphML uses -- elements,
1133/// attributes, character data, CDATA, comments, processing instructions and a
1134/// skipped DOCTYPE -- and rejects everything else with the line number, which
1135/// is what `XML_GetCurrentLineNumber` gives the C++ (`graphml.cpp:74`).
1136/// Entities beyond the five predefined ones and numeric references are not
1137/// supported; neither are they by any GraphML writer.
1138struct Xml<'a> {
1139    s: &'a str,
1140    b: &'a [u8],
1141    pos: usize,
1142    line: usize,
1143}
1144
1145impl<'a> Xml<'a> {
1146    fn new(s: &'a str) -> Self {
1147        Xml {
1148            s,
1149            b: s.as_bytes(),
1150            pos: 0,
1151            line: 1,
1152        }
1153    }
1154
1155    fn bump(&mut self, n: usize) {
1156        let end = (self.pos + n).min(self.b.len());
1157        self.line += self.b[self.pos..end].iter().filter(|&&c| c == b'\n').count();
1158        self.pos = end;
1159    }
1160
1161    fn rest(&self) -> &'a str {
1162        &self.s[self.pos..]
1163    }
1164
1165    fn err<T>(&self, msg: impl std::fmt::Display) -> Result<T, IoError> {
1166        Err(parse_at(self.line, msg))
1167    }
1168
1169    /// Consume through the first occurrence of `close`, which must be there.
1170    fn through(&mut self, close: &str, what: &str) -> Result<(), IoError> {
1171        match self.rest().find(close) {
1172            Some(i) => {
1173                self.bump(i + close.len());
1174                Ok(())
1175            }
1176            None => self.err(format_args!("unterminated {what}")),
1177        }
1178    }
1179
1180    fn skip_ws(&mut self) {
1181        while let Some(&c) = self.b.get(self.pos) {
1182            if c.is_ascii_whitespace() {
1183                self.bump(1);
1184            } else {
1185                break;
1186            }
1187        }
1188    }
1189
1190    fn next(&mut self) -> Result<Event<'a>, IoError> {
1191        loop {
1192            let Some(&c) = self.b.get(self.pos) else {
1193                return Ok(Event::Eof);
1194            };
1195            if c != b'<' {
1196                let end = self.rest().find('<').map_or(self.b.len(), |i| self.pos + i);
1197                let raw = &self.s[self.pos..end];
1198                self.bump(end - self.pos);
1199                return Ok(Event::Text(decode_entities(raw, self.line)?));
1200            }
1201            let rest = self.rest();
1202            if rest.starts_with("<!--") {
1203                self.bump(4);
1204                self.through("-->", "comment")?;
1205                continue;
1206            }
1207            if rest.starts_with("<![CDATA[") {
1208                self.bump(9);
1209                let start = self.pos;
1210                match self.rest().find("]]>") {
1211                    Some(i) => {
1212                        let raw = &self.s[start..start + i];
1213                        self.bump(i + 3);
1214                        return Ok(Event::Text(Cow::Borrowed(raw)));
1215                    }
1216                    None => return self.err("unterminated CDATA section"),
1217                }
1218            }
1219            if rest.starts_with("<?") {
1220                self.bump(2);
1221                self.through("?>", "processing instruction")?;
1222                continue;
1223            }
1224            if rest.starts_with("<!") {
1225                // A DOCTYPE, with an optional internal subset.
1226                self.bump(2);
1227                let mut depth = 0usize;
1228                loop {
1229                    match self.b.get(self.pos) {
1230                        None => return self.err("unterminated declaration"),
1231                        Some(b'[') => depth += 1,
1232                        Some(b']') => depth = depth.saturating_sub(1),
1233                        Some(b'>') if depth == 0 => {
1234                            self.bump(1);
1235                            break;
1236                        }
1237                        Some(_) => {}
1238                    }
1239                    self.bump(1);
1240                }
1241                continue;
1242            }
1243            if rest.starts_with("</") {
1244                self.bump(2);
1245                let name = self.name()?;
1246                self.skip_ws();
1247                if self.b.get(self.pos) == Some(&b'>') {
1248                    self.bump(1);
1249                    return Ok(Event::End { name });
1250                }
1251                return self.err(format_args!("expected '>' to close </{name}"));
1252            }
1253
1254            self.bump(1);
1255            let name = self.name()?;
1256            let mut attrs = Vec::new();
1257            loop {
1258                self.skip_ws();
1259                if self.rest().starts_with("/>") {
1260                    self.bump(2);
1261                    return Ok(Event::Start {
1262                        name,
1263                        attrs,
1264                        empty: true,
1265                    });
1266                }
1267                if self.b.get(self.pos) == Some(&b'>') {
1268                    self.bump(1);
1269                    return Ok(Event::Start {
1270                        name,
1271                        attrs,
1272                        empty: false,
1273                    });
1274                }
1275                let a = self.name()?;
1276                self.skip_ws();
1277                if self.b.get(self.pos) != Some(&b'=') {
1278                    return self.err(format_args!("expected '=' after attribute {a:?}"));
1279                }
1280                self.bump(1);
1281                self.skip_ws();
1282                let quote = match self.b.get(self.pos) {
1283                    Some(&q @ (b'"' | b'\'')) => q,
1284                    _ => return self.err(format_args!("attribute {a:?} is not quoted")),
1285                };
1286                self.bump(1);
1287                let start = self.pos;
1288                let end = match self.rest().find(quote as char) {
1289                    Some(i) => start + i,
1290                    None => return self.err(format_args!("unterminated value for {a:?}")),
1291                };
1292                let raw = &self.s[start..end];
1293                let line = self.line;
1294                self.bump(end - start + 1);
1295                attrs.push((a.to_owned(), decode_entities(raw, line)?.into_owned()));
1296            }
1297        }
1298    }
1299
1300    /// An element or attribute name.
1301    fn name(&mut self) -> Result<&'a str, IoError> {
1302        let start = self.pos;
1303        while let Some(&c) = self.b.get(self.pos) {
1304            if c.is_ascii_whitespace() || matches!(c, b'/' | b'>' | b'=' | b'<') {
1305                break;
1306            }
1307            self.bump(1);
1308        }
1309        if self.pos == start {
1310            return self.err("expected a name");
1311        }
1312        Ok(&self.s[start..self.pos])
1313    }
1314}

1316/// The five predefined entities and numeric character references.
1317///
1318/// The inverse of `protect_xml_string` (`graphml.cpp:33-41`), which is
1319/// `boost::archive::iterators::xml_escape`: `& < > " '`, and nothing else.
1320fn decode_entities(s: &str, line: usize) -> Result<Cow<'_, str>, IoError> {
1321    if !s.contains('&') {
1322        return Ok(Cow::Borrowed(s));
1323    }
1324    let mut out = String::with_capacity(s.len());
1325    let mut rest = s;
1326    while let Some(i) = rest.find('&') {
1327        out.push_str(&rest[..i]);
1328        let tail = &rest[i..];
1329        let end = tail
1330            .find(';')
1331            .ok_or_else(|| parse_at(line, "unterminated entity reference"))?;
1332        let body = &tail[1..end];
1333        match body {
1334            "amp" => out.push('&'),
1335            "lt" => out.push('<'),
1336            "gt" => out.push('>'),
1337            "quot" => out.push('"'),
1338            "apos" => out.push('\''),
1339            _ => {
1340                let code = if let Some(hex) = body.strip_prefix("#x").or(body.strip_prefix("#X")) {
1341                    u32::from_str_radix(hex, 16).ok()
1342                } else if let Some(dec) = body.strip_prefix('#') {
1343                    dec.parse().ok()
1344                } else {
1345                    None
1346                };
1347                let c = code.and_then(char::from_u32).ok_or_else(|| {
1348                    parse_at(line, format_args!("undefined entity &{body};"))
1349                })?;
1350                out.push(c);
1351            }
1352        }
1353        rest = &tail[end + 1..];
1354    }
1355    out.push_str(rest);
1356    Ok(Cow::Owned(out))
1357}

1363/// `protect_xml_string` (`graphml.cpp:33-41`).
1364///
1365/// Control characters are *not* escaped, because XML 1.0 has no
1366/// representation for them at all -- not even a numeric reference. A
1367/// `string` property is a byte string in the value universe, so a value
1368/// carrying one produces a file expat will refuse; the C++ has the same hole
1369/// and the binary `.gt` format is the format that carries such a value.
1370fn protect(s: &str) -> Cow<'_, str> {
1371    if !s.contains(['&', '<', '>', '"', '\'']) {
1372        return Cow::Borrowed(s);
1373    }
1374    let mut out = String::with_capacity(s.len() + 8);
1375    for c in s.chars() {
1376        match c {
1377            '&' => out.push_str("&amp;"),
1378            '<' => out.push_str("&lt;"),
1379            '>' => out.push_str("&gt;"),
1380            '"' => out.push_str("&quot;"),
1381            '\'' => out.push_str("&apos;"),
1382            _ => out.push(c),
1383        }
1384    }
1385    Cow::Owned(out)
1386}

1388/// The edges of a graph in [`EdgeId`] order.
1389///
1390/// See the crate documentation: adjacency order is not stable across
1391/// removals here, so every reproducible writer sorts.
1392pub(crate) fn edges_by_id<H: Lookup>(g: &AdjList<H>) -> Vec<(EdgeId, VertexId, VertexId)> {
1393    let mut v: Vec<(EdgeId, VertexId, VertexId)> = g
1394        .edges()
1395        .map(|e| (e.id(), e.source(), e.target()))
1396        .collect();

1401/// How many values a column must hold to cover its domain.
1402pub(crate) fn domain_len<H: Lookup>(doc: &Document<H>, d: PropertyDomain) -> usize {
1403    match d {
1404        PropertyDomain::Graph => 1,
1405        PropertyDomain::Vertex => doc.graph.num_vertices(),
1406        PropertyDomain::Edge => doc.graph.edge_bound().len(),
1407    }

1410/// `dp`'s iteration order: by name, ties in insertion order
1411/// (`std::multimap`).
1412pub(crate) fn dp_order<H: Lookup>(doc: &Document<H>) -> Vec<usize> {
1413    let mut order: Vec<usize> = (0..doc.properties.len()).collect();
1414    order.sort_by(|&a, &b| doc.properties[a].name.cmp(&doc.properties[b].name));
1415    order
1416}

1418/// Every column covers its domain, or the first one that does not.
1419pub(crate) fn check_columns<H: Lookup>(doc: &Document<H>) -> Result<(), IoError> {
1420    for p in &doc.properties {
1421        let need = domain_len(doc, p.domain);
1422        if p.values.len() < need {
1423            return Err(IoError::ShortProperty {
1424                name: p.name.clone(),
1425                have: p.values.len(),
1426                need,
1427            });
1428        }
1429    }
1430    Ok(())
1431}

1433/// Write a GraphML document, edges in [`EdgeId`](gt_core::ids::EdgeId) order.
1434///
1435/// Ports `write_graphml` (`graphml.hpp:353-528`) as graph-tool calls it, with
1436/// `ordered_vertices = true` (`graph_io.cc:421`):
1437///
1438/// * Properties are emitted in name order, and the key ids `key0`, `key1`,
1439///   ... follow that order.
1440/// * A vertex property named [`VERTEX_ID_KEY`] becomes the node ids and is
1441///   not emitted as a key; without one the ids are canonical, `n` followed by
1442///   the vertex index. [`EDGE_ID_KEY`] does the same for edges.
1443/// * A value whose text is empty is omitted (`:436-437`, `:468-469`,
1444///   `:516-517`). Reading the file back restores it, because an omitted datum
1445///   leaves the member's default -- but a column *all* of whose values are
1446///   empty is not reconstructed at all, since nothing then names it. That is
1447///   the C++ behaviour and it is why [`gt`](crate::gt) is the format for
1448///   round-tripping a document unchanged.
1449///
1450/// # Errors
1451///
1452/// [`IoError::ShortProperty`] if a column does not cover its domain, and
1453/// [`IoError::Io`] from the stream.
1454pub fn write<W: Write, H: Lookup>(w: W, doc: &Document<H>) -> Result<(), IoError> {
1455    let mut out = BufWriter::with_capacity(1 << 16, w);
1456    check_columns(doc)?;
1457    let order = dp_order(doc);
1458
1459    out.write_all(
1460        b"<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n\
1461          <graphml xmlns=\"http://graphml.graphdrawing.org/xmlns\"\n         \
1462          xmlns:xsi=\"http://www.w3.org/2001/XMLSchema-instance\"\n         \
1463          xsi:schemaLocation=\"http://graphml.graphdrawing.org/xmlns \
1464          http://graphml.graphdrawing.org/xmlns/1.0/graphml.xsd\">\n\n",
1465    )?;
1466    out.write_all(b"  <!-- property keys -->\n")?;
1467
1468    let vertex_ids = order
1469        .iter()
1470        .copied()
1471        .find(|&i| is_id_column(&doc.properties[i], PropertyDomain::Vertex, VERTEX_ID_KEY));
1472    let edge_ids = order
1473        .iter()
1474        .copied()
1475        .find(|&i| is_id_column(&doc.properties[i], PropertyDomain::Edge, EDGE_ID_KEY));
1476
1477    // Key ids, assigned in `dp` order over the properties that get one.
1478    let mut key_id: HashMap<usize, String> = HashMap::new();
1479    let mut count = 0usize;
1480    for &i in &order {
1481        if Some(i) == vertex_ids || Some(i) == edge_ids {
1482            continue;
1483        }
1484        let p = &doc.properties[i];
1485        let id = format!("key{count}");
1486        count += 1;
1487        writeln!(
1488            out,
1489            "  <key id=\"{}\" for=\"{}\" attr.name=\"{}\" attr.type=\"{}\" />",
1490            protect(&id),
1491            match p.domain {
1492                PropertyDomain::Graph => "graph",
1493                PropertyDomain::Vertex => "node",
1494                PropertyDomain::Edge => "edge",
1495            },
1496            protect(&p.name),
1497            protect(graphml_type_name(p.kind())),
1498        )?;
1499        key_id.insert(i, id);
1500    }
1501
1502    writeln!(
1503        out,
1504        "\n  <graph id=\"G\" edgedefault=\"{}\" parse.nodeids=\"{}\" parse.edgeids=\"{}\" parse.order=\"nodesfirst\">\n",
1505        if doc.directed { "directed" } else { "undirected" },
1506        if vertex_ids.is_none() { "canonical" } else { "free" },
1507        if edge_ids.is_none() { "canonical" } else { "free" },
1508    )?;
1509
1510    out.write_all(b"   <!-- graph properties -->\n")?;
1511    for &i in &order {
1512        let p = &doc.properties[i];
1513        if p.domain != PropertyDomain::Graph {
1514            continue;
1515        }
1516        if let Some(val) = nonempty(&p.values, 0) {
1517            writeln!(
1518                out,
1519                "   <data key=\"{}\">{}</data>",
1520                protect(&key_id[&i]),
1521                protect(&val)
1522            )?;
1523        }
1524    }
1525
1526    out.write_all(b"\n   <!-- vertices -->\n")?;
1527    for v in doc.graph.vertices() {
1528        let id = match vertex_ids {
1529            Some(i) => value_at(&doc.properties[i].values, v.index()).unwrap_or_default(),
1530            None => format!("n{}", v.index()),
1531        };
1532        writeln!(out, "    <node id=\"{}\">", protect(&id))?;
1533        for &i in &order {
1534            let p = &doc.properties[i];
1535            if p.domain != PropertyDomain::Vertex || Some(i) == vertex_ids {
1536                continue;
1537            }
1538            if let Some(val) = nonempty(&p.values, v.index()) {
1539                writeln!(
1540                    out,
1541                    "      <data key=\"{}\">{}</data>",
1542                    protect(&key_id[&i]),
1543                    protect(&val)
1544                )?;
1545            }
1546        }
1547        out.write_all(b"    </node>\n")?;
1548    }
1549
1550    out.write_all(b"\n   <!-- edges -->\n")?;
1551    let names = |v: VertexId| match vertex_ids {
1552        Some(i) => value_at(&doc.properties[i].values, v.index()).unwrap_or_default(),
1553        None => format!("n{}", v.index()),
1554    };
1555    for (n, (e, s, t)) in edges_by_id(&doc.graph).into_iter().enumerate() {
1556        let id = match edge_ids {
1557            Some(i) => value_at(&doc.properties[i].values, e.index()).unwrap_or_default(),
1558            None => format!("e{n}"),
1559        };
1560        writeln!(
1561            out,
1562            "    <edge id=\"{}\" source=\"{}\" target=\"{}\">",
1563            protect(&id),
1564            protect(&names(s)),
1565            protect(&names(t)),
1566        )?;
1567        for &i in &order {
1568            let p = &doc.properties[i];
1569            if p.domain != PropertyDomain::Edge || Some(i) == edge_ids {
1570                continue;
1571            }
1572            if let Some(val) = nonempty(&p.values, e.index()) {
1573                writeln!(
1574                    out,
1575                    "      <data key=\"{}\">{}</data>",
1576                    protect(&key_id[&i]),
1577                    protect(&val)
1578                )?;
1579            }
1580        }
1581        out.write_all(b"    </edge>\n")?;
1582    }
1583
1584    out.write_all(b"\n  </graph>\n</graphml>\n")?;
1585    // `BufWriter::drop` swallows the error; the C++ arms the stream with
1586    // `failbit` (`graph_io.cc:485`) so the same condition throws.
1587    out.flush().map_err(IoError::Io)
1588}
