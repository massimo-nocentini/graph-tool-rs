1//! I/O errors.
2
3use gt_core::prop::ValueKind;
4
5/// Anything that can go wrong reading or writing a graph.
6#[derive(Debug, thiserror::Error)]
7pub enum IoError {
8    /// Underlying stream failure.
9    #[error("io: {0}")]
10    Io(#[from] std::io::Error),
11    /// The file is not in the expected format.
12    #[error("bad magic: expected {expected:?}, found {found:?}")]
13    BadMagic {
14        /// What the format requires.
15        expected: &'static [u8],
16        /// What was there.
17        found: Vec<u8>,
18    },
19    /// The file declares a format version this build cannot read.
20    #[error("unsupported format version {0}")]
21    UnsupportedVersion(u8),
22    /// A property map declares a value type this build does not know.
23    #[error("unknown value type name {0:?}")]
24    UnknownValueType(String),
25    /// A property map's declared type and payload disagree.
26    #[error("property {name:?} declares {declared:?} but its payload is not readable as that")]
27    MalformedProperty {
28        /// The property's name.
29        name: String,
30        /// The declared member.
31        declared: ValueKind,
32    },
33    /// A property column handed to a writer does not cover the graph.
34    ///
35    /// The C++ cannot raise this: `write_property_dispatch` walks the graph's
36    /// own range and calls `prop[x]` (`graph_io_binary.hh:319-320`) on an
37    /// `unchecked_vector_property_map`, whose `operator[]`
38    /// (`fast_vector_property_map.hh:218-221`) has no bounds check and whose
39    /// backing vector may be shorter than the range -- which is the class of
40    /// defect `prop::DenseProp::sized_for` exists to close.
41    #[error("property {name:?} holds {have} values, the graph needs {need}")]
42    ShortProperty {
43        /// The property's name.
44        name: String,
45        /// How many values the column holds.
46        have: usize,
47        /// How many the graph's index space requires.
48        need: usize,
49    },
50    /// A descriptor in the file is out of range.
51    #[error("{kind} index {index} exceeds the declared bound {bound}")]
52    IndexOutOfRange {
53        /// "vertex" or "edge".
54        kind: &'static str,
55        /// The offending index.
56        index: u64,
57        /// The declared bound.
58        bound: u64,
59    },
60    /// The graph is too large for the configured index width.
61    #[error("graph exceeds the index width; rebuild with the `wide-index` feature")]
62    IndexWidthExceeded,
63    /// Syntax error in a text format.
64    #[error("parse error at line {line}: {msg}")]
65    Parse {
66        /// One-based line number.
67        line: usize,
68        /// What was wrong.
69        msg: String,
70    },
71}
