1//! Edge lists as delimited text.
2//!
3//! This is the port of `load_graph_from_csv`
4//! (`graph_tool/__init__.py:3768-3893`) together with the two C++ kernels it
5//! dispatches into: `add_edge_list` (`graph_python_interface_imp1.cc:36-94`)
6//! for an index edge list, and `add_edge_list_hash::numpy_dispatch`
7//! (`:217-272`) for a symbolic one.
8//!
9//! ## What the format is, exactly
10//!
11//! One record per line, at least two fields per record, the first two being
12//! the source and the target. graph-tool hands the file to Python's
13//! [`csv.reader`] with `delimiter=","`, `quotechar='"'` and
14//! `strip_whitespace=True`, so the accepted grammar is Python's default
15//! dialect, not "split on the delimiter": a field that opens with `"` runs to
16//! the matching `"`, a doubled `""` inside it is one literal quote, and such a
17//! field may contain the delimiter and even a newline. Both readers here
18//! implement that dialect. Fields past the second are parsed and discarded --
19//! the edge properties `load_graph_from_csv` builds out of them are a
20//! property-map concern, and [`EdgeListOptions`] has no place to put them --
21//! but they are still *parsed*, because a quoted third field is allowed to
22//! contain the record terminator and a reader that stopped at the second
23//! field would mis-frame the next record.
24//!
25//! [`csv.reader`]: https://docs.python.org/3/library/csv.html
26//!
27//! ## Determinism
28//!
29//! Two things fix the graph this module builds, and neither is the machine it
30//! runs on:
31//!
32//! * **Vertex ids** come from first-seen order over the records, source
33//!   before target within a record. That is `get_vertex`
34//!   (`graph_python_interface_imp1.cc:241-253`) called as `s` then `t`
35//!   (`:256-260`), and it is sequential in graph-tool too -- the whole hashed
36//!   path runs under one `GILRelease` with one `vertices` map.
37//! * **Edge ids** come from [`ParBuilder`], whose chunk count this module
38//!   derives from the *record count* alone ([`ROWS_PER_CHUNK`]) and never from
39//!   `available_parallelism()` or the size of the rayon pool. So the `i`-th
40//!   record is always edge `i`, at one worker or at sixteen.
41//!
42//! The second point is what `set_concurrent` costs graph-tool: with per-thread
43//! free lists (`graph_adjacency.hh:460-466`, `:627-655`) the index an edge
44//! receives depends on which worker took it, and every edge property map
45//! written by index afterwards inherits that.
46

58/// Records per staging chunk.
59///
60/// A constant, so the chunk count is a pure function of the input. See the
61/// module documentation, and [`ParBuilder::new`]'s warning that a chunk count
62/// derived from the thread pool is a chunk count that makes the edge ids
63/// depend on the deployment.
64const ROWS_PER_CHUNK: usize = 1 << 16;
65
66/// How to interpret a delimited edge list.
67#[derive(Clone, Copy, Debug)]
68pub struct EdgeListOptions {
69    /// Field separator.
70    pub delimiter: u8,
71    /// Whether the first row names the columns.
72    pub header: bool,
73    /// Whether endpoints are names to be interned rather than indices.
74    pub symbolic: bool,
75}
76
77impl Default for EdgeListOptions {
78    fn default() -> Self {
79        EdgeListOptions {
80            delimiter: b',',
81            header: false,
82            symbolic: true,
83        }
84    }
85}

91/// Build a graph from a delimited edge list.
92///
93/// Uses [`ParBuilder`](gt_core::adj::ParBuilder) when the input can be chunked,
94/// so the edge ids do not depend on the number of worker threads.
95///
96/// The second half of the result is the vertex-name map: `names[v.index()]` is
97/// the string `v` was interned from. It is `load_graph_from_csv`'s `g.vp.name`
98/// (`graph_tool/__init__.py:3891-3892`), and like it, it is empty when
99/// `opts.symbolic` is false -- there the fields *are* the indices and there is
100/// nothing to record.
101///
102/// # Semantics
103///
104/// * Endpoints are stripped of leading and trailing ASCII whitespace, which is
105///   `load_graph_from_csv`'s `strip_whitespace=True` default (`:3832-3836`).
106///   Stripping happens *after* unquoting, exactly as it does there, so
107///   `" a "` and `a` name the same vertex.
108/// * With `opts.symbolic`, vertices are numbered in first-seen order, source
109///   before target. The empty string is a name like any other.
110/// * Without it, each field must parse as a `usize` and the graph gets
111///   `max + 1` vertices -- the fixed point of
112///   `while (s >= num_vertices(g) || t >= num_vertices(g)) add_vertex(g);`
113///   (`graph_python_interface_imp1.cc:75-76`).
114/// * Blank records are skipped. `load_graph_from_csv` does not skip them: a
115///   blank line reaches `add_edge_list` as `[]` and the unconditional
116///   `row[0]` at `graph_tool/__init__.py:3856` raises `IndexError`. A trailing
117///   newline is not a syntax error in any other reader of this format, and it
118///   is not one here.
119///
120/// # Errors
121///
122/// * [`IoError::Io`] from the stream.
123/// * [`IoError::Parse`], with the one-based line the record *started* on, for
124///   a record with fewer than two fields, an unterminated quoted field, a
125///   field that is not UTF-8, or -- when `opts.symbolic` is false -- a field
126///   that is not a non-negative integer.
127/// * [`IoError::IndexWidthExceeded`] if the vertex or edge count does not fit
128///   [`Raw`](gt_core::ids::Raw).
129pub fn read_edge_list<R: Read, H: Lookup>(
130    mut r: R,
131    opts: EdgeListOptions,
132    lookup: H,
133) -> Result<(AdjList<H>, Vec<String>), IoError> {
134    let mut buf = Vec::new();
135    r.read_to_end(&mut buf)?;
136    // A UTF-8 BOM is a byte-order mark, not part of the first vertex's name.
137    let body: &[u8] = buf.strip_prefix(&[0xEF, 0xBB, 0xBF]).unwrap_or(&buf);
138
139    let mut cur = Cursor::new();
140    if opts.header {
141        cur.skip_record(body, opts.delimiter)?;
142    }
143
144    // Phase 1, sequential: resolve every record to a pair of ids. This is the
145    // phase that fixes vertex identity, and it is the phase that reports every
146    // syntax error -- so the parallel phase below cannot fail, and the error a
147    // malformed file produces does not depend on which worker saw it first.
148    let mut pairs: Vec<(VertexId, VertexId)> = Vec::new();
149    let mut names: Vec<String> = Vec::new();
150    let mut index: HashMap<String, VertexId> = HashMap::new();
151    let mut n_vertices = 0usize;
152
153    while let Some(rec) = cur.next_record(body, opts.delimiter)? {
154        let (s, t) = if opts.symbolic {
155            // Source first, then target: `graph_python_interface_imp1.cc:256`
156            // and `:260`. A record `a,a` interns one vertex; a record `b,a`
157            // after `a,b` interns none.
158            let s = intern(&mut index, &mut names, rec.source)?;
159            let t = intern(&mut index, &mut names, rec.target)?;
160            (s, t)
161        } else {
162            let s = parse_index(&rec.source, rec.line)?;
163            let t = parse_index(&rec.target, rec.line)?;
164            n_vertices = n_vertices.max(s.index() + 1).max(t.index() + 1);
165            (s, t)
166        };
167        pairs.push((s, t));
168    }
169    if opts.symbolic {
170        n_vertices = names.len();
171    }
172
173    // Phase 2, parallel: scatter into the staging buffers. `fill` hands chunk
174    // `k` nothing but `k`, which is the whole determinism contract, and here
175    // the generator is a slice copy -- a pure function of `k` by construction.
176    let chunks = pairs.len().div_ceil(ROWS_PER_CHUNK).max(1);
177    let mut builder = ParBuilder::new(n_vertices, chunks);
178    let staged = &pairs[..];
179    builder.fill(|k, out| {
180        let lo = (k * ROWS_PER_CHUNK).min(staged.len());
181        let hi = (lo + ROWS_PER_CHUNK).min(staged.len());
182        out.extend_from_slice(&staged[lo..hi]);
183    });
184    // Dropped before `build`, so the staged pairs never coexist with the
185    // adjacency they become.
186    drop(pairs);
187
188    let g = builder
189        .build(lookup)
190        .map_err(|e| build_error(e, n_vertices))?;
191    Ok((g, names))
192}

194/// Translate the builder's failure into this crate's vocabulary.
195///
196/// `NoSuchVertex` is unreachable: phase 1 either interns the endpoint (so it
197/// is below `names.len()`) or widens `n_vertices` past it. It is mapped rather
198/// than asserted because a reader that aborts the process on a malformed file
199/// is not a reader.
200fn build_error(e: GraphError, n_vertices: usize) -> IoError {
201    match e {
202        GraphError::VertexIdSpaceExhausted { .. } | GraphError::EdgeIdSpaceExhausted { .. } => {
203            IoError::IndexWidthExceeded
204        }
205        GraphError::NoSuchVertex(v) => IoError::IndexOutOfRange {
206            kind: "vertex",
207            index: v.index() as u64,
208            bound: n_vertices as u64,
209        },
210        other => IoError::Parse {
211            line: 0,
212            msg: other.to_string(),
213        },
214    }
215}

217/// First-seen interning.
218///
219/// The `to_owned` on the miss path is the one allocation per *vertex*; a hit
220/// allocates nothing, which matters because a real edge list mentions each
221/// name `degree(v)` times.
222fn intern(
223    index: &mut HashMap<String, VertexId>,
224    names: &mut Vec<String>,
225    name: Cow<'_, str>,
226) -> Result<VertexId, IoError> {
227    if let Some(&v) = index.get(name.as_ref()) {
228        return Ok(v);
229    }
230    let v = VertexId::new(names.len()).ok_or(IoError::IndexWidthExceeded)?;
231    let owned = name.into_owned();
232    names.push(owned.clone());
233    index.insert(owned, v);
234    Ok(v)
235}

237/// An index edge list's endpoint.
238fn parse_index(field: &str, line: usize) -> Result<VertexId, IoError> {
239    let n: usize = field.parse().map_err(|_| IoError::Parse {
240        line,
241        msg: format!("expected a vertex index, found {field:?}"),
242    })?;
243    if n > MAX_INDEX {
244        return Err(IoError::IndexWidthExceeded);
245    }
246    Ok(VertexId::from_index(n))
247}

253/// Where a field stopped.
254#[derive(Clone, Copy, PartialEq, Eq, Debug)]
255enum Stop {
256    /// At a delimiter, which was consumed. Another field follows.
257    Field,
258    /// At a record terminator, which was *not* consumed.
259    Record,
260    /// At end of input.
261    Eof,
262}
263
264/// One record's first two fields, and the line it began on.
265struct Record<'a> {
266    line: usize,
267    source: Cow<'a, str>,
268    target: Cow<'a, str>,
269}
270
271/// A position in the buffer, plus the physical line that position is on.
272///
273/// The line counter is advanced by newlines *inside* quoted fields too, so the
274/// number in a [`IoError::Parse`] is the line an editor would put the cursor
275/// on.
276struct Cursor {
277    pos: usize,
278    line: usize,
279}
280
281impl Cursor {
282    fn new() -> Self {
283        Cursor { pos: 0, line: 1 }
284    }
285
286    /// Consume a `\n`, `\r` or `\r\n` if one is next.
287    fn eat_terminator(&mut self, buf: &[u8]) {
288        match buf.get(self.pos) {
289            Some(&b'\r') => {
290                self.pos += 1;
291                if buf.get(self.pos) == Some(&b'\n') {
292                    self.pos += 1;
293                }
294                self.line += 1;
295            }
296            Some(&b'\n') => {
297                self.pos += 1;
298                self.line += 1;
299            }
300            _ => {}
301        }
302    }
303
304    /// Read one field, leaving the cursor after the delimiter that ended it
305    /// (or on the record terminator).
306    ///
307    /// The returned value is borrowed from `buf` whenever the field carried no
308    /// quotes, which is the common case and the reason this is not a
309    /// `String`-per-field parser.
310    fn field<'a>(&mut self, buf: &'a [u8], delim: u8) -> Result<(Cow<'a, str>, Stop), IoError> {
311        let start_line = self.line;
312        let start = self.pos;
313        // `Some` exactly when the field needed rewriting: it opened with a
314        // quote, so the quotes themselves and any doubled `""` have to go.
315        let mut owned: Option<Vec<u8>> = None;
316        let mut in_quotes = false;
317
318        let (stop, end) = loop {
319            let Some(&c) = buf.get(self.pos) else {
320                break (Stop::Eof, self.pos);
321            };
322            if in_quotes {
323                if c == b'"' {
324                    if buf.get(self.pos + 1) == Some(&b'"') {
325                        // `""` -> one literal quote (Python's `doublequote`).
326                        owned.as_mut().expect("quoted").push(b'"');
327                        self.pos += 2;
328                    } else {
329                        in_quotes = false;
330                        self.pos += 1;
331                    }
332                } else {
333                    if c == b'\n' {
334                        self.line += 1;
335                    }
336                    owned.as_mut().expect("quoted").push(c);
337                    self.pos += 1;
338                }
339            } else if c == delim {
340                let end = self.pos;
341                self.pos += 1;
342                break (Stop::Field, end);
343            } else if c == b'\n' || c == b'\r' {
344                break (Stop::Record, self.pos);
345            } else if c == b'"' && self.pos == start {
346                // A quote opens a field only at its start; anywhere else it is
347                // an ordinary character, which is what `QUOTE_MINIMAL` does on
348                // the way in.
349                owned = Some(Vec::new());
350                in_quotes = true;
351                self.pos += 1;
352            } else {
353                // Text after a closing quote is appended, not rejected: that
354                // is the reader's behaviour, and rejecting it would turn a
355                // file graph-tool loads into a file this crate refuses.
356                if let Some(o) = owned.as_mut() {
357                    o.push(c);
358                }
359                self.pos += 1;
360            }
361        };
362
363        if in_quotes {
364            return Err(IoError::Parse {
365                line: start_line,
366                msg: "unterminated quoted field".to_owned(),
367            });
368        }
369
370        let value = match owned {
371            Some(v) => {
372                let s = String::from_utf8(v).map_err(|_| utf8_error(start_line))?;
373                match s.trim() {
374                    t if t.len() == s.len() => Cow::Owned(s),
375                    t => Cow::Owned(t.to_owned()),
376                }
377            }
378            None => {
379                let s =
380                    std::str::from_utf8(&buf[start..end]).map_err(|_| utf8_error(start_line))?;
381                Cow::Borrowed(s.trim())
382            }
383        };
384        Ok((value, stop))
385    }
386
387    /// Skip whatever is left of the current record, terminator included.
388    fn drain_record(&mut self, buf: &[u8], delim: u8) -> Result<(), IoError> {
389        loop {
390            let (_, stop) = self.field(buf, delim)?;
391            match stop {
392                Stop::Field => continue,
393                Stop::Record => {
394                    self.eat_terminator(buf);
395                    return Ok(());
396                }
397                Stop::Eof => return Ok(()),
398            }
399        }
400    }
401
402    /// Skip one whole record, blank lines first. This is `skip_first`
403    /// (`graph_tool/__init__.py:3838-3839`), which discards the line without
404    /// looking at how many columns it has.
405    fn skip_record(&mut self, buf: &[u8], delim: u8) -> Result<(), IoError> {
406        while matches!(buf.get(self.pos), Some(&b'\n') | Some(&b'\r')) {
407            self.eat_terminator(buf);
408        }
409        if self.pos >= buf.len() {
410            return Ok(());
411        }
412        self.drain_record(buf, delim)
413    }
414
415    /// The next non-blank record, or `None` at end of input.
416    fn next_record<'a>(&mut self, buf: &'a [u8], delim: u8) -> Result<Option<Record<'a>>, IoError> {
417        loop {
418            if self.pos >= buf.len() {
419                return Ok(None);
420            }
421            if matches!(buf[self.pos], b'\n' | b'\r') {
422                self.eat_terminator(buf);
423                continue;
424            }
425            let line = self.line;
426            let (source, stop) = self.field(buf, delim)?;
427            if stop != Stop::Field {
428                if stop == Stop::Record {
429                    self.eat_terminator(buf);
430                }
431                // A line of nothing but whitespace is blank, not malformed.
432                if source.is_empty() {
433                    continue;
434                }
435                return Err(too_few(line));
436            }
437            let (target, stop) = self.field(buf, delim)?;
438            match stop {
439                Stop::Field => self.drain_record(buf, delim)?,
440                Stop::Record => self.eat_terminator(buf),
441                Stop::Eof => {}
442            }
443            return Ok(Some(Record {
444                line,
445                source,
446                target,
447            }));
448        }
449    }
450}

452fn utf8_error(line: usize) -> IoError {
453    IoError::Parse {
454        line,
455        msg: "field is not valid UTF-8".to_owned(),
456    }

473/// Write a view as a delimited edge list, in
474/// [`EdgeId`](gt_core::ids::EdgeId) order.
475///
476/// Option (b) of the crate's ordering contract: `AdjList` removes adjacency
477/// entries by swapping with the back of the half, so `EdgeList::edges` yields
478/// an order that depends on the removal history, while `EdgeId` does not. One
479/// sort of `num_edges()` ids buys a byte-reproducible file.
480///
481/// Endpoints are written as vertex *indices*, `opts.symbolic` notwithstanding:
482/// there is no name map in this signature to write instead, and inventing one
483/// from the indices would produce a file that reads back as a different graph.
484/// `opts.header` prepends `source<delim>target`, so a file written with it set
485/// reads back with it set.
486///
487/// Isolated vertices are not representable in this format and are not written;
488/// a round trip through it preserves the edge set and the vertex *count* only
489/// as far as the largest endpoint. That is the format's limitation, and it is
490/// `load_graph_from_csv`'s too.
491pub fn write_edge_list<W: Write, G>(w: W, g: G, opts: EdgeListOptions) -> Result<(), IoError>
492where
493    G: GraphRef + VertexList + EdgeList,
494{
495    let mut out = BufWriter::new(w);
496    if opts.header {
497        out.write_all(b"source")?;
498        out.write_all(&[opts.delimiter])?;
499        out.write_all(b"target\n")?;
500    }
501
502    let mut edges: Vec<_> = Vec::with_capacity(g.num_edges());
503    edges.extend(g.edges().map(|e| (e.id().index(), e.source(), e.target())));
504    // Ids are unique, so this is a total order and the unstable sort is
505    // deterministic.
506    edges.sort_unstable_by_key(|&(id, _, _)| id);
507
508    // One reusable scratch buffer: the steady-state inner loop allocates
509    // nothing.
510    let mut line = Vec::with_capacity(32);
511    for (_, s, t) in edges {
512        line.clear();
513        write_index(&mut line, s.index());
514        line.push(opts.delimiter);
515        write_index(&mut line, t.index());
516        line.push(b'\n');
517        out.write_all(&line)?;
518    }
519    out.flush()?;
520    Ok(())
521}
522
523/// Decimal, appended. `write!` would go through `fmt::Arguments` and a
524/// formatting machine this does not need.
525fn write_index(out: &mut Vec<u8>, mut n: usize) {
526    let mut digits = [0u8; 20];
527    let mut i = digits.len();
528    loop {
529        i -= 1;
530        digits[i] = b'0' + (n % 10) as u8;
531        n /= 10;
532        if n == 0 {
533            break;
534        }
535    }
536    out.extend_from_slice(&digits[i..]);
537}
