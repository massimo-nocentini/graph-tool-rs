1//! The native binary `.gt` format.
2//!
3//! Ports `src/graph/graph_io_binary.hh` in full. The layout, byte for byte:
4//!
5//! ```text
6//! "\u{26fe} gt"          6 bytes, `_magic` (`graph_io_binary.hh:30-31`)
7//! version               u8, `_version == 1` (`:32`)
8//! big_endian            u8, `is_bigendian()` of the *writing* host (`:447`)
9//! comment               string (`:449-456`)
10//! directed              u8  \
11//! N                     u64  |  `write_adjacency` (`:186-202`)
12//! N x vector<Vint>           /  `Vint` is the narrowest of u8/u16/u32/u64 that holds N-1
13//! nprops                u64 (`:459-460`)
14//! nprops x property          `write_property` (`:367-382`)
15//! ```
16//!
17//! where a *string* is a `u64` length followed by that many bytes (`:71-76`),
18//! a *vector* is a `u64` count followed by that many fixed-width elements
19//! (`:62-68`), and a *property* is a domain byte (`:248-253`), a name string,
20//! a value-type index byte, and then one value per element of the domain's
21//! range -- one for a graph property, `N` for a vertex property, `num_edges`
22//! for an edge property.
23//!
24//! ## Endianness
25//!
26//! Every multi-byte field after the `big_endian` flag is in the byte order
27//! that flag declares; [`write`] emits the host's own order and sets the flag
28//! to match, exactly as `write_graph` does (`:447`). The *comment*, however,
29//! is read by the C++ with `read<false>` (`:534`) -- that is, always as
30//! little-endian -- while it is written with the host's order (`:456`), so a
31//! big-endian build of graph-tool writes a header it cannot itself read. This
32//! port reads the comment's length in the order the flag declares, which is
33//! identical on every little-endian host and correct on the others.
34//!
35//! ## `python::object`
36//!
37//! The C++ serialises the Python member as `lexical_cast<std::string>(v)`
38//! (`:86-90`), and that cast is `object_pickler` (`graph_io.cc:62-68`), which
39//! is `pickle.dump(obj, sstream, -1)` (`graph_tool/gt_io.py:63-67`). So a
40//! `python::object` column *in the file* is a column of opaque pickle
41//! payloads, and nothing about reading or writing one needs an interpreter.
42//! [`PropertyColumn::PyObject`] carries those payloads as bytes, for the same
43//! reason [`LongDouble`] carries sixteen: the bytes round-trip unchanged
44//! whether or not this build has the `python` feature, and unpickling is the
45//! boundary layer's business, not the format's.
46//!
47//! ## Ordering
48//!
49//! Edge property values are positional: the reader binds the `i`-th value to
50//! the `i`-th edge of `edges_range(g)` (`:398-399`), so the adjacency section
51//! and every edge property column must agree on one traversal order. This port
52//! uses *out-edges of each vertex in ascending vertex order, and within a
53//! vertex in ascending [`EdgeId`](gt_core::ids::EdgeId) order*. On a graph
54//! that has never had an edge removed that is byte-for-byte graph-tool's own
55//! order; after removals it is the port's, because `AdjList` removes by
56//! swapping with the back of the half where `graph_adjacency.hh:1257-1263`
57//! erases and shifts ([DESIGN](gt_core::design) defect #52). Sorting makes the output a
58//! function of the graph rather than of its removal history.
59

69/// `_magic` (`graph_io_binary.hh:30`): U+26FE SESQUIQUADRATE, a space, `gt`.
70pub const MAGIC: &[u8] = "⛾ gt".as_bytes();

73// and `strncmp(magic, _magic, _magic_length)` (`:524`) trusts it. Here the

75// time, so the two cannot drift apart.

81/// Number of elements a reader will reserve before it has seen the bytes.
82///
83/// `read(std::istream&, std::vector<T>&)` (`:107-117`) does `v.resize(size)`
84/// on a length it has not validated, so a corrupt or truncated file asks for
85/// that allocation before the stream can refuse it. Growth here is driven by
86/// the bytes that actually arrive; this is only the head start.
87const RESERVE_CAP: usize = 1024;

89/// A property map read from, or to be written to, a file.
90///
91/// Named, typed and *valued*: the reader reconstructs a whole column, so the
92/// caller does not have to declare the concrete
93/// [`DenseProp`](gt_core::prop::DenseProp) in advance and does not have to
94/// re-read the stream to fill it.
95#[derive(Clone, Debug, PartialEq)]
96pub struct NamedProperty {
97    /// The property's name, as the Python layer sees it.
98    pub name: String,
99    /// Which identifier space it is keyed on.
100    pub domain: PropertyDomain,
101    /// The values, one per element of `domain`'s range.
102    ///
103    /// A vertex column is indexed by [`VertexId`](gt_core::ids::VertexId), an
104    /// edge column by [`EdgeId`](gt_core::ids::EdgeId) -- *not* by position in
105    /// the file's traversal order. A graph column holds exactly one value.
106    pub values: PropertyColumn,
107}
108
109impl NamedProperty {
110    /// Which member of the value universe this column stores.
111    ///
112    /// A method rather than a field: `value_types` and `type_names[]`
113    /// (`graph_properties.hh:61-76`) are two declarations held in
114    /// correspondence by position and by nothing else, and a `kind` field
115    /// beside a `values` field is the same shape of mistake one level down --
116    /// it can disagree with the payload it describes, and the writer would
117    /// then emit a type byte that the values do not match.
118    #[inline]
119    pub fn kind(&self) -> ValueKind {
120        self.values.kind()
121    }
122}

124/// Which identifier space a property map is keyed on.
125///
126/// The `property_type` enum (`graph_io_binary.hh:248-253`); the discriminants

130pub enum PropertyDomain {

272/// A Rust type that can be a `.gt` property column.
273///
274/// The mapping is [`PropValue`](gt_core::prop::PropValue)'s, member for
275/// member, with one exception it cannot express: `python::object` and
276/// `vector<bool>` are both `Vec<Vec<u8>>` here, because this crate stores the
277/// Python member as its pickle payload rather than as a live handle. `Vec<u8>`
278/// therefore means `vector<bool>`, and a `python::object` column is built
279/// with [`PropertyColumn::PyObject`] directly.
280pub trait ColumnValue: Sized {
281    /// The member this type occupies.
282    const KIND: ValueKind;
283    /// Wrap a column.
284    fn into_column(values: Vec<Self>) -> PropertyColumn;
285    /// Borrow a column of this type, or `None` if it stores another member.
286    fn column_slice(column: &PropertyColumn) -> Option<&[Self]>;
287}
288
289macro_rules! impl_column_value {
290    ($($t:ty => $v:ident),+ $(,)?) => {$(
291        impl ColumnValue for $t {
292            const KIND: ValueKind = ValueKind::$v;
293            #[inline]
294            fn into_column(values: Vec<Self>) -> PropertyColumn {
295                PropertyColumn::$v(values)
296            }
297            #[inline]
298            fn column_slice(column: &PropertyColumn) -> Option<&[Self]> {
299                match column {
300                    PropertyColumn::$v(v) => Some(v),
301                    _ => None,
302                }
303            }
304        }
305    )+};
306}

318    Vec<i64> => VecI64,
319    Vec<f64> => VecF64,

324/// Everything a `.gt` file holds.
325#[derive(Clone, Debug)]
326pub struct Document<H: Lookup> {
327    /// The graph.
328    pub graph: AdjList<H>,
329    /// Whether the file declares the graph directed.
330    ///
331    /// Storage is always the directed adjacency: `write_to_file` sets
332    /// `_directed = true` for the duration of the write (`graph_io.cc:507`)
333    /// so that an undirected graph still emits each edge once, from its
334    /// stored source, and records the flag separately.
335    pub directed: bool,
336    /// The property maps, erased.
337    pub properties: Vec<NamedProperty>,
338    /// The header comment, verbatim.
339    ///
340    /// `None` on write means "generate the one this version of graph-tool
341    /// would" (`graph_io_binary.hh:449-456`). [`read`] fills it with what the
342    /// file actually said, which is the only way a file written by an older
343    /// graph-tool can be re-saved byte-identically: the comment embeds the
344    /// writing version's own text, and the shipped collection (for instance
345    /// `karate.gt.gz`) still carries 2.2.32dev's wording.
346    pub comment: Option<String>,
347}
348
349impl<H: Lookup> Document<H> {
350    /// A document with no properties and no comment.
351    pub fn new(graph: AdjList<H>, directed: bool) -> Self {
352        Document {
353            graph,
354            directed,
355            properties: Vec::new(),
356            comment: None,
357        }
358    }
359
360    /// `write_graph`'s comment (`graph_io_binary.hh:449-456`), regenerated.
361    fn generated_comment(&self) -> String {
362        let (ng, nv, ne) = self.domain_counts();
363        format!(
364            "graph-tool binary file version {VERSION} (https://graph-tool.skewed.de), stats: {} vertices, {} edges, {}, {ng} graph props, {nv} vertex props, {ne} edge props",
365            self.graph.num_vertices(),
366            self.graph.num_edges(),
367            if self.directed {
368                "directed"
369            } else {
370                "undirected"
371            },
372        )
373    }
374
375    fn domain_counts(&self) -> (usize, usize, usize) {
376        let mut c = (0usize, 0usize, 0usize);
377        for p in &self.properties {
378            match p.domain {
379                PropertyDomain::Graph => c.0 += 1,
380                PropertyDomain::Vertex => c.1 += 1,
381                PropertyDomain::Edge => c.2 += 1,
382            }
383        }
384        c
385    }
386}

392/// The one shape of syntax error a binary format has.
393///
394/// `Parse::line` is a text-format concept, so it is zero here and the byte
395/// offset goes in the message. graph-tool signals the same condition by
396/// arming the stream with `failbit | eofbit` (`graph_io.cc:346`) and catching
397/// `ios_base::failure` two frames up, which loses the offset entirely.
398fn corrupt(at: u64, msg: impl std::fmt::Display) -> IoError {
399    IoError::Parse {
400        line: 0,
401        msg: format!("at byte {at}: {msg}"),
402    }
403}
404
405fn from_graph_error(e: GraphError) -> IoError {
406    match e {
407        GraphError::VertexIdSpaceExhausted { .. } | GraphError::EdgeIdSpaceExhausted { .. } => {
408            IoError::IndexWidthExceeded
409        }
410        other => IoError::Parse {
411            line: 0,
412            msg: other.to_string(),
413        },
414    }
415}

421/// A counted, endian-aware byte source.
422struct Source<R: Read> {
423    r: R,
424    /// Byte offset, for diagnostics. graph-tool reports none.
425    pos: u64,
426    /// What the file's `big_endian` flag said.
427    big_endian: bool,
428}
429
430/// `byte_swap<BE>` (`graph_io_binary.hh:45-52`): a no-op when the file's
431/// declared order is the host's, a full reversal of the field otherwise.
432macro_rules! read_scalar {
433    ($name:ident, $t:ty) => {
434        #[inline]
435        fn $name(&mut self) -> Result<$t, IoError> {
436            let mut b = [0u8; size_of::<$t>()];
437            self.exact(&mut b)?;
438            Ok(if self.big_endian {
439                <$t>::from_be_bytes(b)
440            } else {
441                <$t>::from_le_bytes(b)
442            })
443        }
444    };
445}
446
447impl<R: Read> Source<R> {
448    fn new(r: R) -> Self {
449        Source {
450            r,
451            pos: 0,
452            big_endian: false,
453        }
454    }
455
456    /// Fill `buf` or fail with [`IoError::Parse`].
457    ///
458    /// `read_exact`'s `UnexpectedEof` is remapped rather than forwarded: a
459    /// truncated file is a malformed file, not an I/O failure, and the
460    /// acceptance criterion for this unit is that truncation is diagnosed and
461    /// never panics.
462    fn exact(&mut self, buf: &mut [u8]) -> Result<(), IoError> {
463        match self.r.read_exact(buf) {
464            Ok(()) => {
465                self.pos += buf.len() as u64;
466                Ok(())
467            }
468            Err(e) if e.kind() == std::io::ErrorKind::UnexpectedEof => Err(corrupt(
469                self.pos,
470                format_args!("unexpected end of file, wanted {} more bytes", buf.len()),
471            )),
472            Err(e) => Err(IoError::Io(e)),
473        }
474    }
475
476    #[inline]
477    fn u8(&mut self) -> Result<u8, IoError> {
478        let mut b = [0u8; 1];
479        self.exact(&mut b)?;
480        Ok(b[0])
481    }
482
483    read_scalar!(i16, i16);
484    read_scalar!(i32, i32);
485    read_scalar!(i64, i64);
486    read_scalar!(u16, u16);
487    read_scalar!(u32, u32);
488    read_scalar!(u64, u64);
489    read_scalar!(f64, f64);
490
491    /// `long double` is carried opaquely, so the "swap" is the whole field
492    /// reversed -- which is what `byte_swap<BE>` does to it (`:50-51`).
493    fn long_double(&mut self) -> Result<LongDouble, IoError> {
494        let mut b = [0u8; 16];
495        self.exact(&mut b)?;
496        if self.big_endian != cfg!(target_endian = "big") {
497            b.reverse();
498        }
499        Ok(LongDouble(b))
500    }
501
502    /// A length-prefixed byte run (`:127-134`).
503    ///
504    /// The length is *not* trusted into an allocation: growth follows the
505    /// bytes that arrive, so a corrupt length yields a parse error rather than
506    /// the `v.resize(size)` the C++ performs first (`:131`).
507    fn bytes(&mut self) -> Result<Vec<u8>, IoError> {
508        let n = self.u64()?;
509        let at = self.pos;
510        let mut out = Vec::new();
511        let got = self
512            .r
513            .by_ref()
514            .take(n)
515            .read_to_end(&mut out)
516            .map_err(IoError::Io)?;
517        self.pos += got as u64;
518        if got as u64 != n {
519            return Err(corrupt(
520                at,
521                format_args!("string declares {n} bytes, {got} are present"),
522            ));
523        }
524        Ok(out)
525    }
526
527    fn string(&mut self, what: &str) -> Result<String, IoError> {
528        let at = self.pos;
529        let b = self.bytes()?;
530        String::from_utf8(b).map_err(|e| {
531            corrupt(
532                at,
533                format_args!(
534                    "{what} is not valid UTF-8 at offset {}",
535                    e.utf8_error().valid_up_to()
536                ),
537            )
538        })
539    }
540
541    /// The element count of a vector field, with the reservation capped.
542    fn count(&mut self) -> Result<(usize, usize), IoError> {
543        let n = self.u64()?;
544        let n = usize::try_from(n).map_err(|_| {
545            corrupt(
546                self.pos,
547                format_args!("vector of {n} exceeds the address space"),
548            )
549        })?;
550        Ok((n, n.min(RESERVE_CAP)))
551    }
552}

554/// Read a `.gt` stream.
555pub fn read<R: Read, H: Lookup>(r: R, lookup: H) -> Result<Document<H>, IoError> {
556    let mut src = Source::new(BufReader::new(r));
557
558    // `strncmp(magic, _magic, _magic_length) != 0` (`:524`). A short read is
559    // reported as a bad magic rather than as truncation: it is the first
560    // thing in the file, so "not a .gt file" is the better diagnosis.
561    let mut magic = [0u8; 6];
562    let mut got = 0;
563    while got < magic.len() {
564        match src.r.read(&mut magic[got..]).map_err(IoError::Io)? {
565            0 => break,
566            n => got += n,
567        }
568    }
569    src.pos += got as u64;
570    if got != magic.len() || magic != *MAGIC {
571        return Err(IoError::BadMagic {
572            expected: MAGIC,
573            found: magic[..got].to_vec(),
574        });
575    }
576
577    let version = src.u8()?;
578    if version != VERSION {
579        return Err(IoError::UnsupportedVersion(version));
580    }
581    // `read<false>` on a one-byte field (`:531-532`): the swap is a no-op, so
582    // the flag is readable before the order it declares is known.
583    src.big_endian = src.u8()? != 0;
584    let comment = src.string("the header comment")?;
585
586    let (graph, directed, num_edges) = read_adjacency(&mut src, lookup)?;
587
588    let nprops = src.u64()?;
589    let mut properties = Vec::new();
590    for _ in 0..nprops {
591        let at = src.pos;
592        let byte = src.u8()?;
593        let domain = PropertyDomain::from_byte(byte)
594            .ok_or_else(|| corrupt(at, format_args!("invalid property domain {byte}")))?;
595        let name = src.string("a property name")?;
596        let at = src.pos;
597        let index = src.u8()?;
598        // `val_types` is `value_types` with `size_t` appended (`:257-258`), so
599        // index 15 is a dispatch-table artefact that `write_property_dispatch`
600        // never emits -- its `size_t` overload writes `int64_t`'s index
601        // instead (`:335`). Rejecting it here costs nothing a real file uses.
602        let kind = ValueKind::ALL
603            .get(index as usize)
604            .copied()
605            .ok_or_else(|| IoError::UnknownValueType(format!("value type index {index}")))?;
606        let n = match domain {
607            PropertyDomain::Graph => 1,
608            PropertyDomain::Vertex => graph.num_vertices(),
609            PropertyDomain::Edge => num_edges,
610        };
611        let values = read_column(&mut src, kind, n, &name)?;
612        properties.push(NamedProperty {
613            name,
614            domain,
615            values,
616        });
617    }
618
619    Ok(Document {
620        graph,
621        directed,
622        properties,
623        comment: Some(comment),
624    })
625}

627/// `read_adjacency` (`:223-244`), returning the edge count as well.
628fn read_adjacency<R: Read, H: Lookup>(
629    src: &mut Source<R>,
630    lookup: H,
631) -> Result<(AdjList<H>, bool, usize), IoError> {
632    let directed = src.u8()? != 0;
633    let n = src.u64()?;
634    let n = usize::try_from(n).map_err(|_| IoError::IndexWidthExceeded)?;
635    if n > 0 && n - 1 > MAX_INDEX {
636        return Err(IoError::IndexWidthExceeded);
637    }
638
639    let mut g = AdjList::with_lookup(lookup);
640    let width = vint_width(n as u64);
641    let mut edges = 0usize;
642
643    // `add_vertex(g, N)` (`:232`) allocates all N blocks before a single
644    // adjacency byte is read, so a file whose N is corrupt asks for the
645    // allocation first and discovers the truncation second. Growing to the
646    // highest index actually mentioned, and only then to N, makes the two
647    // happen in the other order; for a well-formed file the result is
648    // identical, because every list is read and N blocks exist at the end.
649    for v in 0..n {
650        let (k, _) = src.count()?;
651        grow_to(&mut g, v + 1)?;
652        let v = VertexId::from_index(v);
653        for _ in 0..k {
654            let u = read_vint(src, width)?;
655            // `:215-217`, verbatim, including the bound being N and not the
656            // number of vertices created so far.
657            if u >= n as u64 {
658                return Err(IoError::IndexOutOfRange {
659                    kind: "vertex",
660                    index: u,
661                    bound: n as u64,
662                });
663            }
664            let u = VertexId::from_index(u as usize);
665            grow_to(&mut g, u.index() + 1)?;
666            let e = g.add_edge(v, u).map_err(from_graph_error)?;
667            // The reader binds the i-th edge property value to the i-th edge
668            // of this traversal, so the traversal index and the edge id must
669            // agree. On a graph built only by `add_edge` they do, because the
670            // free list is empty and `alloc` hands out 0, 1, 2, ...
671            debug_assert_eq!(e.id().index(), edges);
672            edges += 1;
673        }
674    }
675    grow_to(&mut g, n)?;
676
677    Ok((g, directed, edges))
678}
679
680fn grow_to<H: Lookup>(g: &mut AdjList<H>, n: usize) -> Result<(), IoError> {
681    while g.num_vertices() < n {
682        g.add_vertex().map_err(from_graph_error)?;
683    }
684    Ok(())
685}

687/// The `Vint` dispatch of `write_adjacency`/`read_adjacency` (`:194-201`,
688/// `:234-241`), as a width in bytes.
689const fn vint_width(n: u64) -> u8 {
690    if n <= u8::MAX as u64 {
691        1
692    } else if n <= u16::MAX as u64 {
693        2
694    } else if n <= u32::MAX as u64 {
695        4
696    } else {
697        8
698    }
699}
700
701fn read_vint<R: Read>(src: &mut Source<R>, width: u8) -> Result<u64, IoError> {
702    Ok(match width {
703        1 => u64::from(src.u8()?),
704        2 => u64::from(src.u16()?),
705        4 => u64::from(src.u32()?),
706        _ => src.u64()?,
707    })
708}

710/// A property payload that declares `string` but is not UTF-8.
711///
712/// `std::string` is a byte string, so the C++ accepts anything here; the value
713/// universe spells the member `String` (`prop/value.rs`), so this port has to
714/// say so rather than silently substitute. Truncation is *not* routed through
715/// here -- that is a `Parse`, raised by `Source::bytes` before this is called.
716fn utf8(b: Vec<u8>, name: &str, declared: ValueKind) -> Result<String, IoError> {
717    String::from_utf8(b).map_err(|_| IoError::MalformedProperty {
718        name: name.to_owned(),
719        declared,
720    })
721}

723/// Read `n` values of one member.
724fn read_column<R: Read>(
725    src: &mut Source<R>,
726    kind: ValueKind,
727    n: usize,
728    name: &str,
729) -> Result<PropertyColumn, IoError> {
730    /// `for (auto x : range) read<BE>(s, prop[x])` (`:398-399`), with the
731    /// reservation capped at [`RESERVE_CAP`].
732    macro_rules! column {
733        ($variant:ident, $read:expr) => {{
734            let mut out = Vec::with_capacity(n.min(RESERVE_CAP));
735            for _ in 0..n {
736                out.push($read?);
737            }
738            PropertyColumn::$variant(out)
739        }};
740    }
741
742    /// A vector-valued member: a count and then that many elements (`:107-117`).
743    macro_rules! vector_column {
744        ($variant:ident, $elem:ident) => {{
745            let mut out = Vec::with_capacity(n.min(RESERVE_CAP));
746            for _ in 0..n {
747                let (k, cap) = src.count()?;
748                let mut inner = Vec::with_capacity(cap);
749                for _ in 0..k {
750                    inner.push(src.$elem()?);
751                }
752                out.push(inner);
753            }
754            PropertyColumn::$variant(out)
755        }};
756    }
757
758    Ok(match kind {
759        ValueKind::Bool => column!(Bool, src.u8()),
760        ValueKind::I16 => column!(I16, src.i16()),
761        ValueKind::I32 => column!(I32, src.i32()),
762        ValueKind::I64 => column!(I64, src.i64()),
763        ValueKind::F64 => column!(F64, src.f64()),
764        ValueKind::LongDouble => column!(LongDouble, src.long_double()),
765        ValueKind::Str => {
766            let mut out = Vec::with_capacity(n.min(RESERVE_CAP));
767            for _ in 0..n {
768                out.push(utf8(src.bytes()?, name, ValueKind::Str)?);
769            }
770            PropertyColumn::Str(out)
771        }
772        ValueKind::VecBool => {
773            // `vector<uint8_t>` is a byte run, so it takes the bulk path
774            // rather than one `read_exact` per element.
775            let mut out = Vec::with_capacity(n.min(RESERVE_CAP));
776            for _ in 0..n {
777                out.push(src.bytes()?);
778            }
779            PropertyColumn::VecBool(out)
780        }
781        ValueKind::VecI16 => vector_column!(VecI16, i16),
782        ValueKind::VecI32 => vector_column!(VecI32, i32),
783        ValueKind::VecI64 => vector_column!(VecI64, i64),
784        ValueKind::VecF64 => vector_column!(VecF64, f64),
785        ValueKind::VecLongDouble => vector_column!(VecLongDouble, long_double),
786        ValueKind::VecStr => {
787            let mut out = Vec::with_capacity(n.min(RESERVE_CAP));
788            for _ in 0..n {
789                let (k, cap) = src.count()?;
790                let mut inner = Vec::with_capacity(cap);
791                for _ in 0..k {
792                    inner.push(utf8(src.bytes()?, name, ValueKind::VecStr)?);
793                }
794                out.push(inner);
795            }
796            PropertyColumn::VecStr(out)
797        }
798        // A pickle payload, kept opaque: `read(s, boost::python::object&)`
799        // (`:156-162`) reads a `std::string` and then unpickles it, and the
800        // unpickling is the boundary layer's business.
801        ValueKind::PyObject => {
802            let mut out = Vec::with_capacity(n.min(RESERVE_CAP));
803            for _ in 0..n {
804                out.push(src.bytes()?);
805            }
806            PropertyColumn::PyObject(out)
807        }
808    })
809}

815/// A byte sink that writes the host's own order, as `write` does (`:55-59`).
816struct Sink<W: Write> {
817    w: W,
818}
819
820macro_rules! write_scalar {
821    ($name:ident, $t:ty) => {
822        #[inline]
823        fn $name(&mut self, v: $t) -> Result<(), IoError> {
824            self.raw(&v.to_ne_bytes())
825        }
826    };
827}
828
829impl<W: Write> Sink<W> {
830    #[inline]
831    fn raw(&mut self, b: &[u8]) -> Result<(), IoError> {
832        self.w.write_all(b).map_err(IoError::Io)
833    }
834
835    #[inline]
836    fn u8(&mut self, v: u8) -> Result<(), IoError> {
837        self.raw(&[v])
838    }
839
840    write_scalar!(i16, i16);
841    write_scalar!(i32, i32);
842    write_scalar!(i64, i64);
843    write_scalar!(u16, u16);
844    write_scalar!(u32, u32);
845    write_scalar!(u64, u64);
846    write_scalar!(f64, f64);
847
848    #[inline]
849    fn long_double(&mut self, v: LongDouble) -> Result<(), IoError> {
850        self.raw(&v.0)
851    }
852
853    /// A length-prefixed byte run (`:71-76`).
854    fn bytes(&mut self, b: &[u8]) -> Result<(), IoError> {
855        self.u64(b.len() as u64)?;
856        self.raw(b)
857    }
858}

860/// Write a `.gt` stream.
861///
862/// Edges are emitted in [`EdgeId`](gt_core::ids::EdgeId) order within each
863/// vertex's out-half, not in adjacency order, so the output does not depend on
864/// the removal history. See the module docs.
865pub fn write<W: Write, H: Lookup>(w: W, doc: &Document<H>) -> Result<(), IoError> {
866    // One buffer for many small fields; the C++ writes into a
867    // `filtering_stream`, which is buffered for the same reason.
868    let mut sink = Sink {
869        w: BufWriter::with_capacity(1 << 16, w),
870    };
871
872    sink.raw(MAGIC)?;
873    sink.u8(VERSION)?;
874    sink.u8(u8::from(cfg!(target_endian = "big")))?;
875    match &doc.comment {
876        Some(c) => sink.bytes(c.as_bytes())?,
877        None => sink.bytes(doc.generated_comment().as_bytes())?,
878    }
879
880    let order = write_adjacency(&mut sink, doc)?;
881
882    let (ng, nv, ne) = doc.domain_counts();
883    sink.u64((ng + nv + ne) as u64)?;
884    // Grouped by domain, in graph/vertex/edge order, preserving the relative
885    // order within each group: `write_graph` walks three separate vectors
886    // (`:461-466`), and a file graph-tool wrote is therefore already grouped.
887    for domain in [
888        PropertyDomain::Graph,
889        PropertyDomain::Vertex,
890        PropertyDomain::Edge,
891    ] {
892        for p in doc.properties.iter().filter(|p| p.domain == domain) {
893            write_property(&mut sink, doc, p, &order)?;
894        }
895    }
896
897    // `BufWriter` swallows a write error in its own `Drop`; the C++ arms the
898    // stream with `failbit` (`graph_io.cc:485`) so that the same condition
899    // throws. Flushing explicitly is what turns it back into a `Result`.
900    sink.w.flush().map_err(IoError::Io)
901}

903/// `write_adjacency` (`:186-202`), returning the canonical edge order so that
904/// every edge property column can be emitted against it.
905fn write_adjacency<W: Write, H: Lookup>(
906    sink: &mut Sink<W>,
907    doc: &Document<H>,
908) -> Result<Vec<usize>, IoError> {
909    let g = &doc.graph;
910    let n = g.num_vertices();
911    sink.u8(u8::from(doc.directed))?;
912    sink.u64(n as u64)?;
913
914    let width = vint_width(n as u64);
915    let mut order = Vec::with_capacity(g.num_edges());
916    // Reused across vertices: a per-vertex `Vec` would be one allocation per
917    // vertex on a path that runs once per file.
918    let mut scratch: Vec<Incident> = Vec::new();
919
920    for v in g.vertices() {
921        scratch.clear();
922        scratch.extend(g.out_edges(v));
923        // Already sorted on a graph that has never had an edge removed, which
924        // is the case that has to stay byte-identical to graph-tool's output.
925        scratch.sort_unstable_by_key(|i| i.edge);
926        sink.u64(scratch.len() as u64)?;
927        for i in &scratch {
928            let u = i.other.index() as u64;
929            match width {
930                1 => sink.u8(u as u8)?,
931                2 => sink.u16(u as u16)?,
932                4 => sink.u32(u as u32)?,
933                _ => sink.u64(u)?,
934            }
935            order.push(i.edge.index());
936        }
937    }
938    Ok(order)
939}

941/// `write_property` (`:367-382`).
942fn write_property<W: Write, H: Lookup>(
943    sink: &mut Sink<W>,
944    doc: &Document<H>,
945    p: &NamedProperty,
946    order: &[usize],
947) -> Result<(), IoError> {
948    // Checked before a byte of this property reaches the sink: a refused
949    // write should not leave half a property header in the stream.
950    //
951    // A vertex column is keyed by vertex index and a graph column holds one
952    // value; an edge column is keyed by `EdgeId`, and the file's positional
953    // order is therefore the traversal `order`, not `0..num_edges`.
954    let need = match p.domain {
955        PropertyDomain::Graph => 1,
956        PropertyDomain::Vertex => doc.graph.num_vertices(),
957        // The *bound*, not the count: after a removal the live ids are sparse,
958        // which is exactly why `_get_any` sizes from `edge_index_range`
959        // (`graph_tool/__init__.py:369`).
960        PropertyDomain::Edge => doc.graph.edge_bound().len(),
961    };
962    if p.values.len() < need {
963        return Err(IoError::ShortProperty {
964            name: p.name.clone(),
965            have: p.values.len(),
966            need,
967        });
968    }
969
970    sink.u8(p.domain as u8)?;
971    sink.bytes(p.name.as_bytes())?;
972    sink.u8(p.kind() as u8)?;
973
974    match p.domain {
975        PropertyDomain::Graph => write_values(sink, &p.values, std::iter::once(0)),
976        PropertyDomain::Vertex => write_values(sink, &p.values, 0..doc.graph.num_vertices()),
977        PropertyDomain::Edge => write_values(sink, &p.values, order.iter().copied()),
978    }
979}

981/// Emit `column[i]` for each `i`, in the order given.
982///
983/// Monomorphised over the index iterator rather than taking `&[usize]`, so the
984/// vertex case walks a range with no indirection and no `order` vector to
985/// build.
986fn write_values<W, I>(
987    sink: &mut Sink<W>,
988    column: &PropertyColumn,
989    indices: I,
990) -> Result<(), IoError>
991where
992    W: Write,
993    I: Iterator<Item = usize>,
994{
995    /// One scalar per element.
996    macro_rules! scalars {
997        ($v:expr, $put:ident) => {{
998            for i in indices {
999                sink.$put($v[i])?;
1000            }
1001        }};
1002    }
1003    /// A count and then that many elements (`:62-68`).
1004    macro_rules! vectors {
1005        ($v:expr, $put:ident) => {{
1006            for i in indices {
1007                let inner = &$v[i];
1008                sink.u64(inner.len() as u64)?;
1009                for x in inner {
1010                    sink.$put(*x)?;
1011                }
1012            }
1013        }};
1014    }
1015
1016    match column {
1017        PropertyColumn::Bool(v) => scalars!(v, u8),
1018        PropertyColumn::I16(v) => scalars!(v, i16),
1019        PropertyColumn::I32(v) => scalars!(v, i32),
1020        PropertyColumn::I64(v) => scalars!(v, i64),
1021        PropertyColumn::F64(v) => scalars!(v, f64),
1022        PropertyColumn::LongDouble(v) => scalars!(v, long_double),
1023        PropertyColumn::Str(v) => {
1024            for i in indices {
1025                sink.bytes(v[i].as_bytes())?;
1026            }
1027        }
1028        // `vector<uint8_t>` and a pickle payload are both byte runs.
1029        PropertyColumn::VecBool(v) | PropertyColumn::PyObject(v) => {
1030            for i in indices {
1031                sink.bytes(&v[i])?;
1032            }
1033        }
1034        PropertyColumn::VecI16(v) => vectors!(v, i16),
1035        PropertyColumn::VecI32(v) => vectors!(v, i32),
1036        PropertyColumn::VecI64(v) => vectors!(v, i64),
1037        PropertyColumn::VecF64(v) => vectors!(v, f64),
1038        PropertyColumn::VecLongDouble(v) => vectors!(v, long_double),
1039        PropertyColumn::VecStr(v) => {
1040            for i in indices {
1041                let inner = &v[i];
1042                sink.u64(inner.len() as u64)?;
1043                for s in inner {
1044                    sink.bytes(s.as_bytes())?;
1045                }
1046            }
1047        }
1048    }
1049    Ok(())
1050}
